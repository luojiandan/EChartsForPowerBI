/**
 * C010 - IBCS C11 风格盈利能力指标关系树
 * 直接粘贴到 ECharts 官方编辑器 JS 面板；option 不用 var 声明。
 * 2021—2025 苹果官方财务数据，单位：百万美元。
 * 2020 年期末资产只用于计算 2021 年平均资产。
 * 参考： https://www.sec.gov/Archives/edgar/data/320193/000032019325000079/aapl-20250927.htm
 *         https://www.sec.gov/Archives/edgar/data/320193/000032019323000106/aapl-20230930.htm
 *         https://www.annualreports.com/HostedData/AnnualReportArchive/a/NASDAQ_AAPL_2021.pdf
 * 原理： ROA = 净利率 × 资产周转率
 * 注意：这是 IBCS C11 风格的复刻，并非官方认证图表。
 */

// ====================================================================
// 01 示例数据｜每个原始字段对应一维数组；空值用于期初辅助数据
// ====================================================================
var 年份 = ['2020','2021','2022','2023','2024','2025'];
var 营业收入 = [null,365817,394328,383285,391035,416161];
var 净利润 = [null,94680,99803,96995,93736,112010];
var 期末总资产 = [323888,351002,352755,352583,364980,359241];

// ====================================================================
// 02 用户设置｜所有可调项目放这里
// ====================================================================
var setting = {
  title:'苹果盈利能力为什么提升？',
  subtitle:'IBCS C11 风格 · 六图财务指标关系树 · 2021—2025 财年',
  entity:'Apple Inc.',
  moneyUnit:'亿美元',
  showMoneyUnitInCards:false,  // 货币卡片只显示数值，统一单位放在副标题
  valueDivisor:100,
  percentDigits:1,
  numberDigits:2,
  source:'数据来源：Apple 2021、2023、2025 Form 10-K（SEC / Apple 年报）',
  description:'图表说明：ROA=净利率×周转率；趋势采用首年=100的指数尺度（可切换）；红绿表示数值升降，非好坏；资产采用两期平均值。',
  layout: {
    left:26,right:26,top:144,bottom:76,
    gapX:44,gapY:18,
    columnRatios:[0.98,1.06,1.06],
    cardHeight:155,
    fallbackWidth:1280,fallbackHeight:770
  },
  colors: {
    positive:'#239263',negative:'#D9544D',
    current:'#263238',historical:'#AEB7C1',
    bar:'#BFC6CE',secondary:'#8493A2',line:'#E5E9ED',background:'transparent',white:'#FFFFFF'
  },
  showPointLabels:false,
  colorTrendSegments:true,         // 按相邻年度升降为趋势线段着色
  showLatestChange:true,           // 卡片右下角展示最新同比变化
  showYearAxis:false,             // 默认改为卡片内仅显示首末年度
  showEndpointYears:true,          // 在每张趋势图底部显示首末年份
  trendScale:'indexed',          // 'indexed': 六图统一2021=100基准；'local': 每图独立尺度
  showIndexBaseline:true,       // 首年=100参考虚线
  historicalLineOpacity:0.68,  // 历史线段弱化（最新段保持醒目）
  showROAAttribution:true,     // ROA的两因素对称分解（贡献单位：百分点）
  showOperators:true,          // 在连线上显示×、÷数学符号
  indexReference:100,          // 指数化的基准值
  showRelationships:true,
  symbolSize:7,
  lineWidth:2.3,
  animationDuration:550
};

// ====================================================================
// 03 数据处理与校验｜不绑定具体年度名称
// ====================================================================
if ([营业收入.length,净利润.length,期末总资产.length].some(function(n){return n!==年份.length;}))
  throw new Error('年份、营业收入、净利润、期末总资产必须为相同长度的一维数组');
function valid(n){return n!==null&&n!==undefined&&n!==''&&isFinite(Number(n));}
function nf(v,d){return Number(v).toLocaleString('zh-CN',{minimumFractionDigits:d,maximumFractionDigits:d});}
function closeEnough(a,b){return Math.abs(a-b)<1e-8*Math.max(1,Math.abs(b));}
var raw=年份.map(function(y,i){return {year:String(y),revenue:valid(营业收入[i])?Number(营业收入[i]):null,profit:valid(净利润[i])?Number(净利润[i]):null,asset:valid(期末总资产[i])?Number(期末总资产[i]):null};});
raw.sort(function(a,b){return Number(a.year)-Number(b.year);});
var records=[];
for(var i=1;i<raw.length;i++){
  var r=raw[i],p=raw[i-1];
  if(!valid(r.revenue)||!valid(r.profit)||!valid(r.asset)||!valid(p.asset))continue;
  var avg=(r.asset+p.asset)/2;
  if(!r.revenue||!avg)continue;
  var margin=r.profit/r.revenue,turnover=r.revenue/avg,roa=r.profit/avg;
  if(!closeEnough(roa,margin*turnover))throw new Error('ROA 勾稽失败 '+r.year);
  records.push({year:r.year,revenue:r.revenue,profit:r.profit,asset:avg,margin:margin*100,turnover:turnover,roa:roa*100});
}
if(records.length<2)throw new Error('至少需要两个有效财年，且前一年有期末总资产数据');
var latest=records[records.length-1],previous=records[records.length-2];
// ROA变化精确分解：m、t 均取原始比例/倍数，贡献换算为百分点。
// Δ(m×t) = Δm×(t0+t1)/2 + Δt×(m0+m1)/2，无残差。
function roaAttribution(before,after){
  var marginContribution=(after.margin-before.margin)*(before.turnover+after.turnover)/2;
  var turnoverContribution=(after.turnover-before.turnover)*(before.margin+after.margin)/2;
  var total=marginContribution+turnoverContribution;
  if(!closeEnough(total,after.roa-before.roa))throw new Error('ROA双因素归因计算未勾稽');
  return {margin:marginContribution,turnover:turnoverContribution,total:total};
}
var attribution=roaAttribution(previous,latest);
var C=setting.colors,L=setting.layout;
function metricFmt(key,v,forCard){
  return ['profit','revenue','asset'].indexOf(key)>=0?nf(v/setting.valueDivisor,setting.numberDigits)+(forCard&&!setting.showMoneyUnitInCards?'':' '+setting.moneyUnit):
    key==='turnover'?nf(v,setting.numberDigits)+' 次':nf(v,setting.percentDigits)+'%';
}
function metricChange(key){return latest[key]-previous[key];}
function changeFmt(key,d){
  var v=Math.abs(d);
  if(key==='turnover')return nf(v,setting.numberDigits)+' 次';
  if(['profit','revenue','asset'].indexOf(key)>=0)return nf(v/setting.valueDivisor,setting.numberDigits)+' '+setting.moneyUnit;
  return nf(v,setting.percentDigits)+' 个百分点';
}
function trendColor(d){return d>1e-8?C.positive:d< -1e-8?C.negative:C.secondary;}
function trendLabel(key,d){return (d>1e-8?'▲ +':d< -1e-8?'▼ −':'● ')+changeFmt(key,d);}

function changeText(key){var d=metricChange(key);return (d>0?'▲ ':d<0?'▼ ':'● ')+metricFmt(key,Math.abs(d)).replace(setting.moneyUnit,setting.moneyUnit);}
function html(s){return String(s).replace(/[&<>"']/g,function(c){return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];});}
var specs=[
 {key:'roa',name:'资产收益率',formula:'净利润 / 平均总资产',col:0,row:1},
 {key:'margin',name:'销售净利率',formula:'净利润 / 营业收入',col:1,row:0},
 {key:'turnover',name:'资产周转率',formula:'营业收入 / 平均总资产',col:1,row:2},
 {key:'profit',name:'净利润',formula:'Net Income',col:2,row:0},
 {key:'revenue',name:'营业收入',formula:'Total Net Sales',col:2,row:1},
 {key:'asset',name:'平均总资产',formula:'(期初 + 期末总资产) / 2',col:2,row:2}
];
var W=(typeof myChart!=='undefined'&&myChart&&myChart.getWidth)?myChart.getWidth():L.fallbackWidth;
var H=(typeof myChart!=='undefined'&&myChart&&myChart.getHeight)?myChart.getHeight():L.fallbackHeight;
W=Math.max(850,W||L.fallbackWidth);H=Math.max(600,H||L.fallbackHeight);
var totalW=W-L.left-L.right-2*L.gapX,sumRat=L.columnRatios.reduce(function(a,b){return a+b;},0);
var colWidths=L.columnRatios.map(function(v){return Math.round(v*totalW/sumRat);});
var colLeft=[L.left,L.left+colWidths[0]+L.gapX,L.left+colWidths[0]+L.gapX+colWidths[1]+L.gapX];
// 三行采用紧凑等距排布，保持关系树顺序，同时为折线留出足够高度。
var usableH=H-L.top-L.bottom;
var cardHeight=Math.min(L.cardHeight,Math.max(91,(usableH-2*L.gapY)/3));
var rowGap=Math.max(10,Math.min(36,(usableH-3*cardHeight)/2));
var blockHeight=3*cardHeight+2*rowGap;
var blockTop=L.top+Math.max(0,(usableH-blockHeight)/2);
var rowCenters=[0,1,2].map(function(i){return blockTop+cardHeight/2+i*(cardHeight+rowGap);});
var cards=specs.map(function(s){return {x:colLeft[s.col],y:rowCenters[s.row]-cardHeight/2,w:colWidths[s.col],h:cardHeight};});
function lineG(x1,y1,x2,y2,color,width){return {type:'line',silent:true,z:0,shape:{x1:x1,y1:y1,x2:x2,y2:y2},style:{stroke:color||C.secondary,lineWidth:width||1.3}};}
function textG(x,y,t,size,bold,color,align,width){return {type:'text',silent:true,z:20,style:{x:x,y:y,text:String(t),font:(bold?'bold ':'')+size+'px Microsoft YaHei, Arial',fill:color||C.current,align:align||'left',verticalAlign:'middle',width:width,overflow:'truncate'}};}
var graphics=[
 {type:'rect',silent:true,z:10,shape:{x:L.left,y:23,width:20,height:4},style:{fill:C.negative}},
 textG(L.left,54,setting.title,21,true,C.current),
 textG(L.left,86,setting.subtitle+' · 金额单位：'+setting.moneyUnit,12,false,C.secondary),
 lineG(L.left,109,W-L.right,109,C.line,1),
 textG(L.left,H-53,setting.source,10,false,C.secondary),
 textG(L.left,H-31,setting.description,10,false,C.secondary)
];
function connector(a,b,branch){
 var ac=cards[a],bc=cards[b],xa=ac.x+ac.w+7,xb=bc.x-7,ya=ac.y+ac.h/2,yb=bc.y+bc.h/2,mid=(xa+xb)/2;
 graphics.push(lineG(xa,ya,mid,ya,C.secondary,1.1));
 graphics.push(lineG(mid,Math.min(ya,yb),mid,Math.max(ya,yb),C.secondary,1.1));
 graphics.push(lineG(mid,yb,xb,yb,C.secondary,1.1));
}
if(setting.showRelationships){
 connector(0,1);connector(0,2);connector(1,3);connector(1,4);connector(2,4);connector(2,5);
 // 运算符是关系提示，不代替连线：ROA = 净利率 × 资产周转率
 var between1=(cards[0].x+cards[0].w+cards[1].x)/2;
 graphics.push({type:'circle',silent:true,z:14,shape:{cx:between1,cy:rowCenters[1],r:13},style:{fill:C.white,stroke:C.line,lineWidth:1}});
 if(setting.showOperators)graphics.push(textG(between1,rowCenters[1],'×',19,true,C.current,'center'));
 // 两个分式节点：销售净利率 = 净利润 ÷ 营业收入；资产周转率 = 营业收入 ÷ 平均总资产。
 if(setting.showOperators){
   var divideX=(cards[1].x+cards[1].w+cards[3].x)/2;
   [rowCenters[0],rowCenters[2]].forEach(function(y){
     graphics.push({type:'circle',silent:true,z:14,shape:{cx:divideX,cy:y,r:12},style:{fill:C.white,stroke:C.line,lineWidth:1}});
     graphics.push(textG(divideX,y,'÷',20,true,C.current,'center'));
   });
 }

}
if(setting.showROAAttribution){
 var root=cards[0],ax=root.x+12;
 // 归因分析直接附着在 ROA 节点下方，形成同一信息单元。
 var boxTop=root.y+root.h+7,boxH=113;
 graphics.push({type:'rect',silent:true,z:1,shape:{x:root.x,y:boxTop,width:root.w,height:boxH,r:5},
   style:{fill:'#F8FAFB',stroke:C.line,lineWidth:1}});
 graphics.push(textG(ax,boxTop+18,'ROA变动归因 · '+previous.year+'→'+latest.year,12,true,C.current,'left',root.w-18));
 graphics.push(textG(ax,boxTop+39,'净利率贡献  '+(attribution.margin>=0?'+':'')+nf(attribution.margin,2)+' 个百分点',11,true,trendColor(attribution.margin),'left',root.w-18));
 graphics.push(textG(ax,boxTop+57,'周转率贡献  '+(attribution.turnover>=0?'+':'')+nf(attribution.turnover,2)+' 个百分点',11,true,trendColor(attribution.turnover),'left',root.w-18));
 graphics.push(textG(ax,boxTop+75,'合计  '+(attribution.total>=0?'+':'')+nf(attribution.total,2)+' 个百分点',11,true,C.current,'left',root.w-18));
 var leader=Math.abs(attribution.margin)>=Math.abs(attribution.turnover)?'销售净利率':'资产周转率';
 var direction=attribution.total>1e-8?'上升':attribution.total< -1e-8?'下降':'保持稳定';
 var summary='ROA'+direction+'，主要驱动：'+leader;
 graphics.push(textG(ax,boxTop+96,summary,11,true,C.current,'left',root.w-18));
}
// 指数化模式：每一小图首年都等于100，且六图共用同一纵轴尺度，以比较相对变化。
var indexSeries=specs.map(function(s){var base=records[0][s.key];if(!(base>0))throw new Error('指数化需要基期指标为正数：'+s.name);
  return records.map(function(r){return setting.indexReference*r[s.key]/base;});});
var indexedMin=Math.min.apply(null,[setting.indexReference].concat.apply([setting.indexReference],indexSeries));
var indexedMax=Math.max.apply(null,[setting.indexReference].concat.apply([setting.indexReference],indexSeries));
var indexedPadding=Math.max(3,(indexedMax-indexedMin)*0.13);
var useIndexed=setting.trendScale==='indexed';
if(['indexed','local'].indexOf(setting.trendScale)<0)throw new Error('trendScale 只能是 indexed 或 local');
var grids=[],xAxes=[],yAxes=[],series=[];
specs.forEach(function(s,i){
 var card=cards[i],innerLeft=card.x+16,innerTop=card.y+65,innerWidth=card.w-32,innerHeight=card.h-(setting.showEndpointYears?94:82);
 graphics.push({type:'rect',silent:true,z:1,shape:{x:card.x,y:card.y,width:card.w,height:card.h,r:5},style:{fill:C.white,stroke:C.line,lineWidth:1}});
 graphics.push(textG(card.x+12,card.y+20,s.name,15,true,C.current));
 graphics.push(textG(card.x+12,card.y+44,s.formula+(useIndexed?' · 首年=100':''),11,false,C.secondary));
 graphics.push(textG(card.x+card.w-12,card.y+20,metricFmt(s.key,latest[s.key],true),16,true,C.current,'right',card.w*0.53));
 if(setting.showLatestChange){
   var diff=metricChange(s.key);
   graphics.push(textG(card.x+card.w-12,card.y+44,'较'+previous.year+'年 '+trendLabel(s.key,diff),11,true,trendColor(diff),'right',card.w*0.59));
 }
 if(setting.showEndpointYears && records.length){
   var yearY=innerTop+innerHeight+16;
   graphics.push(textG(innerLeft,yearY,records[0].year,10,false,C.secondary,'left'));
   graphics.push(textG(innerLeft+innerWidth,yearY,records[records.length-1].year,10,false,C.secondary,'right'));
 }
 grids.push({left:Math.round(innerLeft),top:Math.round(innerTop),width:Math.round(innerWidth),height:Math.round(innerHeight),containLabel:false});
 var v=useIndexed?indexSeries[i]:records.map(function(r){return r[s.key];});
 var lo=Math.min.apply(null,v),hi=Math.max.apply(null,v),span=Math.max(0.001,hi-lo),pad=span*0.24;
 if(s.key==='turnover')pad=Math.max(0.025,pad);
 xAxes.push({type:'category',gridIndex:i,data:records.map(function(r){return r.year;}),boundaryGap:false,axisLine:{show:false},axisTick:{show:false},axisLabel:{show:setting.showYearAxis,color:C.secondary,fontSize:10,interval:0,margin:5}});
 var axisMin=useIndexed?Math.max(0,indexedMin-indexedPadding):Math.max(0,lo-pad);
 var axisMax=useIndexed?indexedMax+indexedPadding:hi+pad;
 yAxes.push({type:'value',gridIndex:i,min:axisMin,max:axisMax,show:false,scale:true});
 if(useIndexed&&setting.showIndexBaseline){
   var baseY=innerTop+innerHeight*(axisMax-setting.indexReference)/(axisMax-axisMin);
   graphics.push({type:'line',silent:true,z:2,shape:{x1:innerLeft,y1:baseY,x2:innerLeft+innerWidth,y2:baseY},
     style:{stroke:C.line,lineWidth:1,lineDash:[3,4]}});
 }
 // 历史连线按相邻两年的涨跌分别染色，避免整条线只用一个颜色。
 if(setting.colorTrendSegments){
   for(var k=1;k<records.length;k++){
     var d=v[k]-v[k-1],segment=Array(records.length).fill(null);
     segment[k-1]=v[k-1];segment[k]=v[k];
     series.push({name:s.name+' · 趋势',type:'line',xAxisIndex:i,yAxisIndex:i,
       data:segment,connectNulls:false,showSymbol:false,silent:true,
       lineStyle:{width:setting.lineWidth+0.3,color:trendColor(d),opacity:k===records.length-1?1:setting.historicalLineOpacity},
       z:4,animationDuration:setting.animationDuration,tooltip:{show:false}});
   }
 }
 // 使用独立 scatter 呈现灰色历史点与深色最新点，避免折线被额外折返。
 series.push({name:s.name,type:'scatter',metricIndex:i,xAxisIndex:i,yAxisIndex:i,
   symbol:'circle',symbolSize:function(val,params){return params.dataIndex===records.length-1?10:setting.symbolSize;},
   itemStyle:{color:C.historical,borderColor:C.white,borderWidth:1.2},
   data:records.map(function(r,k){return {value:[r.year,v[k]],itemStyle:{color:k===records.length-1?trendColor(r[s.key]-records[k-1<0?0:k-1][s.key]):C.historical}};}),
   emphasis:{scale:1.35},z:6,animationDuration:setting.animationDuration,
   label:{show:setting.showPointLabels,position:'top',formatter:function(p){return useIndexed?nf(p.value[1],1):metricFmt(s.key,p.value[1]);},fontSize:9,color:C.current}
 });
});
option={
 backgroundColor:C.background,animation:true,
 grid:grids,xAxis:xAxes,yAxis:yAxes,series:series,graphic:graphics,
 tooltip:{trigger:'item',confine:true,backgroundColor:C.white,borderColor:C.line,borderWidth:1,padding:[10,13],textStyle:{color:C.current,fontSize:12},extraCssText:'box-shadow:0 4px 16px rgba(0,0,0,.08);',
  formatter:function(p){
    var serie=series[p.seriesIndex];
    if(!serie || serie.metricIndex===undefined)return '';
    var s=specs[serie.metricIndex];
    var rowHtml=records.map(function(r,k){
      var prev=k>0?records[k-1]:null;
      var d=prev?r[s.key]-prev[s.key]:null;
      var deltaText=d===null?'基期':trendLabel(s.key,d);
      var color=d===null?C.secondary:trendColor(d);
      var weight=k===records.length-1?'font-weight:700;':'';
      return '<tr style="'+weight+'border-bottom:1px solid '+C.line+'">'+
       '<td style="padding:6px 12px 6px 0;color:'+C.secondary+'">'+html(r.year)+'</td>'+
       '<td style="padding:6px 8px;text-align:right">'+html(metricFmt(s.key,r[s.key]))+'</td>'+
       '<td style="padding:6px 0 6px 12px;text-align:right;color:'+color+'">'+html(deltaText)+'</td></tr>';
    }).join('');
    var diff=metricChange(s.key);
    return '<div style="min-width:330px;font-family:Microsoft YaHei,Arial;color:'+C.current+';line-height:1.5">'+
     '<div style="font-size:14px;font-weight:700;margin-bottom:3px">'+html(s.name)+'</div>'+
     '<div style="font-size:11px;color:'+C.secondary+';margin-bottom:9px">'+html(setting.entity)+' · 全部财年趋势</div>'+
     '<table style="width:100%;border-collapse:collapse;font-size:12px"><thead><tr style="color:'+C.secondary+';border-bottom:1px solid '+C.line+'"><th align="left">财年</th><th align="right">指标值</th><th align="right">较上年变化</th></tr></thead><tbody>'+rowHtml+'</tbody></table>'+
     '<div style="margin-top:8px;padding-top:8px;border-top:1px solid '+C.line+';color:'+trendColor(diff)+';font-weight:700">'+
     '最近一年：'+html(trendLabel(s.key,diff))+'</div>'+
     (s.key==='roa'&&setting.showROAAttribution?'<div style="margin-top:8px;padding:8px;background:#F7F9FA;font-size:11px">ROA变动归因（'+html(previous.year)+'→'+html(latest.year)+'）<div>净利率：'+html((attribution.margin>=0?'+':'')+nf(attribution.margin,2))+' 个百分点</div><div>周转率：'+html((attribution.turnover>=0?'+':'')+nf(attribution.turnover,2))+' 个百分点</div><b>合计：'+html((attribution.total>=0?'+':'')+nf(attribution.total,2))+' 个百分点</b></div>':'')+
     '<div style="margin-top:6px;font-size:11px;color:'+C.secondary+'">计算关系：'+html(s.formula)+'</div>'+ 
     '<div style="margin-top:2px;font-size:10px;color:'+C.secondary+'">注：红绿表示指标数值升降，并不等同于经营好坏。</div></div>';
  }
 }
};
