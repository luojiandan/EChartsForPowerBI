function isNil(v) { return v === null || v === undefined || String(v).trim() === ''; }
function toNumber(v) { return isNil(v) ? null : Number(String(v).replace(/,/g,'')); }
function escapeHtml(s) { return String(s).replace(/[&<>"']/g, function(k) {
  return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[k];
}); }
function formatNumber(v, digits) {
  return v === null || !isFinite(v) ? '—' :
    (Math.abs(v)<1e-9?0:v).toLocaleString('en-US', {
      minimumFractionDigits:digits, maximumFractionDigits:digits
    });
}
function money(v) { return v===null?'—':formatNumber(v/setting.valueDivisor,setting.decimals); }
function signedMoney(v) { return v===null?'—':(v>1e-9?'+':'')+money(v); }
function percent(v) { return v===null?'—':(v>1e-9?'+':'')+formatNumber(v,setting.percentDecimals)+'%'; }
function pctOf(delta, base) { return base===null||Math.abs(base)<1e-9?null:100*delta/Math.abs(base); }
function sign(v) { return v>1e-9?'▲':v< -1e-9?'▼':'●'; }
function semanticColor(v) { return v===null?C.secondary:v>1e-9?C.positive:v< -1e-9?C.negative:C.secondary; }
function labelValue(v) { return formatNumber(v/setting.valueDivisor,setting.decimals); }

function bridgeDomain(steps){
  var values=[0];
  steps.forEach(function(s){values.push(s.start/setting.valueDivisor,s.end/setting.valueDivisor);});
  var lo=Math.min.apply(null,values),hi=Math.max.apply(null,values),span=Math.max(1,hi-lo);
  return {min:lo-Math.max(5,span*0.16),max:hi+Math.max(8,span*0.16)};
}

}
// 规整轴刻度：自动选 1/2/2.5/5/10 × 10^n 的等距步长。
// 不写死苹果示例的 -50~300、-10~70、0~6000，换数据后仍自动适配。
function niceStep(span, targetSegments){
  var raw=Math.max(1e-12,span/Math.max(2,targetSegments));
  var scale=Math.pow(10,Math.floor(Math.log10(raw)));
  var choices=[1,2,2.5,5,10].map(function(n){return n*scale;});
  var best=choices[0];
  choices.forEach(function(v){if(Math.abs(v-raw)<Math.abs(best-raw))best=v;});
  return best;
}
function niceDomain(domain,targetSegments,anchorZero){
  var range=Math.max(1e-10,domain.max-domain.min);
  var step=niceStep(range,targetSegments);
  var lo=anchorZero?0:Math.floor((domain.min+step*1e-9)/step)*step;
  var hi=Math.ceil((domain.max-step*1e-9)/step)*step;
  if(hi<=lo)hi=lo+step;
  var precision=function(v){return Math.round(v*1e9)/1e9;};
  return {min:precision(lo),max:precision(hi),interval:precision(step)};
}

function gText(x,y,str,size,bold,color,align){
  return {type:'text',silent:true,z:20,style:{x:x,y:y,text:String(str),
    font:(bold?'bold ':'')+size+'px '+setting.font,fill:color||C.text,
    align:align||'left',verticalAlign:'middle'}};
}
function gLine(x1,y1,x2,y2,color,width){
  return {type:'line',silent:true,z:1,shape:{x1:x1,y1:y1,x2:x2,y2:y2},style:{stroke:color||C.grid,lineWidth:width||1}};
}
function tooltipHtml(p){
  var idx=p.data&&p.data.rowIndex!==undefined?p.data.rowIndex:p.dataIndex;
  var r=rows[idx];if(!r)return '';
  function tr(name,value,color){return '<tr><td style="padding:4px 14px 4px 0;color:'+C.secondary+'">'+escapeHtml(name)+'</td><td style="text-align:right;color:'+(color||C.text)+';font-weight:600">'+escapeHtml(value)+'</td></tr>';}
  return '<div style="font-family:'+setting.font+';color:'+C.text+';min-width:225px;line-height:1.5">'+
    '<b style="font-size:14px">'+escapeHtml(r.name)+'</b><table style="width:100%;margin-top:7px;border-collapse:collapse">'+
    tr('AC '+ac,money(r.ac)+' '+setting.unit)+tr('PY '+py,money(r.py)+' '+setting.unit)+
    (hasPL?tr('PL',r.pl===null?'—':money(r.pl)+' '+setting.unit):'')+
    tr('△PY',signedMoney(r.deltaPY)+' '+setting.unit,semanticColor(r.deltaPY))+
    tr('△PY%',percent(r.pctPY),semanticColor(r.deltaPY))+
    (hasPL?tr('△PL',signedMoney(r.deltaPL)+' '+(r.deltaPL===null?'':setting.unit),semanticColor(r.deltaPL))+
           tr('△PL%',percent(r.pctPL),semanticColor(r.deltaPL)):'')+
    '</table></div>';
}

}
function bridgeSeries(steps,ix,name){
  return {name:name,type:'custom',coordinateSystem:'cartesian2d',xAxisIndex:ix,yAxisIndex:ix,
    data:steps.map(function(s,i){return {value:[i,s.end/setting.valueDivisor],rowIndex:i};}),
    encode:{x:0,y:1},clip:true,z:5,animationDuration:setting.animationDuration,
    emphasis:{disabled:true},renderItem:function(params,api){
      var i=params.dataIndex,s=steps[i];
      if(s.delta===null)return {type:'group',children:[]};
      var before=api.coord([i,s.start/setting.valueDivisor]);
      var after=api.coord([i,s.end/setting.valueDivisor]);
      var bottomCoord=api.coord([i,0])[1];
      var x=before[0], y1=s.total?bottomCoord:before[1], y2=after[1];
      var bw=Math.max(8,Math.min(32,categoryWidth*0.23));
      var color=s.total?(s.delta>1e-9?C.totalPositive:s.delta< -1e-9?C.totalNegative:C.ac):semanticColor(s.delta);
      var h=Math.max(1.5,Math.abs(y2-y1));
      var shapes=[{type:'rect',shape:{x:x-bw/2,y:Math.min(y1,y2),width:bw,height:h},style:{fill:color}}];
      if(setting.showConnectors&&i<steps.length-1&&!s.total){
        var nx=api.coord([i+1,0])[0];
        shapes.push({type:'line',shape:{x1:x+bw/2,y1:after[1],x2:nx-bw/2,y2:after[1]},
          style:{stroke:C.secondary,opacity:0.52,lineDash:[3,3],lineWidth:1}});
      }
      if(setting.showLabels && categoryWidth>64){
        var v=s.total?money(s.end):sign(s.delta)+' '+formatNumber(Math.abs(s.delta)/setting.valueDivisor,setting.decimals);
        var minTop=bridgeTops[ix]+10;
        var labelY=Math.max(minTop,Math.min(y1,y2)-8);
        shapes.push({type:'text',silent:true,style:{x:x,y:labelY,text:v,
          textAlign:'center',textVerticalAlign:'bottom',fill:color,
          font:'bold '+(compact?9:10)+'px '+setting.font}});
      }
      shapes.push({type:'rect',shape:{x:x-categoryWidth*0.42,y:Math.min(y1,y2)-16,
        width:categoryWidth*0.84,height:Math.max(28,h+29)},style:{fill:'rgba(255,255,255,0)'},z:6});
      return {type:'group',children:shapes};
    }};
}
