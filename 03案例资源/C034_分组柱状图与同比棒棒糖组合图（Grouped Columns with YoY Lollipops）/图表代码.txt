// ───────────────────────── 3. 通用数据处理与同比计算 ─────────────────────────
var C = setting.colors;
function fmtValue(raw, digits) {
  if (raw === null || raw === undefined || !Number.isFinite(Number(raw))) return '—';
  return (Number(raw) * setting.valueScale).toLocaleString('zh-CN', {
    minimumFractionDigits: digits, maximumFractionDigits: digits
  });
}
function fmtRate(rate) {
  return rate === null || !Number.isFinite(rate) ? '—' : Math.abs(rate).toFixed(1) + '%';
}
function fmtSignedRate(rate,digits) {
  if (rate === null || !Number.isFinite(rate)) return '—';
  return (rate > 0 ? '+' : '') + rate.toFixed(digits) + '%';
}
function signSymbol(n) { return n > 0 ? '▲' : n < 0 ? '▼' : '·'; }
function semColor(n) { return n > 0 ? C.positive : n < 0 ? C.negative : C.neutral; }
function safeHtml(value) {
  return String(value).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')
    .replace(/"/g,'&quot;').replace(/'/g,'&#39;');
}
if (年份.length !== 分类.length || 年份.length !== 度量值.length) {
  throw new Error('年份、分类、度量值三个输入数组长度必须一致。');
}
var allYears = Array.from(new Set(年份.map(v => String(v)))).sort(function(a,b) {
  return (Number.isFinite(Number(a)) && Number.isFinite(Number(b)))
    ? Number(a) - Number(b) : a.localeCompare(b, 'zh-CN', {numeric: true});
});
var allGroups = Array.from(new Set(分类.map(v => String(v))));
var lookup = new Map();
for (var i = 0; i < 年份.length; i++) {
  var year = String(年份[i]), group = String(分类[i]);
  var numericValue = Number(度量值[i]);
  if (度量值[i] === null || 度量值[i] === '' || !Number.isFinite(numericValue)) continue;
  var key = JSON.stringify([group, year]);
  lookup.set(key, (lookup.get(key) || 0) + numericValue);
}
function rawValue(g,y) {
  var k = JSON.stringify([g,y]);
  return lookup.has(k) ? lookup.get(k) : null;
}
if (setting.sortGroupsByLatest && allYears.length) {
  var latestYear = allYears[allYears.length - 1];
  allGroups.sort((a,b) => (rawValue(b,latestYear) ?? -Infinity) - (rawValue(a,latestYear) ?? -Infinity));
}
var gapSize = 1;
var axisLabels = [], barPoints = [], growthPoints = [], linePoints = [], groupCenters = [], separatorIndices = [];
var groupSummaries = new Map();
var firstYear = allYears.length ? allYears[0] : null;
var lastYear = allYears.length ? allYears[allYears.length-1] : null;
var maxRevenue = 0, maxYoY = 0, minYoY = 0;
var barColors = [C.historical1, C.historical2, C.historical3];
allGroups.forEach(function(group, groupIndex) {
  var start = axisLabels.length;
  var firstRaw = firstYear !== null ? rawValue(group, firstYear) : null;
  var lastRaw = lastYear !== null ? rawValue(group, lastYear) : null;
  var totalGrowthRate = firstRaw !== null && lastRaw !== null && firstRaw !== 0 && allYears.length >= 2
    ? 100 * (lastRaw - firstRaw) / Math.abs(firstRaw) : null;
  // 比例类度量默认首期为正数；如首期为负，分母用绝对值避免符号歧义。
  groupSummaries.set(group, {fromYear:firstYear, toYear:lastYear,
    firstRaw:firstRaw, lastRaw:lastRaw, totalGrowthRate:totalGrowthRate});
  allYears.forEach(function(year, yearIndex) {
    var raw = rawValue(group, year);
    var prevYear = yearIndex > 0 ? allYears[yearIndex-1] : null;
    var prevRaw = prevYear !== null ? rawValue(group,prevYear) : null;
    var hasYoY = raw !== null && prevRaw !== null && prevRaw !== 0;
    var delta = hasYoY ? raw-prevRaw : null;
    var rate = hasYoY ? 100*delta/prevRaw : null;
    maxRevenue = raw !== null ? Math.max(maxRevenue,raw*setting.valueScale) : maxRevenue;
    var axisIndex = axisLabels.length;
    axisLabels.push(year);
    var fill = (setting.emphasizeLatestYear && yearIndex === allYears.length-1)
      ? C.latest : barColors[Math.min(yearIndex,barColors.length-1)];
    barPoints.push({
      value: raw === null ? '-' : raw * setting.valueScale,
      itemStyle: {color:fill, borderRadius:[2,2,0,0]},
      _meta: {group:group, year:year, raw:raw, prevYear:prevYear, prevRaw:prevRaw, delta:delta, rate:rate}
    });
    if (hasYoY) {
      maxYoY = Math.max(maxYoY,rate);
      minYoY = Math.min(minYoY,rate);
      growthPoints.push({
        value: [axisIndex,rate],
        _meta: {group:group, year:year, raw:raw, prevYear:prevYear, prevRaw:prevRaw, delta:delta, rate:rate},
        symbolSize: yearIndex === allYears.length - 1 ? 14 : 10,
        itemStyle: {color:semColor(rate), borderColor:'#FFFFFF',borderWidth:2},
        label: {
          show:setting.showYoYLabel,
          position: rate >= 0 ? 'top' : 'bottom',
          distance: 8,
          color:semColor(rate),
          fontSize:11,
          fontWeight: yearIndex === allYears.length-1 ? 700 : 500,
          formatter:function(p) {
            var r=p.data._meta.rate;
            return signSymbol(r) + ' ' + fmtRate(r);
          }
        }
      });
      linePoints.push({value:[axisIndex,rate], _meta:{group:group,year:year,delta:delta,rate:rate}});
    }
  });
  groupCenters.push({name:group,startIndex:start,endIndex:start+allYears.length-1,
    axisCenter:start + (allYears.length-1)/2,summary:groupSummaries.get(group)});
  if (groupIndex < allGroups.length-1) {
    separatorIndices.push(axisLabels.length);
    for (var j=0;j<gapSize;j++) {axisLabels.push('');barPoints.push('-');}
  }
});

// ───────────────────────── 4. 自适应像素布局（Grid 四边均为像素） ─────────────────────────
var chartInstance = typeof myChart !== 'undefined' && myChart && typeof myChart.getWidth === 'function'

var bottomHeight = Math.max(110, H - bottomY - 106);

var rangeHi = Math.max(20,Math.ceil((maxYoY+4)/5)*5);
var maxAxisRevenue = Math.max(1,Math.ceil(maxRevenue*1.18/100)*100);
var zeroPixelY = topY + topHeight * rangeHi/(rangeHi-rangeLo);
function niceMajorStep(maxValue) {
  // 目标约每半图 1 条主刻度，如当前营收上限约 2200，则刻度间隔 1000。
  if (!(maxValue > 0)) return 1;
  var approximate = maxValue / 2.4;
  var power = Math.pow(10, Math.floor(Math.log10(approximate)));
  var factor = approximate / power;
  return (factor <= 1.25 ? 1 : factor <= 2.5 ? 2 : factor <= 5 ? 5 : 10) * power;
}
var revenueMajorStep = niceMajorStep(maxAxisRevenue);
function isMultipleOfStep(value, step) {
  return Math.abs(value / step - Math.round(value / step)) < 1e-7;
}
// V5：从数据中分别挑选“正增长最高”与“负增长最低”的分组；不按绝对值 Top N 排名。
// 如果某个方向不存在，则不强行补一个名额；两个方向各最多强调一个分组。
var emphasisGroups = new Set();
if (setting.summaryHighlightExtremes) {
  var highestPositive = null;
  var lowestNegative = null;
  groupCenters.forEach(function(g) {
    var rate = g.summary && g.summary.totalGrowthRate;
    if (!Number.isFinite(rate)) return;
    if (rate > 0 && (!highestPositive || rate > highestPositive.rate)) {
      highestPositive = {name:g.name, rate:rate};
    }
    if (rate < 0 && (!lowestNegative || rate < lowestNegative.rate)) {
      lowestNegative = {name:g.name, rate:rate};
    }
  });
  if (highestPositive) emphasisGroups.add(highestPositive.name);
  if (lowestNegative) emphasisGroups.add(lowestNegative.name);
}

if (setting.showGridLines) {
  [-10, 10].forEach(function(v) {

    graphics.push({type:'line',silent:true,z:-3,

  }
}
if (setting.showZeroBaseline) {
  graphics.push({type:'line',silent:true,z:-2,
    shape:{x1:left,y1:zeroPixelY,x2:W-right,y2:zeroPixelY},

}
var groupTitleY = topY - 81;
var summaryTextY = topY - 53;
var summaryLineY = topY - 33;
groupCenters.forEach(function(g) {
  var cx=left+(g.axisCenter+0.5)*tickWidth;
  graphics.push({type:'text',silent:true,z:30,x:cx,y:groupTitleY,
    style:{text:g.name,fill:C.text,font:'600 14px sans-serif',textAlign:'center',textVerticalAlign:'middle'}});

  // 同一类别的第一年到最后一年：淡灰跨度括号 + 正负语义色标签。
  // 图形按数据位置生成，不在代码中写死地区名称或具体年份。
  var summary=g.summary;
  if (setting.showTotalGrowth && summary.totalGrowthRate !== null) {
    var r=summary.totalGrowthRate;
    var spanText=summary.fromYear+' → '+summary.toYear;
    var rateText=fmtSignedRate(r,setting.totalGrowthDecimals);
    var braceX1=left+(g.startIndex+0.5)*tickWidth-setting.summaryBracketOverhang;
    var braceX2=left+(g.endIndex+0.5)*tickWidth+setting.summaryBracketOverhang;
    var drop=setting.summaryBracketDrop;
    graphics.push({type:'line',silent:true,z:25,shape:{x1:braceX1,y1:summaryLineY+drop,x2:braceX1,y2:summaryLineY},
      style:{stroke:C.bracket,lineWidth:1}});
    graphics.push({type:'line',silent:true,z:25,shape:{x1:braceX1,y1:summaryLineY,x2:braceX2,y2:summaryLineY},
      style:{stroke:C.bracket,lineWidth:1}});
    graphics.push({type:'line',silent:true,z:25,shape:{x1:braceX2,y1:summaryLineY,x2:braceX2,y2:summaryLineY+drop},
      style:{stroke:C.bracket,lineWidth:1}});
    // 保持两个独立文字元素；仅强调增长最快和下降最多的数值，字号与颜色规则不变。
    graphics.push({type:'text',silent:true,z:30,x:cx-7,y:summaryTextY,
      style:{text:spanText,font:'11px sans-serif',fill:C.textSecondary,
        textAlign:'right',textVerticalAlign:'middle'}});
    var summaryWeight = emphasisGroups.has(g.name)
      ? setting.summaryEmphasisWeight : setting.summaryNormalWeight;
    graphics.push({type:'text',silent:true,z:30,x:cx+7,y:summaryTextY,
      style:{text:rateText,font:summaryWeight+' 12px sans-serif',fill:semColor(r),
        textAlign:'left',textVerticalAlign:'middle'}});
  }
});
separatorIndices.forEach(function(idx) {
  var px=left+(idx+0.5)*tickWidth;
  graphics.push({type:'line',silent:true,
    shape:{x1:px,y1:groupTitleY-13,x2:px,y2:bottomY+bottomHeight+27},
    style:{stroke:C.rule,lineWidth:1,opacity:0.55}});
});
var yearLegendStart = left;
var legendY = 112;
allYears.forEach(function(year,yi) {

  graphics.push({type:'rect',shape:{x:lx,y:legendY,width:12,height:12},style:{fill:lc}});
  graphics.push({type:'text',x:lx+18,y:legendY+6,

if (setting.showChangeAmount) {
  growthPoints.forEach(function(pt) {

}

function tooltipHtml(p) {
  if (!p || !p.data || !p.data._meta) return '';
  var m=p.data._meta;
  var summary=groupSummaries.get(m.group);
  var yoyKnown = m.rate !== null && Number.isFinite(m.rate);
  var headline='<div style="font-size:13px;font-weight:700;color:'+C.text+';margin-bottom:10px;">'+safeHtml(m.group)
    +'<span style="font-weight:400;color:'+C.textSecondary+';margin-left:12px;">'+safeHtml(m.year)+' 财年</span></div>';
  var revenue='<div style="display:flex;justify-content:space-between;gap:35px;margin:6px 0;">'
    +'<span style="color:'+C.textSecondary+';">净销售额</span><b>'+fmtValue(m.raw,2)+' '+safeHtml(setting.unit)+'</b></div>';
  var comparison=yoyKnown
    ? '<div style="display:flex;justify-content:space-between;gap:35px;margin:6px 0;">'
      +'<span style="color:'+C.textSecondary+';">'+safeHtml(m.prevYear)+' 财年</span><span>'+fmtValue(m.prevRaw,2)+' '+safeHtml(setting.unit)+'</span></div>'
      +'<div style="border-top:1px solid '+C.rule+';margin:8px 0;"></div>'
      +'<div style="display:flex;justify-content:space-between;gap:35px;margin:6px 0;">'
      +'<span style="color:'+C.textSecondary+';">同比增减额</span><b style="color:'+semColor(m.delta)+';">'
      +signSymbol(m.delta)+' '+fmtValue(Math.abs(m.delta),2)+' '+safeHtml(setting.unit)+'</b></div>'
      +'<div style="display:flex;justify-content:space-between;gap:35px;margin:6px 0;">'
      +'<span style="color:'+C.textSecondary+';">同比增长率</span><b style="color:'+semColor(m.rate)+';">'
      +signSymbol(m.rate)+' '+fmtRate(m.rate)+'</b></div>'
    : '<div style="font-size:11px;color:'+C.textSecondary+';margin-top:8px;">首期或缺少同期数据，不计算同比</div>';
  var totalGrowthHtml = setting.showTotalGrowth && summary && summary.totalGrowthRate !== null
    ? '<div style="border-top:1px solid '+C.rule+';margin:9px 0 6px;"></div>'

      + '<span style="color:'+C.textSecondary+';">'+safeHtml(summary.fromYear)+' → '+safeHtml(summary.toYear)+' 累计增长</span>'
      + '<b style="color:'+semColor(summary.totalGrowthRate)+';">'
      +fmtSignedRate(summary.totalGrowthRate,setting.totalGrowthDecimals)+'</b></div>'
    : '';
  return '<div style="min-width:220px;font:12px/1.6 -apple-system,BlinkMacSystemFont,Segoe UI,sans-serif;color:'+C.text+';">'
    +headline+revenue+comparison+totalGrowthHtml+'</div>';
}
