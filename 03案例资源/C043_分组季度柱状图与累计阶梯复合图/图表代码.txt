// ===== 3 通用数据校验与派生计算 =====
if (!Array.isArray(类别) || !Array.isArray(季度) || !Array.isArray(营收) ||

var c043Format = function (value, digits) {
  return Number(value).toLocaleString('en-US', {
    minimumFractionDigits: digits,
    maximumFractionDigits: digits
  });
};
var c043Escape = function (text) {
  return String(text).replace(/&/g, '&amp;').replace(/</g, '&lt;')
    .replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#39;');
};
var c043Rgb = function (hex) {
  var v = hex.replace('#', '');
  return [parseInt(v.slice(0, 2), 16), parseInt(v.slice(2, 4), 16), parseInt(v.slice(4, 6), 16)];
};
var c043Mix = function (a, b, ratio) {
  var t = Math.max(0, Math.min(1, ratio));
  var aa = c043Rgb(a), bb = c043Rgb(b);
  return '#' + aa.map(function (n, i) {
    return Math.round(n * (1 - t) + bb[i] * t).toString(16).padStart(2, '0');
  }).join('').toUpperCase();
};
var c043Short = function (s, count) {
  var chars = Array.from(String(s));
  return chars.length <= count ? chars.join('') : chars.slice(0, Math.max(1, count - 1)).join('') + '…';
};

var c043Categories = [];
var c043Groups = Object.create(null);

  ? setting.periods.map(function (x) { return String(x).trim(); })
  : Array.from(new Set(季度.map(function (x) { return String(x).trim(); })))
      .sort(function (a,b) {return a.localeCompare(b,'zh-CN',{numeric:true});});

}
if (new Set(c043InputPeriods).size !== c043InputPeriods.length) {
  throw new Error('C043：periods 周期列表不能重复。');
}
for (var c043i = 0; c043i < 类别.length; c043i++) {
  var c043Cat = String(类别[c043i]).trim();
  var c043Period = String(季度[c043i]).trim();
  var c043Val = Number(营收[c043i]);
  if (!c043Cat || !c043Period || !Number.isFinite(c043Val) || c043Val < 0) {
    throw new Error('C043：第 ' + (c043i + 1) + ' 行类别、季度或营收无效，营收须为非负有限数。');
  }
  if (c043InputPeriods.indexOf(c043Period) < 0) {

  }
  if (!Object.prototype.hasOwnProperty.call(c043Groups, c043Cat)) {
    c043Groups[c043Cat] = {name: c043Cat, byPeriod: Object.create(null), rows: [], total: 0};
    c043Categories.push(c043Cat);
  }
  if (Object.prototype.hasOwnProperty.call(c043Groups[c043Cat].byPeriod, c043Period)) {
    throw new Error('C043：类别「' + c043Cat + '」的周期「' + c043Period + '」重复，请确保每组每期仅一条。');

  c043Groups[c043Cat].byPeriod[c043Period] = c043Val;
}

// 统一周期排序；每组必须包含同样的全部周期，避免误判环比和累计。
var c043PeriodCount = c043InputPeriods.length;
var c043SlotNames = [];
var c043Rows = [];
var c043MaxCumulative = 0;
var c043GroupTotals = [];
for (var c043g = 0; c043g < c043Categories.length; c043g++) {
  var c043Name = c043Categories[c043g];
  var c043Group = c043Groups[c043Name];
  var c043Amounts = c043InputPeriods.map(function (p) {
    if (!Object.prototype.hasOwnProperty.call(c043Group.byPeriod, p)) {
      throw new Error('C043：类别「' + c043Name + '」缺少周期「' + p + '」。');
    }
    return c043Group.byPeriod[p];
  });
  var c043AnnualTotal = c043Amounts.reduce(function (sum, v) { return sum + v; }, 0);
  if (!(c043AnnualTotal > 0)) { throw new Error('C043：类别「' + c043Name + '」所选周期的' + setting.metricName + '合计必须大于0。'); }
  c043Group.total = c043AnnualTotal;
  c043GroupTotals.push(c043AnnualTotal);
  c043MaxCumulative = Math.max(c043MaxCumulative, c043AnnualTotal);
  var c043Cum = 0;
  var c043ChangeList = c043Amounts.map(function (amount, k) {
    return k > 0 && c043Amounts[k - 1] !== 0
      ? (amount - c043Amounts[k - 1]) / c043Amounts[k - 1] * 100 : null;
  });
  
  for (var c043q = 0; c043q < c043PeriodCount; c043q++) {
    c043SlotNames.push(c043InputPeriods[c043q]);
    c043Cum += c043Amounts[c043q];
    var c043Row = {
      category: c043Name,
      groupIndex: c043g,
      quarter: c043InputPeriods[c043q],
      periodIndex: c043q,
      slotIndex: c043SlotNames.length - 1,
      revenue: c043Amounts[c043q],
      cumulative: c043Cum,
      total: c043AnnualTotal,
      qoq: c043ChangeList[c043q]
    };
    c043Rows.push(c043Row);
    c043Group.rows.push(c043Row);
  }
  if (c043g < c043Categories.length - 1) c043SlotNames.push('');
}
// 纵轴最大值按数据数量级自动计算，不再固定按 20 为步长取整。
var c043TargetMax = c043MaxCumulative * (1 + setting.maxYExtraRatio);
var c043Magnitude = Math.pow(10, Math.floor(Math.log10(c043TargetMax)));
var c043UnitY = c043TargetMax / c043Magnitude;
var c043NiceY = [1,1.2,1.5,2,2.5,3,4,5,6,8,10].filter(function(x){return x>=c043UnitY;})[0] || 10;
var c043YMax = c043NiceY * c043Magnitude;
var c043ColorFor = function (index) {
  var base = setting.seriesColors;
  if (!Array.isArray(base) || base.length < 1) throw new Error('C043：请至少配置一个系列颜色。');
  if (index < base.length) return base[index];
  var cycle = Math.floor(index / base.length);
  var color = base[index % base.length];
  // 以既有基础色衍生明度，不新增未经确认的颜色体系；不再简单循环相同颜色。
  return cycle % 2 ? c043Mix(color, '#263238', Math.min(0.15+0.06*Math.floor(cycle/2),0.34))
    : c043Mix(color,'#FFFFFF',Math.min(0.15+0.06*Math.floor(cycle/2),0.34));
};
var c043Title = setting.title || (setting.periodLabel + setting.metricName + '与累计：' + c043Categories.length + ' 个' + setting.categoryLabel + '表现如何？');
var c043Subtitle = setting.subtitle || (
  (setting.fiscalYear ? setting.fiscalYear + ' · ' : '') +
  c043PeriodCount + '期 · 柱体=单期' + setting.metricName +
  ' · 阶梯=所选期累计 · 单位：' + setting.unit
);
var c043FooterNote = setting.noteText || (
  '图表说明：柱体=单期' + setting.metricName + '，阶梯=所选期累计' + setting.metricName +
  '；▲▼表示较上期增加／减少；首期无环比。'
);

// 按真实数据自动提炼最值得强调的一项结论；替换 Power BI 数据后不会沿用 Apple 文案。
// 优先选择所有类别均环比下滑的周期（若有多个，选整体降幅更大的一期）。
var c043Insight = {ordinary:'', emphasized:''};
var c043WorstCommonPeriod = -1;
var c043WorstCommonAverage = Infinity;
for (var c043p = 1; c043p < c043PeriodCount; c043p++) {
  var c043ThisRows = c043Categories.map(function (cat) {return c043Groups[cat].rows[c043p];});
  if (c043ThisRows.every(function (row) {return row.qoq !== null && row.qoq < 0;})) {
    var c043Avg = c043ThisRows.reduce(function (sum,row) {return sum + row.qoq;},0) / c043ThisRows.length;
    if (c043Avg < c043WorstCommonAverage) {
      c043WorstCommonAverage = c043Avg;
      c043WorstCommonPeriod = c043p;
    }
  }
}
if (c043WorstCommonPeriod >= 0) {
  var c043WorstDrop = c043Categories.map(function (cat) {
    return c043Groups[cat].rows[c043WorstCommonPeriod];
  }).reduce(function (a,b) {return a.qoq <= b.qoq ? a : b;});
  c043Insight.ordinary = c043InputPeriods[c043WorstCommonPeriod] + '：' +
    c043Categories.length + ' 个' + setting.categoryLabel + '的' + setting.metricName + '均较上期下降；最大降幅 ';
  c043Insight.emphasized = '▼' + c043Format(Math.abs(c043WorstDrop.qoq),setting.growthDigits) +
    '%（' + c043WorstDrop.category + '）';
  c043Insight.briefOrdinary = c043InputPeriods[c043WorstCommonPeriod] + ' 全线环比下滑；最大 ';
  c043Insight.briefEmphasized = '▼' + c043Format(Math.abs(c043WorstDrop.qoq),setting.growthDigits) + '%';
} else {
  var c043BiggestChange = c043Rows.filter(function (row) {return row.qoq !== null;})
    .sort(function (a,b) {return Math.abs(b.qoq)-Math.abs(a.qoq);})[0];
  if (c043BiggestChange) {
    c043Insight.ordinary = '关键变化：' + c043BiggestChange.quarter + ' ' + c043BiggestChange.category + ' 较上期变化 ';
    c043Insight.emphasized = (c043BiggestChange.qoq >= 0 ? '▲' : '▼') +
      c043Format(Math.abs(c043BiggestChange.qoq),setting.growthDigits) + '%';
    c043Insight.briefOrdinary = c043BiggestChange.quarter + ' 最大环比变动 ';
    c043Insight.briefEmphasized = c043Insight.emphasized;
  }
}


// ===== 4 图元：用 custom series 画柱、阶梯、节点、重要标注 =====
// 坐标使用 ECharts 类目轴和数值轴，每次 chart.resize() 会自动重算像素位置。
var c043Line = function (x1, y1, x2, y2, color, thickness, dashed, opacity) {
  return {type:'line',silent:true,shape:{x1:x1,y1:y1,x2:x2,y2:y2},
    style:{stroke:color,lineWidth:thickness||1,lineDash:dashed||null,
      opacity: opacity == null ? 1 : opacity}};
};
var c043Text = function (x, y, label, color, size, align, weight, opt) {
  var style = {
    x:x,y:y,text:label,fill:color,fontSize:size,
    fontFamily:'Arial, Microsoft YaHei, sans-serif',
    fontWeight:weight||400,
    textAlign:align||'center',align:align||'center',
    textVerticalAlign:'middle',verticalAlign:'middle'
  };
  if (opt && opt.width) {
    style.width = opt.width;
    style.overflow = 'truncate';
    style.ellipsis = '…';
  }
  return {type:'text',silent:true,style:style};
};
var c043SeriesData = c043Rows.map(function (row, i) {
  return {name:row.category + ' · ' + row.quarter,
    value:[row.slotIndex,row.revenue,row.cumulative,row.qoq === null ? 0 : row.qoq],
    rawIndex:i};
});

var c043RenderItem = function (params, api) {
  var row = c043Rows[params.dataIndex];
  if (!row) return null;
  var area = params.coordSys;
  var band = area.width / c043SlotNames.length;
  var compact = band < setting.minBandToShowAllValues;
  var narrow = band < 26;
  var center = api.coord([row.slotIndex, 0])[0];
  var baseY = api.coord([row.slotIndex, 0])[1];
  var barY = api.coord([row.slotIndex, row.revenue])[1];
  var cumY = api.coord([row.slotIndex, row.cumulative])[1];
  var barWidth = Math.min(64, band * setting.barWidthRatio);
  var color = c043ColorFor(row.groupIndex);
  var pale = c043Mix(color,'#FFFFFF',setting.barTopWhiteMix);
  var group = {type:'group',children:[]};
  var cs = group.children;
  // 首行负责绘制顶部数据结论，不引入额外 DOM、graphic 或宽度初始化值。
  if (params.dataIndex === 0 && setting.showInsight && c043Insight.ordinary) {
    var useBrief = area.width < 510;
    var prefix = useBrief ? c043Insight.briefOrdinary : c043Insight.ordinary;
    var emphasis = useBrief ? c043Insight.briefEmphasized : c043Insight.emphasized;
    // 避免字段名称中的大括号、竖线干扰 ECharts 富文本标记解析。
    emphasis = String(emphasis).replace(/[{}|]/g,'');
    prefix = String(prefix).replace(/[{}|]/g,'');
    cs.push({type:'text',silent:true,style:{
      x:32,y:area.y - (setting.gridTop - setting.hintTop),
      text:'{description|' + prefix + '}{change|' + emphasis + '}',
      width:Math.max(100,area.width + area.x - 40),overflow:'truncate',ellipsis:'…',
      fontFamily:'Arial, Microsoft YaHei, sans-serif',
      rich:{
        description:{fill:setting.mutedColor,fontSize:11,fontWeight:400},
        change:{fill:emphasis.charAt(0) === '▼' ? setting.negativeColor : setting.positiveColor,
          fontSize:11,fontWeight:600}
      }
    }});
  }
  var barHeight = Math.max(0, baseY - barY);
  var column = {
    type:'rect',
    shape:{x:center - barWidth / 2,y:barY,width:barWidth,height:barHeight,r:[3,3,0,0]},
    style:{fill:{type:'linear',x:0,y:0,x2:0,y2:1,
      colorStops:[{offset:0,color:pale},{offset:1,color:c043Mix(color,'#FFFFFF',setting.barBottomWhiteMix)}]},
      stroke:c043Mix(color,'#FFFFFF',0.48),lineWidth:0.6},
    emphasis:{style:{stroke:color,lineWidth:1.7}}
  };
  cs.push(column);

  // 保留阶梯的真实累计高度；水平段降低线宽和不透明度，纵向跃升稍强。
  // Q1 累计等于 Q1 季度营收，首节点与柱顶重合是数学上的必然关系。
  if (row.periodIndex > 0) {
    var prev = c043GroupForRow(row).rows[row.periodIndex - 1];
    var prevX = api.coord([prev.slotIndex, 0])[0];
    var prevY = api.coord([prev.slotIndex, prev.cumulative])[1];
    cs.push(c043Line(prevX,prevY,center,prevY,color,
      setting.stepHorizontalWidth,null,setting.stepHorizontalOpacity));
    cs.push(c043Line(center,prevY,center,cumY,color,
      setting.stepVerticalWidth,null,setting.stepVerticalOpacity));
  }
  cs.push({type:'circle',silent:true,
    shape:{cx:center,cy:cumY,
      r:row.periodIndex === c043PeriodCount - 1 ? setting.lastDotRadius : setting.dotRadius},
    style:{fill:color,stroke:'#FFFFFF',lineWidth:1.15}});

  // 重点阅读逻辑：上方标签以累计最终值为主；柱体只保留必要数值。
  if (setting.showBarValues && (!compact || row.periodIndex === c043PeriodCount - 1)) {
    var valueText = c043Format(row.revenue, narrow ? 0 : setting.valueDigits);
    // Q1 柱顶与阶梯首节点重合；Q2 通常离上一期累计水平线很近：文字放入柱顶部避让。
    var insideBar = row.periodIndex < 2;
    cs.push(c043Text(center,insideBar ? barY + 13 : barY - 11,
      valueText,insideBar ? setting.textColor : setting.textColor,
      narrow ? 8 : compact ? 9 : 10, 'center', 500));
  }
  // 逐期环比：从第二期起，增加显示绿色▲，减少显示红色▼，零变化显示中性 0.0%。
  // 在密集类别/柱体很短时退回 Tooltip，以防压盖；三、五类常见布局全部可展示。
  if (setting.showQoQLabels && row.periodIndex > 0 && row.qoq !== null &&
      band >= setting.minBandToShowQoQ && barHeight >= setting.minBarHeightToShowQoQ) {
    var sign = row.qoq > 0 ? '▲' : row.qoq < 0 ? '▼' : '';
    var percent = c043Format(Math.abs(row.qoq),band < 43 ? 0 : setting.growthDigits) + '%';
    var qText = sign ? sign + (band < 48 ? '' : ' ') + percent : '0.0%';
    var qColor = row.qoq > 0 ? setting.positiveColor : row.qoq < 0 ? setting.negativeColor : setting.mutedColor;
    // 数值位于柱顶上方或内部第一行，环比位于柱体第二行，避免与累计节点重叠。
    cs.push(c043Text(center,barY + (row.periodIndex===1 ? 34 : 20),qText,qColor,
      band < 46 ? 8.4 : 10,'center',600));
  }

  // 三组标题：绘制在对应分组的中心，宽度随绘图区域自动伸缩。
  if (row.periodIndex === 0) {
    var groupRows = c043GroupForRow(row).rows;
    var lastX = api.coord([groupRows[groupRows.length - 1].slotIndex,0])[0];
    var titleX = (center + lastX) / 2;
    var titleMax = Math.max(4,Math.min(setting.groupTitleMaxChars,Math.floor((lastX - center + band) / 13.3)));
    cs.push(c043Text(titleX,area.y - 27,c043Short(row.category,titleMax),color,
      band < 25 ? 10 : 12,'center',700,{width:Math.max(32,lastX - center + band * 0.8)}));
  }

  // 默认从 Q2 起，给每个累计节点标值；Q1与柱顶同值，不重复。
  // 仅在空间很窄时收起中间节点数值，保留末期结果。
  var showThisCumulative = setting.showCumulativeLabels &&
    row.periodIndex >= Math.max(1,setting.cumulativeLabelStartIndex) &&
    (band >= setting.minBandToShowCumulative ||
      (row.periodIndex === c043PeriodCount - 1 && band >= 23));
  if (showThisCumulative) {
    cs.push(c043Text(center,cumY - 14,c043Format(row.cumulative,
      band < 39 ? 0 : setting.cumulativeDigits),color,
      band < 48 ? 9 : 10.5,'center',row.periodIndex===c043PeriodCount-1 ? 700 : 600));
  }

  // 不改变柱和阶梯的真实比例，仅用极浅分界线强化组别归属。
  if (row.periodIndex === c043PeriodCount - 1 && row.groupIndex < c043Categories.length - 1) {
    var separator = center + band * 0.92;
    cs.push(c043Line(separator,area.y - 12,separator,baseY + 14,setting.gridLineColor,1));
  }
  return group;
};
var c043GroupForRow = function (row) { return c043Groups[row.category]; };

// ===== 5 富文本 Tooltip（四项核心值，其余只在悬停时出现） =====
var c043Tooltip = function (params) {
  if (!params || params.dataIndex == null) return '';
  var row = c043Rows[params.dataIndex];
  if (!row) return '';
  var base = setting.textColor, sec = setting.mutedColor;
  var color = c043ColorFor(row.groupIndex);
  var qoq = row.qoq === null ? '<span style="color:'+sec+'">无上期基数</span>' :
    '<span style="font-weight:700;color:'+(row.qoq >= 0 ? setting.positiveColor : setting.negativeColor)+'">'+
    (row.qoq >= 0 ? '▲ ' : '▼ ') + c043Format(Math.abs(row.qoq),setting.growthDigits) + '%</span>';
  var line = function (label,value) {
    return '<div style="display:flex;justify-content:space-between;gap:28px;padding:4px 0;">'+
      '<span style="color:'+sec+'">'+label+'</span><span style="color:'+base+';font-weight:600">'+value+'</span></div>';
  };
  return '<div style="min-width:216px;max-width:292px;font-family:Arial,Microsoft YaHei,sans-serif;font-size:12px;line-height:1.5;color:'+base+'">'+
    '<div style="font-size:13px;font-weight:700;margin-bottom:3px">'+
    '<span style="display:inline-block;width:9px;height:9px;border-radius:2px;background:'+color+';margin-right:8px"></span>'+c043Escape(row.category)+'</div>'+
    '<div style="color:'+sec+';margin-bottom:6px">'+
    (setting.fiscalYear ? c043Escape(setting.fiscalYear)+' · ' : '')+c043Escape(row.quarter)+'</div>'+
    line('本'+c043Escape(setting.periodLabel)+c043Escape(setting.metricName),c043Format(row.revenue,2)+' '+c043Escape(setting.unit))+
    '<div style="display:flex;justify-content:space-between;gap:28px;padding:4px 0;">'+
      '<span style="color:'+sec+'">较上'+c043Escape(setting.periodLabel)+'变化</span>'+qoq+'</div>'+
    '<div style="border-top:1px solid '+setting.gridLineColor+';margin:8px 0 4px"></div>'+
    line('截至本期累计'+c043Escape(setting.metricName),c043Format(row.cumulative,2)+' '+c043Escape(setting.unit))+
    line('本期占本类别合计',c043Format(row.revenue / row.total * 100,1)+'%')+

    '</div>';
};
