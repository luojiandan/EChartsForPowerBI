// 【三】通用数据处理：只按输入字段索引对应，不做任何国家/年份假设。
function 解析有限数值(v, row, col) {
    if (v === null || v === undefined || String(v).trim() === '') {
        throw new Error('第 ' + (row + 1) + ' 行 ' + col + ' 为空；不允许以 0 替代缺失。');
    }
    var n = typeof v === 'number' ? v : Number(String(v).replace(/,/g, '').trim());
    if (!Number.isFinite(n)) {
        throw new Error('第 ' + (row + 1) + ' 行 ' + col + ' 不是有效数字：' + String(v));
    }
    return n;
}
function 创建行数据(labels, columns, axisDefs, types) {
    if (!Array.isArray(labels) || !Array.isArray(columns) || columns.length < 2) {
        throw new Error('至少需要对象名称和两个数值指标数组。');
    }
    if (axisDefs.length !== columns.length || types.length !== columns.length) {
        throw new Error('setting.axisDefs 与 axisTypes 数量必须等于指标数组数量。');
    }
    var seen = Object.create(null);
    var rows = [];
    for (var j = 0; j < columns.length; j++) {
        if (!Array.isArray(columns[j]) || columns[j].length !== labels.length) {
            throw new Error('指标' + (j + 1) + ' 的行数与对象名称不一致。');
        }
    }
    for (var i = 0; i < labels.length; i++) {
        var label = String(labels[i] == null ? '' : labels[i]).trim();
        if (!label) { throw new Error('第 ' + (i + 1) + ' 行对象名称为空。'); }
        if (seen[label]) { throw new Error('对象名称重复：' + label + '；请先确认数据粒度。'); }
        seen[label] = true;
        var nums = columns.map(function (col, j) {
            var n = 解析有限数值(col[i], i, axisDefs[j].name);
            if (types[j] === 'log' && n <= 0) {
                throw new Error('对数轴 ' + axisDefs[j].name + ' 存在非正数：' + label);
            }
            return n;
        });
        rows.push({ name: label, value: nums, originalIndex: i });
    }
    if (rows.length < 2) { throw new Error('至少需要两个对象进行比较。'); }
    return rows;
}
function 转义HTML(v) {
    return String(v).replace(/&/g, '&amp;').replace(/</g, '&lt;')
        .replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}
function 数值格式(v, decimals) {
    var a = Number(v);
    return a.toLocaleString('en-US', { minimumFractionDigits: decimals, maximumFractionDigits: decimals });
}
function 轴短格式(v, definition) {
    if (definition.axisAbbrev === 'k' && Math.abs(v) >= 1000) {
        var num = v / 1000;
        return (Math.abs(num) >= 100 ? num.toFixed(0) : num.toFixed(1)) + 'k';
    }
    if (Math.abs(v) >= 100) { return Number(v).toFixed(0); }
    if (Math.abs(v) >= 10) { return Number(v).toFixed(0); }
    return Number(v).toFixed(1);
}

function 生成提示(params, axisDefs) {
    var dataItem = params.data || {};
    var values = Array.isArray(dataItem) ? dataItem : dataItem.value;
    if (!Array.isArray(values)) { return ''; }
    var label = dataItem.name || params.seriesName || '对象';
    var html = '<div style="font-family:Arial,Microsoft YaHei,sans-serif;color:#263238;min-width:220px;max-width:' + setting.tooltipMaxWidth + 'px">';
    html += '<div style="font-size:14px;font-weight:700;margin-bottom:9px">' + 转义HTML(label) + '</div>';
    html += '<div style="border-top:1px solid #E5E9ED;margin-bottom:9px"></div>';
    var displayOrder = 验证坐标轴顺序(setting.axisDisplayOrder, axisDefs.length);
    for (var j = 0; j < displayOrder.length; j++) {
        var i = displayOrder[j];
        var d = axisDefs[i];
        html += '<div style="display:flex;justify-content:space-between;align-items:baseline;gap:18px;margin:5px 0">';
        html += '<span style="color:#8493A2">' + 转义HTML(d.name) + '</span>';
        html += '<b style="font-variant-numeric:tabular-nums;white-space:nowrap">' + 数值格式(values[i], d.decimals) + (d.unit === '%' ? '%' : (' ' + 转义HTML(d.unit))) + '</b></div>';
    }
    return html + '</div>';
}

// 只改变轴的视觉排列，不修改数据数组顺序及 Power BI 字段绑定关系。
function 验证坐标轴顺序(order, size) {
    if (order == null) { return Array.from({ length: size }, function (_, i) { return i; }); }
    if (!Array.isArray(order) || order.length !== size ||
        order.some(function (n) { return !Number.isInteger(n) || n < 0 || n >= size; }) ||
        (new Set(order)).size !== size) {
        throw new Error('setting.axisDisplayOrder 必须是从 0 到 ' + (size - 1) + ' 的无重复索引排列。');
    }
    return order.slice();
}

// 【四·一】选择状态：点击折线、下拉选择、文字搜索共用一套渲染状态。
// 文字搜索可同时突出多个匹配项；锁定单个对象时在相应坐标轴标出五个数值。
var 当前选中名称 = null;
var 当前搜索词 = '';
function 标准化搜索词(v) { return String(v == null ? '' : v).trim().toLocaleLowerCase(); }
function 是否命中搜索(name) {
    var keyword = 标准化搜索词(当前搜索词);
    return !!keyword && 标准化搜索词(name).indexOf(keyword) !== -1;
}
function 取得搜索匹配项() {
    var keyword = 标准化搜索词(当前搜索词);
    return keyword ? rows.filter(function(r) { return 是否命中搜索(r.name); }) : [];
}
function 取得详情对象() {
    if (当前选中名称 != null) {
        return rows.find(function (r) { return r.name === 当前选中名称; }) || null;
    }
    var matches = 取得搜索匹配项();
    return matches.length === 1 ? matches[0] : null;
}
function 生成可选中系列数据() {
    var searching = !!标准化搜索词(当前搜索词);
    var focusing = searching || 当前选中名称 !== null;
    var normalItems = [], highlightedItems = [];
    rows.forEach(function (r) {
        var hit = searching ? 是否命中搜索(r.name)
                            : (当前选中名称 !== null && r.name === 当前选中名称);
        var item = {
            name: r.name,
            value: r.value.slice(),
            lineStyle: {
                color: hit ? setting.clickSelectedColor : setting.colors.line,
                width: hit ? setting.lineWidth + 1.65 : setting.lineWidth,
                opacity: !focusing ? setting.lineOpacity
                         : (hit ? setting.selectedOpacity : setting.unselectedOpacity)
            },
            emphasis: { disabled: true }
        };
        (hit ? highlightedItems : normalItems).push(item);
    });
    return normalItems.concat(highlightedItems);
}

function 取得指标坐标(instance, chosen, originalIndex, visualIndex) {
    // 优先从实际 ECharts 极坐标以外的 parallel 坐标系计算像素位置；适配缩放与尺寸变化。
    try {
        if (instance && typeof instance.getModel === 'function') {
            var model = instance.getModel();
            var parallelModel = model && model.getComponent('parallel', 0);
            var coordSys = parallelModel && parallelModel.coordinateSystem;
            if (coordSys && typeof coordSys.dataToPoint === 'function') {
                var pos = coordSys.dataToPoint(chosen.value[originalIndex], 'dim' + originalIndex);
                if (Array.isArray(pos) && Number.isFinite(pos[0]) && Number.isFinite(pos[1])) {
                    return pos;
                }
            }
        }
    } catch (err) { /* 部分 Power BI 容器不暴露内部坐标系，使用同尺度后备计算。 */ }
    // 标准平行坐标线性/对数轴的后备坐标映射。
    var width = instance && typeof instance.getWidth === 'function' ? instance.getWidth() : box.width;
    var height = instance && typeof instance.getHeight === 'function' ? instance.getHeight() : box.height;
    var left = width < 760 ? 64 : setting.layout.chartLeft;
    var right = width < 760 ? 64 : setting.layout.chartRight;
    var bottom = width < 560 ? Math.max(setting.layout.chartBottom, 120)
               : width < 760 ? Math.max(setting.layout.chartBottom, 108) : setting.layout.chartBottom;
    var axis = axes[visualIndex];
    var v = chosen.value[originalIndex];
    var frac = setting.axisTypes[originalIndex] === 'log'
        ? (Math.log(v) - Math.log(axis.min)) / (Math.log(axis.max) - Math.log(axis.min))
        : (v - axis.min) / (axis.max - axis.min);
    frac = Math.min(1, Math.max(0, frac));
    var x = left + (width - left - right) * visualIndex / Math.max(1, 显示顺序.length - 1);
    var y = setting.layout.chartTop + (height - setting.layout.chartTop - bottom) * (1 - frac);
    return [x, y];
}
// ECharts graphic 的 $action: 'remove' 不能用于尚未创建的元素：
// 否则 GraphicView.updateLeaveTo 可能读取 undefined 的内部状态并报 __ec_inner_* 错误。
// 这里改用稳定 ID + 普通 merge 更新，并以 invisible 隐藏已经创建的标注。
function 创建指标标签元素(instance, chosen) {
    var elements = [];
    var color = setting.clickSelectedColor;
    var order = 显示顺序;
    for (var j = 0; j < order.length; j++) {
        var dotId = 'c036-selected-dot-' + j;
        var textId = 'c036-selected-value-' + j;
        var originalIndex = order[j];
        var xy = chosen ? 取得指标坐标(instance, chosen, originalIndex, j) : [0, 0];
        var valueStr = chosen ? 选中数值格式(chosen.value[originalIndex], setting.axisDefs[originalIndex]) : '';
        var isLast = j === order.length - 1;
        var fs = setting.axisValueLabelFontSize || 12;
        var shift = setting.axisValueLabelOffset || 9;
        var below = chosen && xy[1] < setting.layout.chartTop + 35;
        elements.push({
            id: dotId, type: 'circle', silent: true, invisible: !chosen, z: 100,
            shape: { cx: xy[0], cy: xy[1], r: 3.6 },
            style: { fill: '#FFFFFF', stroke: color, lineWidth: 1.8 }
        });
        elements.push({
            id: textId, type: 'text', silent: true, invisible: !chosen, z: 101,
            x: xy[0] + (isLast ? -shift : shift), y: xy[1] + (below ? 11 : -12),
            style: {
                text: valueStr, fill: color,
                font: '600 ' + fs + 'px Arial,Microsoft YaHei,sans-serif',
                align: isLast ? 'right' : 'left', verticalAlign: below ? 'top' : 'bottom',
                backgroundColor: 'rgba(255,255,255,0.96)',
                padding: [3, 5, 3, 5], borderRadius: 3
            }
        });
    }
    return elements;
}
function 更新指标轴标签(instance) {
    if (!instance || typeof instance.setOption !== 'function') { return; }
    var chosen = setting.axisValueLabelsEnabled ? 取得详情对象() : null;
    // 关键：未曾创建任何标签且当前未选中时，直接跳过 setOption。
    // 避免在 ECharts 主 option 初始化以前提交无意义的 graphic 更新。
    if (!chosen && !instance.__c036AxisLabelsCreated) { return; }
    instance.setOption({ graphic: 创建指标标签元素(instance, chosen) }, { lazyUpdate: false });
    instance.__c036AxisLabelsCreated = true;
}
function 更新搜索状态提示(instance) {

    if (ui && ui.input && ui.clear) {
        ui.clear.style.display = ui.input.value ? 'block' : 'none';
    }
    更新指标轴标签(instance);
}
function 绑定折线点击选择() {
    var instance = 获得实例();
    if (!instance || !setting.clickSelectEnabled) { return; }
    if (instance.__c036ParallelClick) { instance.off('click', instance.__c036ParallelClick); }
    if (instance.__c036ParallelBrush) { instance.off('axisareaselected', instance.__c036ParallelBrush); }
    instance.__c036ParallelClick = function (params) {
        if (params.componentType !== 'series' || params.seriesType !== 'parallel') { return; }
        var name = params.name || (params.data && params.data.name);
        if (!name) { return; }
        当前搜索词 = '';
        if (instance.__c036SearchUI) {
            instance.__c036SearchUI.input.value = '';
            instance.__c036SearchUI.closeList();
        }
        当前选中名称 = (name === 当前选中名称) ? null : name;
        if (instance.__c036SearchUI && 当前选中名称) {
            instance.__c036SearchUI.input.value = 当前选中名称;
        }
        刷新折线高亮(instance);
        更新搜索状态提示(instance);
    };
    instance.__c036ParallelBrush = function () {
        if (当前选中名称 !== null) {
            当前选中名称 = null;
            if (instance.__c036SearchUI) { instance.__c036SearchUI.input.value = ''; }
            刷新折线高亮(instance);
            更新搜索状态提示(instance);
        }
    };
    instance.on('click', instance.__c036ParallelClick);
    instance.on('axisareaselected', instance.__c036ParallelBrush);
}

// 【四·二】组合搜索框：本 JS 在 ECharts 容器内创建 input 与滚动下拉列表。
// 不需要外部 HTML/CSS。实际 Power BI 容器若禁用 DOM 写入，则保留图表与原生 Tooltip。
function 安装搜索框() {
    var instance = 获得实例();
    if (!instance || !setting.searchEnabled || typeof document === 'undefined' ||
        typeof document.createElement !== 'function' || typeof instance.getDom !== 'function') { return; }
    var root = instance.getDom();
    if (!root || typeof root.appendChild !== 'function') { return; }
    // 避免在同一 ECharts 实例重跑脚本后保留旧节点及 document 事件。
    if (instance.__c036SearchUI) {
        var oldUI = instance.__c036SearchUI;
        if (typeof oldUI.cleanup === 'function') { oldUI.cleanup(); }
        if (oldUI.host && oldUI.host.parentNode) { oldUI.host.parentNode.removeChild(oldUI.host); }
        instance.__c036SearchUI = null;
    }
    if (typeof window !== 'undefined' && typeof window.getComputedStyle === 'function' &&
        window.getComputedStyle(root).position === 'static') { root.style.position = 'relative'; }

    var host = document.createElement('div');
    var hostTop = (typeof instance.getWidth === 'function' && instance.getWidth() < 760) ? 89 : 22;
    host.style.cssText = 'position:absolute;top:' + hostTop + 'px;right:30px;z-index:50;width:' +
        Math.max(160, Number(setting.searchWidth) || 238) + 'px;max-width:calc(100% - 60px);font-family:Arial,Microsoft YaHei,sans-serif;pointer-events:auto;';
    var wrap = document.createElement('div');
    wrap.style.cssText = 'display:flex;align-items:center;height:35px;background:#FFFFFF;border:1px solid #D9DEE3;border-radius:7px;box-sizing:border-box;padding:0 8px;';
    var glass = document.createElement('span');
    glass.textContent = '⌕';
    glass.setAttribute('aria-hidden', 'true');
    glass.style.cssText = 'color:#8493A2;font-size:20px;line-height:1;margin-right:5px;';
    var input = document.createElement('input');
    input.type = 'text';
    input.placeholder = setting.searchPlaceholder || '搜索对象…';
    input.autocomplete = 'off';
    input.setAttribute('role', 'combobox');
    input.setAttribute('aria-label', '搜索并选择' + setting.itemLabel);
    input.setAttribute('aria-haspopup', 'listbox');
    input.setAttribute('aria-expanded', 'false');
    input.style.cssText = 'min-width:0;flex:1;width:100%;height:30px;padding:0;border:0;outline:0;box-shadow:none;appearance:none;background:transparent;color:#263238;font:12px Arial,Microsoft YaHei,sans-serif;';
    var clear = document.createElement('button');
    clear.type = 'button'; clear.textContent = '×'; clear.title = '清除搜索或选择';
    clear.setAttribute('aria-label', '清除搜索或选择');
    clear.style.cssText = 'display:none;cursor:pointer;border:0;background:transparent;color:#8493A2;font:20px Arial,sans-serif;padding:0 2px 1px;margin:0;';
    var arrow = document.createElement('button');
    arrow.type = 'button'; arrow.textContent = '▾'; arrow.title = '展开全部' + setting.itemLabel;
    arrow.setAttribute('aria-label', '展开候选列表');
    arrow.style.cssText = 'background:none;border:0;color:#8493A2;font:13px Arial,sans-serif;cursor:pointer;margin:0 0 0 4px;padding:0 2px;';
    var dropdown = document.createElement('div');
    dropdown.setAttribute('role','listbox');
    dropdown.setAttribute('aria-label', setting.itemLabel + '列表');
    dropdown.style.cssText = 'display:none;position:absolute;top:39px;left:0;right:0;max-height:'+
        Math.max(120, Number(setting.dropdownMaxHeight) || 230)+'px;overflow-y:auto;overscroll-behavior:contain;background:#FFFFFF;border:1px solid #D9DEE3;border-radius:7px;box-shadow:0 5px 18px rgba(38,50,56,0.13);padding:4px;box-sizing:border-box;z-index:51;';
    wrap.appendChild(glass); wrap.appendChild(input); wrap.appendChild(clear); wrap.appendChild(arrow);
    host.appendChild(wrap); host.appendChild(dropdown);
    root.appendChild(host);

    var ui = {host:host, input:input, clear:clear, arrow:arrow, dropdown:dropdown,
              filteredNames:[], activeIndex:-1, opened:false};
    instance.__c036SearchUI = ui;
    function closeList() {
        dropdown.style.display = 'none'; ui.opened = false; ui.activeIndex = -1;
        input.setAttribute('aria-expanded', 'false');
    }
    function onSelection(name) {
        当前搜索词 = '';
        当前选中名称 = name;
        input.value = name;
        closeList();
        刷新折线高亮(instance);
        更新搜索状态提示(instance);
    }
    // 列表有 30+ 项时内置滚动，不使整个图表或 Power BI 页面滚动。
    function renderList(query) {
        var q = 标准化搜索词(query);
        var matches = rows.filter(function (r) { return !q || 标准化搜索词(r.name).indexOf(q) !== -1; });
        ui.filteredNames = matches.map(function (r) { return r.name; });
        ui.activeIndex = -1;
        while (dropdown.firstChild) { dropdown.removeChild(dropdown.firstChild); }
        if (!matches.length) {
            var empty = document.createElement('div');
            empty.textContent = '无匹配的' + setting.itemLabel;
            empty.style.cssText = 'padding:12px 10px;font-size:12px;color:#8493A2;';
            dropdown.appendChild(empty);
        } else {
            matches.forEach(function (r, i) {
                var rowEl = document.createElement('button');
                rowEl.type = 'button'; rowEl.textContent = r.name;
                rowEl.title = r.name;
                rowEl.setAttribute('role', 'option');
                rowEl.setAttribute('aria-selected', String(r.name === 当前选中名称));
                rowEl.style.cssText = 'display:block;width:100%;text-align:left;box-sizing:border-box;border:0;border-radius:4px;background:'+
                    (r.name === 当前选中名称 ? '#FFF1F0' : 'transparent') + ';padding:8px 10px;font:12px Arial,Microsoft YaHei,sans-serif;color:#263238;cursor:pointer;white-space:nowrap;overflow:hidden;text-overflow:ellipsis;';
                rowEl.addEventListener('mouseenter', function () { rowEl.style.background = '#F1F4F7'; });
                rowEl.addEventListener('mouseleave', function () { rowEl.style.background = r.name === 当前选中名称 ? '#FFF1F0' : 'transparent'; });
                // pointerdown 防止 input 失焦导致列表在 click 前消失。
                rowEl.addEventListener('pointerdown', function(evt) { if (evt.preventDefault) { evt.preventDefault(); } });
                rowEl.addEventListener('click', function (evt) { if (evt.stopPropagation) { evt.stopPropagation(); } onSelection(r.name); });
                dropdown.appendChild(rowEl);
            });
        }
    }
    function openList(query) {
        renderList(query);
        dropdown.style.display = 'block'; ui.opened = true;
        input.setAttribute('aria-expanded', 'true');
    }
    ui.closeList = closeList;
    function onTyped() {
        当前搜索词 = input.value;
        当前选中名称 = null;
        刷新折线高亮(instance);
        更新搜索状态提示(instance);
        openList(input.value);
    }
    var composing = false;
    input.addEventListener('compositionstart', function() { composing = true; });
    input.addEventListener('compositionend', function() { composing = false; onTyped(); });
    input.addEventListener('input', function() { if (!composing) { onTyped(); } });
    input.addEventListener('focus', function() { openList(''); }); // 单击立即展示全部候选项
    input.addEventListener('click', function() { if (!ui.opened) { openList(''); } });
    input.addEventListener('keydown', function(evt) {
        if (evt.key === 'Escape') { closeList(); evt.stopPropagation(); return; }
        if (evt.key === 'ArrowDown' || evt.key === 'ArrowUp') {
            if (!ui.opened) { openList(''); }
            else if (ui.filteredNames.length) {
                var dx = evt.key === 'ArrowDown' ? 1 : -1;
                ui.activeIndex = ui.activeIndex < 0 ? (dx > 0 ? 0 : ui.filteredNames.length - 1)
                    : (ui.activeIndex + dx + ui.filteredNames.length) % ui.filteredNames.length;
                for (var k = 0; k < dropdown.children.length; k++) {
                    dropdown.children[k].style.background = k === ui.activeIndex ? '#F1F4F7'
                        : (ui.filteredNames[k] === 当前选中名称 ? '#FFF1F0' : 'transparent');
                }
                var child = dropdown.children[ui.activeIndex];
                if (child && typeof child.scrollIntoView === 'function') { child.scrollIntoView({block:'nearest'}); }
            }
            if (evt.preventDefault) { evt.preventDefault(); }
        }
        if (evt.key === 'Enter' && ui.opened && ui.filteredNames.length) {
            onSelection(ui.filteredNames[ui.activeIndex >= 0 ? ui.activeIndex : 0]);
            if (evt.preventDefault) { evt.preventDefault(); }
        }
    });
    arrow.addEventListener('click', function(evt) {
        if (evt.stopPropagation) { evt.stopPropagation(); }
        if (ui.opened) { closeList(); } else { openList(''); input.focus(); }
    });
    clear.addEventListener('click', function(evt) {
        if (evt.stopPropagation) { evt.stopPropagation(); }
        当前选中名称 = null; 当前搜索词 = ''; input.value = '';
        closeList(); 刷新折线高亮(instance); 更新搜索状态提示(instance);
        input.focus();
    });
    ['click', 'pointerdown', 'mousedown', 'wheel'].forEach(function(eventName) {
        host.addEventListener(eventName, function(evt) { evt.stopPropagation(); });
    });
    function outside(evt) {
        if (!ui.opened || !evt || !evt.target) { return; }
        if (typeof host.contains === 'function' && !host.contains(evt.target)) { closeList(); }
    }
    if (typeof document.addEventListener === 'function') { document.addEventListener('pointerdown', outside); }
    ui.cleanup = function() {
        if (typeof document.removeEventListener === 'function') { document.removeEventListener('pointerdown', outside); }
    };
    更新搜索状态提示(instance);
}
