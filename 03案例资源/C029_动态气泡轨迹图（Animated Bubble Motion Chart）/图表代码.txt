// 二、校验与变换：以下逻辑完全不依赖本案例的国家名称或固定年份。
function c029Num(v) {
  if (v === null || v === undefined || String(v).trim() === '') return NaN;
  return Number(v);
}
function c029Encode(s) {
  return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')
    .replace(/"/g,'&quot;').replace(/'/g,'&#39;');
}
function c029CountryName(name) {
  return (setting.countryAliases && setting.countryAliases[name]) || name;
}
function c029RegionName(region) {
  return (setting.regionLabels && setting.regionLabels[region]) || region;
}
function c029Color(region) {
  // 所有地区使用相同颜色：避免视觉关注点被地区调色板分散。
  return setting.neutralColor;
}
function c029Comma(n) {
  return Math.round(Number(n)).toLocaleString('en-US');
}
function c029Population(n) {
  if (n >= 1e8) return (n / 1e8).toFixed(2) + ' 亿人';
  if (n >= 1e4) return (n / 1e4).toFixed(1) + ' 万人';
  return c029Comma(n) + ' 人';
}
function c029Prepare() {
  var fields = [国家,地区,年份,人均GDP,预期寿命,人口];
  if (!fields.every(Array.isArray)) throw new Error('C029：六个字段都必须是一维数组。');
  var size = fields[0].length;
  if (!fields.every(function(x){return x.length === size;})) throw new Error('C029：六个数组长度不一致。');
  var records = [];
  var lookup = Object.create(null), allYears = Object.create(null), byCountry = Object.create(null);
  var regionList = [];
  for (var i=0; i<size; i++) {
    var c = String(国家[i] == null ? '' : 国家[i]).trim();
    var r = String(地区[i] == null ? '' : 地区[i]).trim();
    var year = c029Num(年份[i]);
    var gdp = c029Num(人均GDP[i]);
    var life = c029Num(预期寿命[i]);
    var pop = c029Num(人口[i]);
    if (!c || !r || !Number.isInteger(year) || !Number.isFinite(gdp) ||
        !Number.isFinite(life) || !Number.isFinite(pop) || gdp<=0 || life<=0 || pop<=0)
      throw new Error('C029：第 ' + (i+1) + ' 行无效；国家/地区不能为空，年份必须为整数，指标必须为正数。');
    var key = c + '\u0001' + year;
    if (lookup[key]) throw new Error('C029：重复国家-年份组合 ' + c + ' / ' + year);
    lookup[key] = true;
    if (!byCountry[c]) byCountry[c] = [];
    if (regionList.indexOf(r)<0) regionList.push(r);
    allYears[year] = true;
    var obj={country:c,region:r,year:year,gdp:gdp,life:life,pop:pop};
    byCountry[c].push(obj);
    records.push(obj);
  }
  var years=Object.keys(allYears).map(Number).sort(function(a,b){return a-b;});
  Object.keys(byCountry).forEach(function(c){
    byCountry[c].sort(function(a,b){return a.year-b.year;});
  });
  var ordered=(setting.regionOrder||[]).filter(function(c){return regionList.indexOf(c)>=0;});
  regionList.forEach(function(c){if(ordered.indexOf(c)<0)ordered.push(c);});
  var minGDP=Infinity,maxGDP=0,minLife=Infinity,maxLife=0,maxPop=0;
  records.forEach(function(r){
    minGDP=Math.min(minGDP,r.gdp);maxGDP=Math.max(maxGDP,r.gdp);
    minLife=Math.min(minLife,r.life);maxLife=Math.max(maxLife,r.life);
    maxPop=Math.max(maxPop,r.pop);
  });
  return {records:records,byCountry:byCountry,years:years,regions:ordered,
    countries:Object.keys(byCountry),minGDP:minGDP,maxGDP:maxGDP,
    minLife:minLife,maxLife:maxLife,maxPop:maxPop};
}

function c029BubbleDiameter(pop) {
  return Math.min(setting.maxBubbleSize,Math.max(setting.minBubbleSize,
    setting.minBubbleSize+Math.sqrt(pop/1000000)*setting.bubbleScale));
}
function c029AxisRange() {
  if (!c029Data.records.length) return {xMin:300,xMax:100000,yMin:30,yMax:90};
  var expMin=Math.pow(10,Math.floor(Math.log(c029Data.minGDP)/Math.LN10));
  var expMax=Math.pow(10,Math.ceil(Math.log(c029Data.maxGDP)/Math.LN10));
  if(expMax<=expMin)expMax=expMin*10;
  var minY=Math.max(0,Math.floor((c029Data.minLife-4)/10)*10);
  var maxY=Math.ceil((c029Data.maxLife+4)/10)*10;
  if(maxY<=minY)maxY=minY+10;
  return {
    xMin:setting.xAxisMin==null?Math.max(1,expMin):Number(setting.xAxisMin),
    xMax:setting.xAxisMax==null?expMax:Number(setting.xAxisMax),
    yMin:setting.yAxisMin==null?minY:Number(setting.yAxisMin),
    yMax:setting.yAxisMax==null?maxY:Number(setting.yAxisMax)
  };
}

// 根据当前坐标系、可用空间和附近气泡的遮挡程度，选择关注国家的标签方向。
// 使用实数坐标进行对数换算；类别、年份和国家名称都不参与硬编码判断。
function c029PixelCoordinate(d, axis, width, height) {
  var plotWidth=Math.max(100,width-setting.grid.left-setting.grid.right);
  var plotHeight=Math.max(100,height-setting.grid.top-setting.grid.bottom);
  var dx=Math.log(axis.xMax)-Math.log(axis.xMin);
  var dy=axis.yMax-axis.yMin;
  return {
    x:setting.grid.left+plotWidth*(Math.log(d.gdp)-Math.log(axis.xMin))/dx,
    y:setting.grid.top+plotHeight*(1-(d.life-axis.yMin)/dy)
  };
}
function c029EstimateLabelPosition(d, yearData) {
  if (!setting.focusLabelAvoidance) return 'right';
  var chart=c029State.chart;
  var w=chart&&typeof chart.getWidth==='function'?chart.getWidth():1200;
  var h=chart&&typeof chart.getHeight==='function'?chart.getHeight():820;
  var axis=c029AxisRange();
  var p=c029PixelCoordinate(d,axis,w,h);
  var radius=c029BubbleDiameter(d.pop)/2;
  var name=c029CountryName(d.country);
  var len=Array.from(String(name)).reduce(function(total,ch){return total+(/[\u0000-\u007f]/.test(ch)?0.61:1);},0);
  var labelWidth=Math.max(25,len*setting.labelFontSize+4);
  var labelHeight=setting.labelFontSize+6;
  var margin=7;
  var plot={left:setting.grid.left+4,right:w-setting.grid.right-4,
    top:setting.grid.top+4,bottom:h-setting.grid.bottom-4};
  var candidates=['right','left','top','bottom'];
  var best='right', bestScore=Infinity;
  candidates.forEach(function(dir){
    var x,y;
    if(dir==='right'){x=p.x+radius+margin;y=p.y-labelHeight/2;}
    else if(dir==='left'){x=p.x-radius-margin-labelWidth;y=p.y-labelHeight/2;}
    else if(dir==='top'){x=p.x-labelWidth/2;y=p.y-radius-margin-labelHeight;}
    else{x=p.x-labelWidth/2;y=p.y+radius+margin;}
    var score=0;
    if(x<plot.left) score+=(plot.left-x)*24;
    if(x+labelWidth>plot.right) score+=(x+labelWidth-plot.right)*24;
    if(y<plot.top) score+=(plot.top-y)*24;
    if(y+labelHeight>plot.bottom) score+=(y+labelHeight-plot.bottom)*24;
    yearData.forEach(function(other){
      if(other.country===d.country)return;
      var q=c029PixelCoordinate(other,axis,w,h);
      var r=c029BubbleDiameter(other.pop)/2;
      var nearX=Math.max(x,Math.min(q.x,x+labelWidth));
      var nearY=Math.max(y,Math.min(q.y,y+labelHeight));
      var dist=Math.hypot(nearX-q.x,nearY-q.y);
      if(dist<r+3) score+=(r+3-dist)*(other.country && (setting.labelCountries||[]).indexOf(other.country)>=0?11:3);
    });
    // Prefer lateral placement when scores are comparable; it keeps the label compact.
    if(dir==='top'||dir==='bottom')score+=4;
    if(score<bestScore){bestScore=score;best=dir;}
  });
  return best;
}

// 自定义箭头使用真实轨迹节点转换后的像素坐标，方向不依赖对数轴近似。
// 在轨迹累计长度处放置 1—2 枚小箭头，避免节点密集区出现大量箭头。
function c029ArrowGraphic(params,api,records) {
  if(records.length<3)return null;
  var coords=records.map(function(d){return api.coord([d.gdp,d.life]);});
  var seg=[],total=0;
  for(var i=1;i<coords.length;i++){
    var a=coords[i-1],b=coords[i];
    var length=Math.hypot(b[0]-a[0],b[1]-a[1]);
    seg.push({a:a,b:b,length:length,start:total});total+=length;
  }
  if(total<45)return null;
  var positions=setting.trailArrowFractions||[0.34,0.70];
  var fraction=Number(positions[params.dataIndex]);
  if(!Number.isFinite(fraction))return null;
  var target=total*Math.min(0.92,Math.max(0.08,fraction));
  var targetSeg=seg[seg.length-1];
  for(var j=0;j<seg.length;j++){
    if(target<=seg[j].start+seg[j].length){targetSeg=seg[j];break;}
  }
  if(targetSeg.length<8)return null;
  var t=Math.min(0.82,Math.max(0.18,(target-targetSeg.start)/targetSeg.length));
  var dx=(targetSeg.b[0]-targetSeg.a[0])/targetSeg.length;
  var dy=(targetSeg.b[1]-targetSeg.a[1])/targetSeg.length;
  var x=targetSeg.a[0]+(targetSeg.b[0]-targetSeg.a[0])*t;
  var y=targetSeg.a[1]+(targetSeg.b[1]-targetSeg.a[1])*t;
  var s=Math.max(4,Number(setting.trailArrowSize)||6);
  return {type:'polygon',silent:true,
    shape:{points:[
      [x+dx*s,y+dy*s],
      [x-dx*s*0.75-dy*s*0.54,y-dy*s*0.75+dx*s*0.54],
      [x-dx*s*0.75+dy*s*0.54,y-dy*s*0.75-dx*s*0.54]
    ]},
    style:{fill:setting.focusColor,opacity:0.96,stroke:'#FFFFFF',lineWidth:0.4}};
}

function c029OneDatum(d,focus,yearData) {
  var focused=!!focus && focus===d.country;
  var labelWanted = focused || (setting.labelCountries||[]).indexOf(d.country)>=0;
  return {
    value:[d.gdp,d.life,d.pop,d.country,d.year,d.region],
    name:d.country,
    symbolSize:c029BubbleDiameter(d.pop),
    label:{
      show:!!labelWanted,
      position:focused?c029EstimateLabelPosition(d,yearData||[]):
        (Math.log(d.gdp)>Math.log(c029AxisRange().xMax)*0.85?'left':'right'),distance:7,
      formatter:c029CountryName(d.country),
      color:focused?COLORS.text:COLORS.muted,
      fontSize:setting.labelFontSize,
      fontWeight:focused?'bold':'normal'
    },
    itemStyle:{
      color:focused?setting.focusColor:setting.neutralColor,
      opacity:focused?setting.focusOpacity:(focus?setting.unfocusedOpacity:setting.neutralOpacity),
      borderColor:focused?COLORS.text:COLORS.neutralBorder,
      borderWidth:focused?1.8:0.8
    }
  };
}

function c029SearchMatches(query) {
  var q=String(query||'').trim().toLocaleLowerCase();
  return c029Data.countries.filter(function(name){
    var label=c029CountryName(name);
    var region=c029Data.byCountry[name]&&c029Data.byCountry[name][0]
      ?c029RegionName(c029Data.byCountry[name][0].region):'';

}

function c029SetFocus(name, deferRenderUntilClickEnds) {
  if(name && !c029Data.byCountry[name])return false;
  c029State.focus=name||'';
  if(name){
    var currentYear=c029Data.years[c029State.index];
    if(!c029Data.byCountry[name].some(function(d){return d.year===currentYear;})){
      // 对稀疏数据：选择该国家最近有观测的年份，而不是让搜索结果指向一个空位置。
      var closest=c029Data.byCountry[name].reduce(function(best,d){
        return !best||Math.abs(d.year-currentYear)<Math.abs(best.year-currentYear)?d:best;
      },null);
      if(closest){var ix=c029Data.years.indexOf(closest.year);if(ix>=0)c029State.index=ix;}
    }
  }
  var chart=c029State.chart;
  if(chart&&typeof chart.setOption==='function'){
    // 不要在 ZRender click 分发尚未结束时重建整个图表。
    // 原来同步 notMerge 重绘与空白点击监听叠加，可能出现“点中后立刻清除”。
    var version=++c029State.clickRenderVersion;
    if(deferRenderUntilClickEnds){
      Promise.resolve().then(function(){
        if(version===c029State.clickRenderVersion && c029State.chart===chart){
          chart.setOption(c029BuildOption(),{notMerge:true,lazyUpdate:false});
        }
      });
    }else{
      chart.setOption(c029BuildOption(),{notMerge:true,lazyUpdate:false});
    }
  }
  return true;
}

// 四、统一点击入口：直接监听 ZRender click，不依赖可能被旧版 off('click') 移除的 ECharts 转发。
//    只有注销本图自己注册的回调时才允许 off('click', callback)；禁止无回调的全量 off。
function c029ClickPosition(event) {
  if (!event) return null;
  // ZRender Handler 的鼠标偏移优先在 event.offsetX / offsetY，某些宿主只给 event.event.zrX/zrY。
  var x=event.offsetX, y=event.offsetY;
  if(!Number.isFinite(x) || !Number.isFinite(y)) {
    var mouse=event.event || {};
    x=mouse.zrX; y=mouse.zrY;
  }
  if(!Number.isFinite(x) || !Number.isFinite(y))return null;
  return [x,y];
}
function c029CountryFromElement(chart,target) {
  // ECharts 给散点图图元添加 __ecData；先使用引擎的真实命中结果解决重叠气泡。
  for(var el=target,depth=0;el&&depth<8;el=el.parent,depth++){
    var meta=el.__ecData;
    if(!meta||!Number.isInteger(meta.seriesIndex)||!Number.isInteger(meta.dataIndex))continue;
    try {
      var model=chart.getModel&&chart.getModel();
      var series=model&&model.getSeriesByIndex&&model.getSeriesByIndex(meta.seriesIndex);
      if(!series)continue;
      var id=String(series.id||'');
      if(id!=='c029-active-country'&&id.indexOf('c029-region-')!==0)continue;
      var data=series.getData&&series.getData();
      var raw=data&&data.getRawDataItem&&data.getRawDataItem(meta.dataIndex);
      var name=raw&&raw.name;
      if(!name&&raw&&Array.isArray(raw.value))name=raw.value[3];
      if(name&&c029Data.byCountry[name])return name;
    } catch(error) { /* Power BI 或自定义渲染宿主可能不暴露 getModel；继续尝试几何检测 */ }
  }
  return '';
}
function c029CountryFromPixel(chart,xy) {
  // 退化方案：只使用当年真实可见的气泡及其实际屏幕半径做圆形命中测试。
  // 这使旧版破坏了 ECharts 内部 click 转发的编辑器实例仍然可以响应点击。
  if(!xy||!chart||typeof chart.convertToPixel!=='function')return '';
  var year=c029Data.years[c029State.index];
  var candidates=c029Data.records.filter(function(d){return d.year===year;});
  var best='',bestPriority=-Infinity;
  candidates.forEach(function(d){
    var center;
    try{center=chart.convertToPixel({xAxisIndex:0,yAxisIndex:0},[d.gdp,d.life]);}
    catch(error){return;}
    if(!center||!Number.isFinite(center[0])||!Number.isFinite(center[1]))return;
    var dx=xy[0]-center[0],dy=xy[1]-center[1];
    var distance=Math.sqrt(dx*dx+dy*dy);
    var radius=c029BubbleDiameter(d.pop)*0.5;
    if(distance>radius+1.5)return;
    // 当前关注系列 z-index 更高；重叠时优先命中前景，再选择圆心更近的气泡。
    var priority=(d.country===c029State.focus?1e6:0)-distance/Math.max(1,radius);
    if(priority>bestPriority){best=d.country;bestPriority=priority;}
  });
  return best;
}

}
function c029BindChart(instance) {
  if(!instance||typeof instance.on!=='function'||typeof instance.setOption!=='function')return false;
  c029State.chart=instance;
  var zr=instance.getZr&&instance.getZr();
  if(!zr||typeof zr.on!=='function')return false;
  // 热更新只清除已有 C029 自己的监听器，不能 zr.off('click') 或 instance.off('click')。
  // v0.1.3 曾全量移除 ZRender click 监听，可能误删 ECharts 自身的事件转发。
  var old=instance.__c029Handlers;
  if(old){
    if(old.timeline)instance.off('timelinechanged',old.timeline);
    if(old.click)instance.off('click',old.click);
    if(old.blank)zr.off('click',old.blank);
    if(old.zrClick)zr.off('click',old.zrClick);
  }
  // v0.1.2 的匿名监听无法被安全地精确注销；永远不能为了清理它而清空整个事件类型。
  var handlers={
    timeline:function(event){
      if(event&&Number.isInteger(event.currentIndex))c029State.index=event.currentIndex;
    },
    zrClick:function(event){
      var name=c029CountryAtClick(instance,event);
      if(name){
        c029SetFocus(c029State.focus===name?'':name,true);
        c029SyncSearchFocus();
      } else if(!event || !event.target) {
        // 只处理真正空白，不因坐标轴、标题和滚动控件误清除选择。
        if(c029State.focus){c029SetFocus('',true);c029SyncSearchFocus();}
      }
    }
  };
  instance.on('timelinechanged',handlers.timeline);
  zr.on('click',handlers.zrClick);
  instance.__c029Handlers=handlers;
  instance.__c029Handlers_v013=handlers;
  instance.__c029Bound=true;
  c029MountSearch(instance);
  return true;
}
