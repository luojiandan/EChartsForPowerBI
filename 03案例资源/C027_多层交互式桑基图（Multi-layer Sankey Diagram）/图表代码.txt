function c027Txt(v){return v == null ? '' : String(v).trim();}
function c027Esc(v){return c027Txt(v).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;').replace(/'/g,'&#39;');}
function c027Round(v, n){return Number(Number(v).toFixed(n));}
function c027Fmt(v, n){var p=(n==null?setting.valueDigits:n); return Number(v||0).toLocaleString('zh-CN',{minimumFractionDigits:p,maximumFractionDigits:p});}
function c027Pct(n,d){return d>0?(n*100/d).toFixed(1)+'%':'—';}
function c027Fail(x){throw Error('C027 输入错误：'+x);}
function c027Color(name,index){return setting.nodeColors[name] || setting.fallbackColors[index % setting.fallbackColors.length];}
function c027Label(name){return setting.nodeAliases[name] || name;}

}
function c027BuildData(){
  var arr=[来源节点,目标节点,流量值];
  if(arr.some(function(a){return !Array.isArray(a);})){c027Fail('3个字段必须为一维数组。');}
  if(!arr[0].length||arr.some(function(a){return a.length!==arr[0].length;})){c027Fail('3个数组必须非空且长度相同。');}
  var map=new Map(), links=[],seen=new Set();
  function node(n){if(!map.has(n))map.set(n,{name:n,input:0,output:0,inlinks:[],outlinks:[]});return map.get(n);}
  for(var i=0;i<arr[0].length;i++){
    var a=c027Txt(来源节点[i]),b=c027Txt(目标节点[i]),v=Number(流量值[i]);
    if(!a||!b||a===b)c027Fail('第'+(i+1)+'行来源/目标为空或构成自环。');
    if(!Number.isFinite(v)||v<=0)c027Fail('第'+(i+1)+'行流量必须为正数。');
    var key=JSON.stringify([a,b]);if(seen.has(key))c027Fail('第'+(i+1)+'行的来源—目标重复，请先聚合。');
    seen.add(key);var s=node(a),t=node(b),edge={source:a,target:b,value:v,index:i};
    s.output+=v;t.input+=v;s.outlinks.push(edge);t.inlinks.push(edge);links.push(edge);
  }
  // 使用 Kahn 算法检查有向无环图，防止错误数据导致 Sankey 运行失败。
  var degree=new Map(),queue=[],visited=0;
  map.forEach(function(n){degree.set(n.name,n.inlinks.length);if(n.inlinks.length===0)queue.push(n.name);});
  while(queue.length){var k=queue.shift();visited++;map.get(k).outlinks.forEach(function(e){var next=degree.get(e.target)-1;degree.set(e.target,next);if(next===0)queue.push(e.target);});}
  if(visited!==map.size)c027Fail('流向数据存在环路。桑基图要求无环，请先处理循环流。');
  var all=Array.from(map.values());
  // 默认按初次出现排序；如果提供 nodeDepth，优先遵循设置的层级。
  var order=new Map(all.map(function(n,i){return[n.name,i];}));
  var fallbackDepth=new Map();
  function depthOf(k){
    if(fallbackDepth.has(k))return fallbackDepth.get(k);
    var nd=map.get(k),d=0;if(nd)nd.inlinks.forEach(function(e){d=Math.max(d,depthOf(e.source)+1);});
    fallbackDepth.set(k,d);return d;
  }
  var maxDepth=0;
  all.forEach(function(n){n.depth=Number.isInteger(setting.nodeDepth[n.name])?setting.nodeDepth[n.name]:depthOf(n.name);maxDepth=Math.max(maxDepth,n.depth);});
  links.forEach(function(e){if(map.get(e.target).depth<=map.get(e.source).depth)c027Fail('层级冲突：'+e.source+' → '+e.target+'；请调整 setting.nodeDepth。');});
  // nodes 原始顺序决定 layoutIterations=0 时的上下位置；保持初次出现，减少重排。
  var originals=all.filter(function(n){return n.inlinks.length===0;});
  var originTotal=originals.reduce(function(v,n){return v+n.output;},0);
  var endpoints=all.filter(function(n){return n.outlinks.length===0;});
  var endpointTotal=endpoints.reduce(function(v,n){return v+n.input;},0);
  var primary=map.get(setting.primaryOutcome),secondary=map.get(setting.secondaryOutcome);
  return {map:map, nodes:all,links:links,originals:originals,endpoints:endpoints,originTotal:originTotal,endpointTotal:endpointTotal,
    primary:primary,secondary:secondary,maxDepth:maxDepth,order:order};
}

function c027SelectionGraph(){
  var sel=c027State.selected,D=c027Data;
  if(!sel)return null;
  var nodes=new Set(),edges=new Set();
  if(sel.kind==='edge'){
    var only=D.links[sel.index];if(!only)return null;
    nodes.add(only.source);nodes.add(only.target);edges.add(only.index);
    return {nodes:nodes,edges:edges};
  }
  if(!D.map.has(sel.name))return null;
  nodes.add(sel.name);
  // 对所选节点执行上游+下游递归：高亮完整流向链，而不是仅突出直接相邻节点。
  var backward=[sel.name],forward=[sel.name],seenB=new Set(),seenF=new Set();
  while(backward.length){var a=backward.pop();if(seenB.has(a))continue;seenB.add(a);D.map.get(a).inlinks.forEach(function(e){edges.add(e.index);nodes.add(e.source);backward.push(e.source);});}
  while(forward.length){var b=forward.pop();if(seenF.has(b))continue;seenF.add(b);D.map.get(b).outlinks.forEach(function(e){edges.add(e.index);nodes.add(e.target);forward.push(e.target);});}
  return {nodes:nodes,edges:edges};
}
function c027RichTooltip(p){
  var D=c027Data;
  if(p.dataType==='edge'){
    var d=p.data||{},from=D.map.get(d.source),to=D.map.get(d.target);
    if(!from||!to)return '';
    return '<div style="font-size:13px;font-weight:700;color:#263238">'+c027Esc(c027Label(d.source))+' <span style="color:#8493A2">→</span> '+c027Esc(c027Label(d.target))+'</div>'+
      '<div style="margin-top:9px;border-top:1px solid #E5E9ED;padding-top:9px">流量 <strong style="float:right">'+c027Fmt(d.value)+' '+c027Esc(setting.unit)+'</strong></div>'+
      '<div style="margin-top:5px;color:#8493A2">占来源总流出 <b style="float:right;color:#263238">'+c027Pct(d.value,from.output)+'</b></div>'+
      '<div style="margin-top:5px;color:#8493A2">占目标总流入 <b style="float:right;color:#263238">'+c027Pct(d.value,to.input)+'</b></div>';
  }
  if(p.dataType==='node'){
    var name=p.data&&p.data.name,nd=D.map.get(name);if(!nd)return '';
    var side=(nd.input>0&&nd.output>0)?Math.max(nd.input,nd.output):nd.input||nd.output;
    var arr=nd.outlinks.length?nd.outlinks:nd.inlinks;
    var sorted=arr.slice().sort(function(a,b){return b.value-a.value;}).slice(0,5);
    return '<div style="font-size:14px;font-weight:700;color:#263238">'+c027Esc(c027Label(name))+'</div>'+
      '<div style="padding:9px 0 7px;border-bottom:1px solid #E5E9ED">节点规模：<b>'+c027Fmt(side)+' '+c027Esc(setting.unit)+'</b></div>'+
      (nd.input>0?'<div style="margin-top:7px;color:#8493A2">总流入：'+c027Fmt(nd.input)+'</div>':'')+
      (nd.output>0?'<div style="margin-top:4px;color:#8493A2">总流出：'+c027Fmt(nd.output)+'</div>':'')+
      sorted.map(function(x){var label=nd.outlinks.length?x.target:x.source;return '<div style="display:flex;justify-content:space-between;gap:18px;margin-top:6px"><span>'+c027Esc(c027Label(label))+'</span><b>'+c027Fmt(x.value)+'</b></div>';}).join('')+
      '<div style="font-size:11px;color:#8493A2;margin-top:8px">点击节点可固定高亮整条上、下游链路</div>';
  }
  return '';
}

function c027BuildSeries(){
  var D=c027Data,selection=c027SelectionGraph(),maxDepth=D.maxDepth;
  var nodes=D.nodes.map(function(n,index){
    var active=!selection||selection.nodes.has(n.name),low=Math.max(n.input,n.output),depth=n.depth;
    var color=c027Color(n.name,index);
    var pos=depth===0?'left':'right';
    return {name:n.name,depth:depth,value:low,itemStyle:{color:color,borderWidth:0,borderRadius:setting.nodeRound,opacity:active?1:setting.fadedNodeOpacity},
      label:{show:low>=setting.minLabelValue,position:pos,distance:6,color:active?setting.text:'#9EABB5',fontSize:setting.labelFontSize,
        fontWeight:selection&&active?600:500,formatter:(setting.showNodeValue?c027Label(n.name)+' '+c027Fmt(low,1):c027Label(n.name))}};
  });
  var links=D.links.map(function(e){
    var on=!selection||selection.edges.has(e.index);
    // 来源节点决定前两段的颜色；进入末端结果的连线改用中性灰色。
    var sourceDepth=D.map.get(e.source).depth;
    var color=sourceDepth>=2?c027Color(e.target,0):c027Color(e.source,D.order.get(e.source));
    return {source:e.source,target:e.target,value:e.value,lineStyle:{color:color,opacity:selection?(on?setting.highlightLinkOpacity:setting.fadedLinkOpacity):setting.normalLinkOpacity}};
  });
  return {id:'c027-main',name:'能源流向',type:'sankey',orient:'horizontal',nodeAlign:setting.nodeAlign,
    draggable:setting.draggable,layoutIterations:setting.layoutIterations,
    left:c027Layout().left,right:c027Layout().right,top:c027Layout().top,bottom:c027Layout().bottom,
    nodeWidth:setting.nodeWidth,nodeGap:setting.nodeGap,
    animation:true,animationDuration:650,animationDurationUpdate:220,
    label:{color:setting.text,fontSize:setting.labelFontSize,fontFamily:'sans-serif'},
    lineStyle:{curveness:setting.lineCurveness,opacity:setting.normalLinkOpacity,color:'source'},
    emphasis:{focus:setting.enableHoverFocus?'adjacency':'none',lineStyle:{opacity:0.9}},
    blur:{itemStyle:{opacity:setting.fadedNodeOpacity},lineStyle:{opacity:setting.fadedLinkOpacity},label:{opacity:0.60}},
    data:nodes,links:links};
}

// ===== 05 旧案例兼容清理：仅作用于当前 ECharts 实例，不操作页面其他图表 =====
function c027CleanupLegacyC026(instance){
  if(!instance)return;
  var host=typeof instance.getDom==='function'?instance.getDom():null;
  if(host){
    // C026 v0.5.7 保存了完整销毁回调：同时移除搜索框和 document 级别的指针监听。
    if(typeof host.__c026SearchFinderCleanup==='function'){
      try{host.__c026SearchFinderCleanup();}catch(_e){}
    }
    // 兼容没有销毁回调的旧页面，仅清理明确标记为 C026 的元素。
    if(typeof host.querySelectorAll==='function'){
