// ========== 通用计算：去空值、同名汇总、排序、累计和阈值 ===========
function c030NumText(n, precision) {
  return Number(n).toLocaleString('zh-CN', {
    minimumFractionDigits: precision,
    maximumFractionDigits: precision
  });
}
function c030Safe(text) {
  return String(text).replace(/[&<>"']/g, function(c) {
    return {'&':'&amp;', '<':'&lt;', '>':'&gt;', '"':'&quot;', "'":'&#39;'}[c];
  });
}
function c030BuildModel(labels, values, config) {
  if (!Array.isArray(labels) || !Array.isArray(values) || labels.length !== values.length) {
    throw new Error('对象与数值数组的长度必须一致。');
  }
  if (!(config.threshold > 0 && config.threshold <= 1)) {
    throw new Error('setting.threshold 应大于0且不超过1。');
  }
  var map = Object.create(null);
  for (var i = 0; i < labels.length; i++) {
    var name = String(labels[i] == null ? '' : labels[i]).trim();
    var raw = values[i];
    if (!name || raw === '' || raw == null) continue;
    var number = Number(raw);
    if (!Number.isFinite(number) || number < 0) {
      throw new Error('第' + (i + 1) + '行数值无效：只允许非负有限数字。');
    }
    if (number === 0) continue;
    map[name] = (map[name] || 0) + number;
  }
  var rows = Object.keys(map).map(function(name) {return {name:name,value:map[name]};});
  rows.sort(function(a,b){return b.value - a.value || a.name.localeCompare(b.name, 'zh-CN');});
  var sampleTotal = rows.reduce(function(s,row){return s + row.value;},0);
  var providedTotal = config.totalValue == null || config.totalValue === '' ? null : Number(config.totalValue);
  if (providedTotal != null && (!Number.isFinite(providedTotal) || providedTotal <= 0)) {
    throw new Error('setting.totalValue 必须是正数或 null。');
  }
  var denominator = providedTotal == null ? sampleTotal : providedTotal;
  if (denominator + 1e-7 < sampleTotal) {
    throw new Error('setting.totalValue 小于所展示的数值总和。替换业务数据时请改为 null 或正确总体量。');
  }
  var accum = 0, thresholdIndex = -1;
  rows.forEach(function(row,i){
    accum += row.value;
    row.rank = i + 1;
    row.share = denominator > 0 ? row.value / denominator * 100 : 0;
    row.cumulative = denominator > 0 ? accum / denominator * 100 : 0;
    if (thresholdIndex < 0 && row.cumulative + 1e-9 >= config.threshold * 100) thresholdIndex = i;
  });
  return {
    rows:rows, sampleTotal:sampleTotal, denominator:denominator,
    displayCoverage: denominator > 0 ? sampleTotal / denominator * 100 : 0,
    thresholdIndex:thresholdIndex,
    thresholdCumulative:thresholdIndex >= 0 ? rows[thresholdIndex].cumulative : null,
    externalDenominator:providedTotal != null
  };
}

// ========== 用户在图面上直接调整阈值（不使用额外 DOM） ==========
function c030ClampThresholdPercent(input) {
  var ctl = setting.thresholdControl || {};
  var lo = Math.max(1, Math.min(100, Number(ctl.minPercent) || 1));
  var hi = Math.max(lo, Math.min(100, Number(ctl.maxPercent) || 100));
  var n = Number(input);
  if (!Number.isFinite(n)) return Math.round(setting.threshold * 100);
  return Math.max(lo, Math.min(hi, Math.round(n)));
}
function c030CurrentThresholdPercent() {
  return Math.round(setting.threshold * 100);
}
function c030SetThresholdPercent(percent) {
  var nextPercent = c030ClampThresholdPercent(percent);
  if (nextPercent === c030CurrentThresholdPercent()) return false;
  // 更新统一的 setting 配置，模型、交点、柱体和标题均从这里重新计算。
  setting.threshold = nextPercent / 100;
  c030Model = c030BuildModel(对象, 数值, setting);
  var instance = c030ChartInstance;
  if (instance && typeof instance.setOption === 'function') {
    var nextOption = c030BuildOption();
    // 保存当前滑块浏览区间：修改阈值时，不让已经滑动到第 N 个类别的视图跳回开头。
    if (typeof instance.getOption === 'function') {
      var prev = instance.getOption();
      if (prev && Array.isArray(prev.dataZoom) && Array.isArray(nextOption.dataZoom)) {
        nextOption.dataZoom.forEach(function(zoom, i) {
          var old = prev.dataZoom[i];
          if (old && Number.isFinite(Number(old.start)) && Number.isFinite(Number(old.end))) {
            zoom.start = Number(old.start);
            zoom.end = Number(old.end);
          }
        });
      }
    }
    instance.setOption(nextOption, true);
  }
  return true;
}
// 左上角阈值增减按钮：不增加右侧面板、滑块或额外 HTML。
// 使用有形状的 ECharts graphic rect 接收点击，text 仅显示字符，避免事件穿透或重复触发。
function c030ThresholdGraphic() {
  var ctl = setting.thresholdControl || {};
  if (ctl.show === false) return null;
  var x = Number.isFinite(Number(ctl.left)) ? Number(ctl.left) : 188;
  var top = Number.isFinite(Number(ctl.top)) ? Number(ctl.top) : 101;
  var w = Math.max(16, Number(ctl.buttonWidth) || 20);
  var h = Math.max(14, Number(ctl.buttonHeight) || 16);
  var gap = Math.max(1, Number(ctl.gap) || 2);
  var current = c030CurrentThresholdPercent();
  var step = Math.max(1, Math.round(Number(ctl.stepPercent) || 1));
  var low = Math.max(1, Math.min(100, Number(ctl.minPercent) || 1));
  var high = Math.max(low, Math.min(100, Number(ctl.maxPercent) || 100));
  var normalColor = setting.colors.text;
  var disabledColor = '#C7CED4';
  var outlineColor = '#E5E9ED';

  function c030Button(id, y, symbol, enabled, delta) {
    return {
      id: id, type: 'group', z: 100,
      children: [
        {
          type: 'rect', cursor: enabled ? 'pointer' : 'default',
          shape: {x:x, y:y, width:w, height:h, r:4},
          style: {fill:enabled ? '#F8F9FA' : '#FBFCFD', stroke:outlineColor, lineWidth:0.8},
          onclick: enabled ? function(){ c030SetThresholdPercent(current + delta); } : undefined
        },
        {
          type:'text', silent:true, style:{
            x:x+w/2, y:y+h/2, text:symbol,
            textAlign:'center', textVerticalAlign:'middle',
            font:'600 13px sans-serif', fill:enabled ? normalColor : disabledColor
          }
        }
      ]
    };
  }
  return {
    id:'c030-threshold-control', type:'group', z:100,
    children:[
      c030Button('c030-threshold-increase', top, '+', current < high, step),
      c030Button('c030-threshold-decrease', top+h+gap, '−', current > low, -step)
    ]
  };
}

function c030BarData() {
  return c030Model.rows.map(function(row,i) {
    var color = c030SelectedIndex === i ? setting.colors.focus :
      ((c030Model.thresholdIndex < 0 || i <= c030Model.thresholdIndex) ? setting.colors.barKey : setting.colors.barOther);
    return {value: row.value, itemStyle: {color:color}};
  });
}
function c030Tooltip(params) {
  var param = Array.isArray(params) ? params[0] : params;
  var i = param && typeof param.dataIndex === 'number' ? param.dataIndex : -1;
  var row = c030Model.rows[i];
  if (!row) return '';
  var approximate = c030Model.externalDenominator && setting.totalValueIsApproximate;
  var line = function(label, value) {
    return '<div style="display:flex;justify-content:space-between;gap:20px;margin-top:5px;">'
      + '<span style="color:' + setting.colors.sub + '">' + label + '</span>'
      + '<strong style="color:' + setting.colors.text + '">' + value + '</strong></div>';
  };
  var html = '<div style="padding:4px 6px;min-width:190px;line-height:1.5;">'
    + '<div style="font-size:14px;font-weight:700;color:'+setting.colors.text+';">'
    + c030Safe(row.name) + ' <span style="color:'+setting.colors.sub+';font-weight:400;">第' + row.rank + '位</span></div>';
  html += line(c030Safe(setting.valueLabel), c030NumText(row.value, setting.decimals) + ' '+c030Safe(setting.valueUnit));
  html += line('单项占总体', (approximate ? '约' : '') + row.share.toFixed(2) + '%');
  html += line('累计贡献', (approximate ? '约' : '') + row.cumulative.toFixed(2) + '%');
  if (c030Model.thresholdIndex >= 0) {
    html += '<div style="padding-top:8px;margin-top:8px;border-top:1px solid #E5E9ED;color:'+setting.colors.sub+';">'
      + (i <= c030Model.thresholdIndex ? '处于达到阈值所需的前' + (c030Model.thresholdIndex + 1) + '项' : '在' + (setting.threshold * 100).toFixed(0) + '%阈值之后') + '</div>';
  }
  return html + '</div>';
}
function c030Text(left,top,text,font,color) {
  return {type:'text',left:left,top:top,silent:true,style:{text:text,font:font,fill:color}};
}
// 与柱顶数值采用同一种千分位格式；小数值则保留有效的小数位，避免全变成 0。
function c030AxisNumber(value) {
  var largest = c030Model.rows.length ? c030Model.rows[0].value : 0;
  var decimals = largest < 1 && largest > 0
    ? Math.min(8, Math.max(2, Math.ceil(-Math.log(largest) / Math.LN10) + 1))
    : largest < 10 ? 2 : largest < 100 ? 1 : 0;
  return c030NumText(value, decimals);
}
function c030CoverageLabel(rows, model, config) {
  if (!rows.length) return '暂无有效数据';
  // 外部总体与输入数组求和是两种不同口径，不将“已加载数据”误写成“当前缩放区间”。
  if (model.externalDenominator) {
    return '前' + rows.length + '项覆盖' + String(config.totalScopeLabel || '总体');
  }
  return '全部' + rows.length + '项占合计';
}
// 对标题与说明中的阈值、类别数量、覆盖率占位符进行动态替换。
function c030DynamicText(text) {
  return String(text == null ? '' : text)
    .replace(/\{threshold\}/g, (setting.threshold * 100).toFixed(0))
    .replace(/\{count\}/g, String(c030Model.rows.length))
    .replace(/\{coverage\}/g, c030Model.displayCoverage.toFixed(1));
}

    {type:'rect',left:30,top:18,silent:true,shape:{width:20,height:4},style:{fill:setting.colors.threshold}},

    c030Text(30,66,c030DynamicText(setting.subtitle),'12px sans-serif',setting.colors.sub),
    c030Text(30,112,'累计贡献阈值','12px sans-serif',setting.colors.sub),
    c030Text(125,107,(setting.threshold*100).toFixed(0)+'%','700 23px sans-serif',setting.colors.threshold),
    c030Text(230,112,thresholdLabel,'12px sans-serif',setting.colors.sub),
    c030Text(335,107,thresholdNumber,'700 19px sans-serif',setting.colors.text),
    c030Text(510,112,coverageLabel,'12px sans-serif',setting.colors.sub),
    c030Text(642,108,coverageValue,'700 18px sans-serif',setting.colors.text),
    c030Text(30,163,setting.valueLabel + '（' + setting.valueUnit + '）','11px sans-serif',setting.colors.sub),
    {type:'text',right:62,top:163,silent:true,style:{text:'累计贡献率',font:'11px sans-serif',fill:setting.colors.sub,textAlign:'right'}},
  ];

  }
  if (c030SelectedIndex >= 0 && rows[c030SelectedIndex]) {
    graphics.push({type:'text',right:62,top:113,silent:true,style:{text:'关注：' + rows[c030SelectedIndex].name,fill:setting.colors.focus,font:'600 12px sans-serif',textAlign:'right'}});

      splitLine:{show:false}
    }],

      {type:'value',name:'',min:0,scale:false,axisLine:{show:false},axisTick:{show:false},
       axisLabel:{color:setting.colors.sub,fontSize:11,formatter:c030AxisNumber},

}

function c030Select(index) {
  c030SelectedIndex = c030SelectedIndex === index ? -1 : index;
  if (!c030ChartInstance) return;
  // 只更新所需的数据和动态提示，不重建坐标轴/缩放状态。
  var partial = {series:[{id:'c030-bars',data:c030BarData()}]};
  c030ChartInstance.setOption(partial,false);
}
function c030BindChart(instance) {
  if (!instance || typeof instance.on !== 'function') return;
  if (c030ChartInstance && c030BoundHandler) c030ChartInstance.off('click',c030BoundHandler);
  c030ChartInstance=instance;
  c030BoundHandler=function(p){
    if (!p || !p.seriesId || (p.seriesId !== 'c030-bars' && p.seriesId !== 'c030-cumulative'))return;
    if (typeof p.dataIndex === 'number') c030Select(p.dataIndex);
  };
  instance.off('click',c030BoundHandler);
