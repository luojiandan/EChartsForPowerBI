// 三、输入校验与日期处理：阶段名称、日期和任务数量均不写死
const DAY_MS = 24 * 60 * 60 * 1000;
function parseDay(value, field, label, allowBlank) {
  if (value === null || value === undefined || String(value).trim() === '') {
    if (allowBlank) return null;
    throw new Error('“' + label + '”的' + field + '不能为空');
  }
  let raw;
  if (value instanceof Date && Number.isFinite(value.getTime())) {
    raw = [value.getFullYear(), String(value.getMonth()+1).padStart(2,'0'),
      String(value.getDate()).padStart(2,'0')].join('-');
  } else {
    raw = String(value).trim().slice(0,10).replace(/\//g,'-');
  }
  const m = /^(\d{4})-(\d{1,2})-(\d{1,2})$/.exec(raw);
  if (!m) throw new Error('“'+label+'”的'+field+'格式无效：'+String(value));
  const y=+m[1],mo=+m[2],d=+m[3];
  const ms=Date.UTC(y,mo-1,d), test=new Date(ms);
  if (test.getUTCFullYear()!==y || test.getUTCMonth()!==mo-1 || test.getUTCDate()!==d)
    throw new Error('“'+label+'”的'+field+'不是有效日期：'+raw);
  return ms;
}
function dayText(ms) {
  if (ms===null || ms===undefined) return '—';
  const d=new Date(ms);
  return d.getUTCFullYear()+'-'+String(d.getUTCMonth()+1).padStart(2,'0')+'-'+String(d.getUTCDate()).padStart(2,'0');
}
function countDays(start,end) {return start===null||end===null ? null : Math.round((end-start)/DAY_MS)+1;}
function escapeHtml(v) {
  return String(v==null?'':v).replace(/[&<>"']/g,function(c){
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
  });
}
function readableNumber(value) {return value===null?'—':(Number.isInteger(value)?String(value):value.toFixed(1));}
function optionalColumn(arr,name,length,blank) {
  if (!Array.isArray(arr)) throw new Error(name+'必须是数组');
  if (arr.length===0) return Array(length).fill(blank);
  if (arr.length!==length) throw new Error(name+'数组长度与任务名称不一致');
  return arr;
}

  const start=parseDay(planStarts[i],'计划开始',text,false);
  const end=parseDay(planEnds[i],'计划结束',text,false);
  if (start>end) throw new Error('任务“'+text+'”计划结束早于开始');
  const actualStart=parseDay(actualStartValues[i],'实际开始',text,true);
  const actualEnd=parseDay(actualEndValues[i],'实际结束',text,true);
  if (actualEnd!==null&&(actualStart===null||actualEnd<actualStart))
    throw new Error('任务“'+text+'”的实际结束与实际开始不匹配');
  const isBlank=progressValues[i]===null||progressValues[i]===undefined||String(progressValues[i]).trim()==='';
  const progress=isBlank ? (actualEnd!==null?100:null) : Number(progressValues[i]);
  if (progress!==null&&(!Number.isFinite(progress)||progress<0||progress>100))

  const type=String(typeValues[i]==null?'':typeValues[i]).trim().toLowerCase();

  if(isMilestone&&start!==end) throw new Error('里程碑“'+text+'”的计划开始、结束须相同');

  return {index:i,name:text,phase:phase,planStart:start,planEnd:end,
    actualStart:actualStart,actualEnd:actualEnd,progress:progress,isMilestone:isMilestone};
});

// 四、动态行与阶段汇总：折叠保留汇总条，统计口径不随折叠变化
const collapsed=new Set((Array.isArray(setting.initiallyCollapsedPhases)?setting.initiallyCollapsedPhases:[])
  .map(function(v){return String(v);}));
function groupSummary(group){
  const tt=group.tasks,started=tt.filter(t=>t.actualStart!==null),ended=tt.filter(t=>t.actualEnd!==null);
  const dates=tt.map(t=>t.planStart),ends=tt.map(t=>t.planEnd);
  const known=tt.filter(t=>t.progress!==null);
  const average=known.length?Math.round(known.reduce((s,t)=>s+t.progress,0)/known.length):null;
  const overdue=tt.filter(t=>t.lateDays>0).length;
  return {phase:group.name,name:setting.stageSummaryText+'（'+tt.length+'项）',
    isSummary:true,group:group,planStart:Math.min.apply(null,dates),planEnd:Math.max.apply(null,ends),
    actualStart:started.length?Math.min.apply(null,started.map(t=>t.actualStart)):null,
    actualEnd:ended.length===tt.length?Math.max.apply(null,ended.map(t=>t.actualEnd)):null,
    progress:average,overdue:overdue,completed:ended.length===tt.length,
    color:overdue>0?setting.colors.negative:(ended.length===tt.length?setting.colors.positive:setting.colors.inProgress)};
}
function makeRows(){
  const rows=[];
  groups.forEach(function(g){
    const folded=collapsed.has(g.name);
    if(folded) rows.push({kind:'summary',task:groupSummary(g),group:g});
    else g.tasks.forEach(t=>rows.push({kind:'task',task:t,group:g}));
  });
  // 保留折叠后的汇总行，确保阶段可以重新展开，且不会丢失阶段时间跨度。
  return rows.map(function(r,i){return Object.assign({},r,{rowIndex:i});});
}

// 五、绘图辅助：计划/实际不混淆；聚合时间条与任务 Tooltip 均为通用计算
function rectClipped(x0,x1,y,h,coord){
  const l=Math.max(coord.x,Math.min(x0,x1)),r=Math.min(coord.x+coord.width,Math.max(x0,x1));
  return r>l?{x:l,y:y,width:r-l,height:h,r:Math.min(4,h/2)}:null;
}
function addDiamond(children,x,cy,size,color,coord,outlined){
  if(x<coord.x-5||x>coord.x+coord.width+5)return;
  const h=size/2;
  children.push({type:'polygon',shape:{points:[[x,cy-h],[x+h,cy],[x,cy+h],[x-h,cy]]},
    style:{fill:color,stroke:outlined?setting.colors.text:'#fff',lineWidth:outlined?1.2:.8}});
}
function taskStatus(t){
  const p=t.progress===null?'进度未提供':readableNumber(t.progress)+'%';
  if(t.lateDays>0)return (t.actualStart===null?'逾期未启动':'延期')+' +'+t.lateDays+'天 · '+p;
  if(t.completed)return '按期 · '+p;
  if(t.actualStart!==null)return '进行中 · '+p;
  return '未开始'+(t.progress===null?'':' · '+p);
}
function summaryStatus(t){
  return (t.overdue?'延期 '+t.overdue+'项':'无延期')+' · '+(t.progress===null?'进度未提供':'均进度 '+t.progress+'%');
}
function tooltipRow(label,value,valueColor){
  const color=valueColor||setting.colors.text;
  return '<div style="display:flex;justify-content:space-between;gap:18px;line-height:22px">'+
    '<span style="font-size:12px;color:#8493A2">'+escapeHtml(label)+'</span>'+
    '<strong style="font-size:12px;color:'+color+';font-weight:600;text-align:right">'+escapeHtml(value)+'</strong></div>';
}
// 阶段汇总 Tooltip 独立于任务 Tooltip；展开及折叠时均使用本阶段全部任务计算。
function phaseTooltip(group){
  const members=group.tasks, t=groupSummary(group);
  const allDone=members.filter(x=>x.completed).length;
  const actualDates=members.filter(x=>x.actualStart!==null).map(x=>x.actualStart);
  const header='<div style="font-size:13px;font-weight:700;border-bottom:1px solid #E5E9ED;padding-bottom:6px;margin-bottom:6px">'+
    escapeHtml(group.name+' · '+setting.stageSummaryText)+'</div>';
  return '<div style="min-width:260px;color:#263238">'+header+
    tooltipRow('阶段任务',members.length+' 项')+
    tooltipRow('已完成 / 进行中 / 未开始',allDone+' / '+members.filter(x=>!x.completed&&x.actualStart!==null).length+' / '+members.filter(x=>!x.completed&&x.actualStart===null).length)+
    tooltipRow('延期风险',t.overdue+' 项',t.overdue>0?setting.colors.negative:null)+
    tooltipRow('计划跨度',dayText(t.planStart)+' ～ '+dayText(t.planEnd))+
    tooltipRow('最早实际启动',actualDates.length?dayText(Math.min.apply(null,actualDates)):'—')+
    tooltipRow('平均完成率',t.progress===null?'未提供':t.progress+'%')+
    tooltipRow('统计基准日',dayText(asOf))+'</div>';
}
function ganttTooltip(params){
  // phaseLabels 拥有独立数据项；阶段标题悬停不再映射为第一项任务。
  const phaseIndex=params&&params.data&&params.data.phaseIndex;
  if(Number.isInteger(phaseIndex)&&groups[phaseIndex])return phaseTooltip(groups[phaseIndex]);
  const idx=params&&params.data&&params.data.rowIndex;
  if(!Number.isInteger(idx)||!visibleRows[idx])return '';
  const row=visibleRows[idx],t=row.task;
  if(row.kind==='summary')return phaseTooltip(row.group);
  const header='<div style="font-size:13px;font-weight:700;border-bottom:1px solid #E5E9ED;padding-bottom:6px;margin-bottom:6px">'+escapeHtml(t.name)+'</div>';
  const last=t.actualEnd===null&&t.actualStart!==null?asOf:t.actualEnd;
  const actual=t.actualStart===null?'尚未开始':dayText(t.actualStart)+' ～ '+
    (t.actualEnd===null?dayText(asOf)+'（执行中）':dayText(t.actualEnd));
  return '<div style="min-width:260px;color:#263238">'+header+
    tooltipRow('所属阶段',t.phase)+tooltipRow('任务类型',t.isMilestone?'里程碑':'普通任务')+
    tooltipRow('计划日期',dayText(t.planStart)+' ～ '+dayText(t.planEnd))+
    tooltipRow('计划工期',countDays(t.planStart,t.planEnd)+' 天')+
    tooltipRow('实际执行',actual)+
    (t.actualStart===null?'':tooltipRow('已历经时间',countDays(t.actualStart,last)+' 天'))+
    tooltipRow('完成率',t.progress===null?'未提供':readableNumber(t.progress)+'%')+
    '<div style="display:flex;justify-content:space-between;gap:18px;line-height:22px"><span style="font-size:12px;color:#8493A2">任务状态</span><strong style="color:'+t.color+';font-size:12px">'+escapeHtml(t.state)+'</strong></div>'+
    (t.lateDays>0?tooltipRow(t.completed?'实际延期':'截至基准日延期',t.lateDays+' 天',setting.colors.negative):'')+

// 六、动态 ECharts 配置：一张 Grid，阶段左栏独立，折叠后使用阶段汇总时间条
function chartSize(){
  if(typeof myChart!=='undefined'&&myChart&&typeof myChart.getHeight==='function')
    return {width:myChart.getWidth(),height:myChart.getHeight()};
  return {width:1200,height:820};
}
function layoutForRows(){
  const h=chartSize().height;
  const maxAvailable=Math.max(80,h-setting.grid.top-setting.grid.bottom);
  const desired=Math.max(1,visibleRows.length)*setting.maxTaskRowHeight;
  const gridHeight=Math.min(maxAvailable,Math.max(74,desired));
  return {height:gridHeight,bottom:Math.max(setting.grid.bottom,h-setting.grid.top-gridHeight)};
}
function textTick(value) {
  const d=new Date(value), m=d.getUTCMonth()+1, day=d.getUTCDate();
  const withYear=setting.crossYearDateStyle==='always' ||
    (setting.crossYearDateStyle==='auto' && crossesYear);
  // 同一时间刻度仍以日为精度；跨年时明确显示年份，避免 12 月/1 月混淆。
  return withYear ? d.getUTCFullYear()+'/'+m+'/'+day : m+'/'+day;
}

// 七、交互：阶段列使用 ZRender 像素命中；保留底部滑块与图内缩放
function activeChart(){return typeof myChart!=='undefined'&&myChart&&typeof myChart.setOption==='function'?myChart:null;}
function readZoom(chart){
  try{
    const dz=chart.getOption().dataZoom;
    if(Array.isArray(dz)&&dz.length){
      const z=dz[0],lo=Number(z.start),hi=Number(z.end);
      if(Number.isFinite(lo)&&Number.isFinite(hi)&&lo>=0&&hi<=100&&hi>lo)return [lo,hi];
    }
  }catch(e){}
  return zoomPercent;
}
function redraw(){
  const chart=activeChart();if(!chart)return;
  const cfg=chartOptions();
  // 使用 notMerge 替换旧可见行、坐标轴与阶段标记；保留 zoomPercent 与 collapsed 状态。
  chart.setOption(cfg,{notMerge:true,lazyUpdate:false});
  option=cfg;
}
function togglePhase(phaseName){
  if(!setting.phaseCollapsible)return;
  const chart=activeChart();if(chart)zoomPercent=readZoom(chart);
  if(collapsed.has(phaseName))collapsed.delete(phaseName);else collapsed.add(phaseName);
  visibleRows=makeRows();redraw();
}
// 折叠点击必须命中鼠标真实位置，而非依赖 custom series 子元素的 info 透传。
// ECharts 的 series click 参数主要面向数据项，不能稳定读取 renderItem 中任意子元素的 info。
// 使用 ZRender 全局点击事件 + 当前可见分组的像素区域；对 canvas/svg 均通用。
function locatePhaseByPoint(chart, x, y) {
  if (!setting.phaseCollapsible || !Number.isFinite(x) || !Number.isFinite(y) || !visibleRows.length) return null;
  // 与 renderItem 中三栏坐标共用相同配置，调整列宽也不会使折叠点击错位。
  const phaseX = setting.grid.left - setting.leftLabelGap - setting.statusColumnWidth -
    setting.statusColumnGap - setting.taskColumnWidth -
    setting.phaseColumnGap - setting.phaseColumnWidth;
  if (x < phaseX - 7 || x > phaseX + setting.phaseColumnWidth + 5) return null;
  // 当前任务行高度由折叠状态及容器尺寸决定；重新计算避免 resize 后点击错位。
  const rowHeight = layoutForRows().height / visibleRows.length;
  let first = 0;
  while (first < visibleRows.length) {
    const group = visibleRows[first].group;
    let last = first;
    while (last + 1 < visibleRows.length && visibleRows[last + 1].group === group) last++;
    // category 轴 inverse:true：第 i 行中心 = gridTop + (i + 0.5) × rowHeight。
    const centerY = setting.grid.top + (first + last + 1) * rowHeight / 2;
    if (Math.abs(y - centerY) <= 15) return group.name;
    first = last + 1;
  }
  return null;
}
function bindGanttInteractions(chart){
  if (!chart || typeof chart.on !== 'function') return;
  const zr = typeof chart.getZr === 'function' ? chart.getZr() : null;
  const old = chart.__c021Handlers;
  // 清除之前版本的 click/zoom 和 ZRender 监听，防止脚本重新执行后重复折叠。
  if (old) {
    if (old.click) chart.off('click', old.click);
    if (old.zoom) chart.off('datazoom', old.zoom);
    if (zr && old.zrClick) zr.off('click', old.zrClick);
  }
  function onZoom(){zoomPercent=readZoom(chart);}
  function onZrClick(event){
    // offsetX / offsetY 为 ZRender 相对当前画布的像素坐标。
    const x=Number(event && event.offsetX), y=Number(event && event.offsetY);
    const phaseName=locatePhaseByPoint(chart,x,y);
    if (phaseName!==null) togglePhase(phaseName);
  }
  chart.on('datazoom',onZoom);
  if (zr && typeof zr.on === 'function') zr.on('click',onZrClick);
  chart.__c021Handlers={zoom:onZoom,zrClick:onZrClick};
}

// 八、输出：官方编辑器直接识别 option；已有图表实例时绑定点击和缩放监听
option=chartOptions();
if(typeof myChart!=='undefined')bindGanttInteractions(myChart);
