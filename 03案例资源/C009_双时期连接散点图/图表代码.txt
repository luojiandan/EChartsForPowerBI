function chooseLabelLayouts(selectedRows){
  const boxW = view.sx / Math.max(gw, 1) * (C.labels.fontSize * 2.2);
  const boxH = view.sy / Math.max(gh, 1) * (C.labels.fontSize + 7);
  const gapX = view.sx / Math.max(gw, 1) * (C.labels.distance + 5);
  const gapY = view.sy / Math.max(gh, 1) * (C.labels.distance + 4);
  const candidates = (C.labels.positionCandidates || ['right','top','bottom','left']).slice();
  const placed = [];
  const layout = new Map();
  function rectFor(pos, x, y, name){
    const w = boxW + name.length * view.sx / Math.max(gw,1) * C.labels.fontSize * 0.9;
    const h = boxH;
    if(pos==='left') return {x1:x-w-gapX,x2:x-gapX,y1:y-h/2,y2:y+h/2};
    if(pos==='top') return {x1:x-w/2,x2:x+w/2,y1:y-h-gapY,y2:y-gapY};
    if(pos==='bottom') return {x1:x-w/2,x2:x+w/2,y1:y+gapY,y2:y+h+gapY};
    return {x1:x+gapX,x2:x+w+gapX,y1:y-h/2,y2:y+h/2};
  }
  function overlapArea(a,b){
    const w = Math.max(0, Math.min(a.x2,b.x2)-Math.max(a.x1,b.x1));
    const h = Math.max(0, Math.min(a.y2,b.y2)-Math.max(a.y1,b.y1));
    return w*h;
  }
  function score(pos, r){
    const rect = rectFor(pos, r.now.x, r.now.y, r.name);
    let penalty = 0;
    if(rect.x1 < view.x) penalty += (view.x-rect.x1) * 50;
    if(rect.x2 > view.x+view.sx) penalty += (rect.x2-(view.x+view.sx)) * 50;
    if(rect.y1 < view.y) penalty += (view.y-rect.y1) * 50;
    if(rect.y2 > view.y+view.sy) penalty += (rect.y2-(view.y+view.sy)) * 50;
    for(const p of placed) penalty += overlapArea(rect, p.rect) * 500;
    return {penalty, rect};
  }
  const rowsSorted = selectedRows.slice().sort((a,b)=>b.now.y-a.now.y);
  for(const r of rowsSorted){
    let cand = candidates.slice();
    if(r.now.x > view.x + view.sx*0.8) cand = ['left','top','bottom','right'];
    else if(r.now.y > view.y + view.sy*0.85) cand = ['bottom','right','left','top'];
    else if(r.now.y < view.y + view.sy*0.18) cand = ['top','right','left','bottom'];
    let best = {penalty: Infinity, pos: cand[0], rect: null};
    for(const pos of cand){
      const s = score(pos, r);
      if(s.penalty < best.penalty) best = {penalty: s.penalty, pos, rect: s.rect};
      if(s.penalty === 0) break;
    }
    placed.push({name:r.name, rect:best.rect});
    layout.set(r.name, best.pos);
  }
  return layout;
}

        html+=sep;
        html+=growthRow(F.yMetric+'增长率',r.gy);
        html+=growthRow(F.xMetric+'增长率',r.gx);
        // 增速差：自动生成直观结论；颜色表示纵轴/横轴增速对比，不代表经济改善。
        const threshold=1e-9;
        const hasDifference=Number.isFinite(r.diff);
        const fasterY=hasDifference && r.diff>threshold;
        const fasterX=hasDifference && r.diff<-threshold;
        const conclusion= !hasDifference ? '增速无法比较' :
          fasterY ? F.yMetric+'增长更快' : fasterX ? F.xMetric+'增长更快' : '两项指标增速持平';
        const conclusionValue=hasDifference ? Math.abs(r.diff).toFixed(cfg.percentDigits)+' 个百分点' : '—';
        const conclusionColor=!hasDifference||(!fasterY&&!fasterX)?clr.secondary:(fasterY?clr.positive:clr.negative);
        html+=sep;
        html+='<div style="'+rowStyle+'font-weight:600;">'+
          '<span style="color:'+clr.text+';white-space:nowrap;">'+esc(conclusion)+'</span>'+
          '<span style="color:'+conclusionColor+';font-variant-numeric:tabular-nums;white-space:nowrap;">'+esc(conclusionValue)+'</span></div>';
      }
      if(cfg.showBalance){html+=sep;html+=valueRow(F.yMetric+'－'+F.xMetric,num(current.y-current.x));}
      html+='</div>';
      return html;
    }
  },
  xAxis:axis(true),yAxis:axis(false),
  series:[
    ...(A.showBalanceLine?[{id:'balance',name:'两指标相等',type:'line',showSymbol:false,silent:true,clip:true,
      data:[[Math.min(full.x,full.y)-full.sx-full.sy,Math.min(full.x,full.y)-full.sx-full.sy],[Math.max(full.x+full.sx,full.y+full.sy)+full.sx+full.sy,Math.max(full.x+full.sx,full.y+full.sy)+full.sx+full.sy]],

// 更新缩放区域并自动显示放大区域中的对象名称
function applyView(){if(typeof myChart==='undefined')return;
 myChart.setOption({xAxis:{min:view.x,max:view.x+view.sx},yAxis:{min:view.y,max:view.y+view.sy},series:[{id:'current',data:currentPoints()}]});}
if(typeof myChart !== 'undefined') {
  const zr = myChart.getZr();
  if (myChart.__c006GenericHandlers) {
    for (const [eventName, fn] of myChart.__c006GenericHandlers) zr.off(eventName, fn);
  }
  const handlers = [];
  function bind(eventName, fn) {
    zr.on(eventName, fn);
    handlers.push([eventName, fn]);
  }

  // 坐标统一使用 ZRender 的画布坐标；兼容编辑器内外层事件包装。
  // 不使用 DOM 的 layerX / offsetX（其参照节点可能是散点或其他元素）。
  function pointer(e) {
    const nested = e && e.event;
    const nativeEvent = nested && nested.event ? nested.event : nested;
    let x = Number.isFinite(e && e.offsetX) ? e.offsetX : NaN;
    let y = Number.isFinite(e && e.offsetY) ? e.offsetY : NaN;
    // 浏览器事件经 ZRender 处理后会带有 zrX/zrY，优先使用。
    if (nativeEvent && Number.isFinite(nativeEvent.zrX) && Number.isFinite(nativeEvent.zrY)) {
      x = nativeEvent.zrX; y = nativeEvent.zrY;
    } else if (nested && Number.isFinite(nested.zrX) && Number.isFinite(nested.zrY)) {
      x = nested.zrX; y = nested.zrY;
    }
    return {x, y, nativeEvent};
  }

  // 根据当前真实坐标系统判断是否在绘图区，而非依赖手工边距。
  function plotRect() {
    const x0 = myChart.convertToPixel({xAxisIndex:0}, view.x);
    const x1 = myChart.convertToPixel({xAxisIndex:0}, view.x + view.sx);
    const y0 = myChart.convertToPixel({yAxisIndex:0}, view.y);
    const y1 = myChart.convertToPixel({yAxisIndex:0}, view.y + view.sy);
    if ([x0,x1,y0,y1].every(Number.isFinite)) {
      return {left:Math.min(x0,x1), right:Math.max(x0,x1),
              top:Math.min(y0,y1), bottom:Math.max(y0,y1),
              width:Math.abs(x1-x0), height:Math.abs(y1-y0)};
    }
    return {left:gx,right:gx+gw,top:gy,bottom:gy+gh,width:gw,height:gh};
  }
  function hitPlot(p) {
    if (!Number.isFinite(p.x) || !Number.isFinite(p.y)) return false;
    return myChart.containPixel({gridIndex:0}, [p.x,p.y]);
  }
  function clamp(v) {
    const mx = v.sx * I.panMarginRatio, my = v.sy * I.panMarginRatio;
    v.x = Math.max(full.x-v.sx+mx, Math.min(full.x+full.sx-mx, v.x));
    v.y = Math.max(full.y-v.sy+my, Math.min(full.y+full.sy-my, v.y));
    return v;
  }
  let dragging = null;
  function wheel(e) {
    if (!I.wheelEnabled) return;
    const p = pointer(e);
    if (!hitPlot(p)) return;
    const ev = p.nativeEvent;
    let dy = ev && typeof ev.deltaY === 'number' ? -ev.deltaY : (e.wheelDelta || 0);
    if (!dy && ev && typeof ev.wheelDelta === 'number') dy = ev.wheelDelta;
    if (!dy) return;
    const ratio = dy > 0 ? I.wheelFactor : 1 / I.wheelFactor;
    const scale = Math.max(I.minZoomRatio, Math.min(1, view.sx / full.sx * ratio));
    const nsx = full.sx * scale, nsy = full.sy * scale;
    const rect = plotRect();
    const fx = Math.max(0,Math.min(1,(p.x-rect.left)/rect.width));
    const fy = Math.max(0,Math.min(1,1-(p.y-rect.top)/rect.height));
    const ax = view.x + fx*view.sx, ay = view.y + fy*view.sy;
    view = clamp({x:ax-fx*nsx,y:ay-fy*nsy,sx:nsx,sy:nsy});
    applyView();
    if (ev && ev.preventDefault) ev.preventDefault();
  }
  function down(e) {
    if (!I.dragEnabled) return;
    const p = pointer(e);
    if (!hitPlot(p)) return;
    const ev = p.nativeEvent;
    if (ev && typeof ev.button === 'number' && ev.button !== 0) return;
    dragging = {px:p.x, py:p.y, x:view.x, y:view.y, sx:view.sx, sy:view.sy};
  }
  function move(e) {
    if (!dragging) return;
    const p = pointer(e);
    if (!Number.isFinite(p.x) || !Number.isFinite(p.y)) return;
    const rect = plotRect();
    view = clamp({...view,
      x:dragging.x-(p.x-dragging.px)*dragging.sx/rect.width,
      y:dragging.y+(p.y-dragging.py)*dragging.sy/rect.height});
    applyView();
  }
  function up() { dragging = null; }
  function reset(e) {
    if (!I.doubleClickReset) return;
    const p=pointer(e);
    if (!hitPlot(p)) return;
    dragging=null;view={...full};applyView();
  }
  bind('mousewheel', wheel);
  bind('mousedown', down);
  bind('mousemove', move);
  bind('mouseup', up);
  bind('globalout', up);
  bind('dblclick', reset);
  myChart.__c006GenericHandlers=handlers;
}
