function fmt(v, d) {
  if (v === null || v === undefined || !isFinite(Number(v))) return '—';
  return Number(v).toLocaleString('zh-CN', {
    minimumFractionDigits: d === undefined ? setting.unitDecimals : d,
    maximumFractionDigits: d === undefined ? setting.unitDecimals : d
  });
}
function esc(v) {
  return String(v).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}
function validNumber(v) {
  return v !== null && v !== undefined && String(v).trim() !== '' && isFinite(Number(v));
}
function directionColor(v) {
  if (v === null || !isFinite(v) || Math.abs(v) < eps) return C.secondary;
  return (v > 0) === setting.positiveIsGood ? C.positive : C.negative;
}
function arrowText(v, pct) {
  if (v === null || !isFinite(v)) return '—';
  return (v > eps ? '▲ ' : v < -eps ? '▼ ' : '') + fmt(Math.abs(v), pct ? setting.percentDecimals : setting.unitDecimals) + (pct ? '%' : '');
}
function signedText(v, pct) {
  if (v === null || !isFinite(v)) return '—';
  return (v > eps ? '+' : '') + fmt(v, pct ? setting.percentDecimals : setting.unitDecimals) + (pct ? '%' : '');
}
function percent(numerator, denominator) {
  return denominator === null || !isFinite(denominator) || Math.abs(denominator) < eps ? null : numerator / Math.abs(denominator) * 100;
}
function niceStep(raw) {
  var t = Math.abs(raw);
  if (!isFinite(t) || t <= 0) return 1;
  var power = Math.pow(10, Math.floor(Math.log(t) / Math.LN10));
  var candidates = [1, 2, 2.5, 5, 10];
  for (var i = 0; i < candidates.length; i++) if (candidates[i] * power >= t - 1e-12) return candidates[i] * power;
  return 10 * power;
}
function niceSignedExtent(values, minPadding, maxPadding, ticks) {
  var lo = Math.min.apply(null, values.concat([0]));
  var hi = Math.max.apply(null, values.concat([0]));
  var span = Math.max(eps, hi - lo);
  var targetStep = span * (1 + minPadding + maxPadding) / (ticks || 5);
  var step = niceStep(targetStep);
  var a = Math.floor((lo - span * minPadding) / step) * step;
  var b = Math.ceil((hi + span * maxPadding) / step) * step;
  if (a === b) b = a + step;
  return { min: Number(a.toPrecision(12)), max: Number(b.toPrecision(12)), interval: Number(step.toPrecision(12)) };
}
function nicePositiveExtent(values, padTop, ticks) {
  var hi = Math.max.apply(null, values.concat([0]));
  var top = hi * (1 + padTop);
  var step = niceStep(top / Math.max(3, ticks || 4));
  var max = Math.ceil(top / step) * step;
  return { min: 0, max: max, interval: step };
}

function makeGraphic() {

    { type: 'rect', silent: true, left: L.titleLeft, top: L.redDashTop, shape: { width: 20, height: 4 }, style: { fill: C.negative } },
    graphicText(L.titleLeft, L.titleTop, setting.title, { font: 'bold ' + (compact ? 19 : 22) + 'px sans-serif', width: W - L.titleLeft * 2, overflow: 'truncate' }),
    graphicText(L.titleLeft, L.subtitleTop, subtitle, { font: '12px sans-serif', fill: C.secondary, width: W - L.titleLeft * 2, overflow: 'truncate' })
  ];
  if (setting.showLegend && W > 760) {
    var lx = L.titleLeft, ly = L.legendTop;
    var legendX = lx;
    [
      ['PY', C.previous], ['AC', C.actual], ['增加', C.positive], ['减少', C.negative]
    ].forEach(function(v){
      g.push({ type: 'rect', silent: true, left: legendX, top: ly + 2, shape: { width: 11, height: 11 }, style: { fill: v[1], stroke: 'none' } });
      g.push(graphicText(legendX + 16, ly - 1, v[0], { font: '11px sans-serif', fill: C.secondary }));
      // 图例按文字宽度紧凑排列，不再固定 92px 一档。
      var textWidth = /[^\x00-\x7F]/.test(v[0]) ? v[0].length * 11 : v[0].length * 7;
      legendX += 16 + textWidth + setting.legendItemGap;
    });
  }
  if (setting.showSectionTitles) {
    g.push(graphicText(gridLeft, topMargin - L.sectionGap, '月度差异累计贡献 · △PY' + (metricUnit ? '（' + metricUnit + '）' : ''), { font: 'bold 12px sans-serif' }));
    g.push(graphicText(gridLeft, midTop - L.sectionGap, '月度同比变化率 · △PY%（末列为累计同比）', { font: 'bold 12px sans-serif' }));
    g.push(graphicText(gridLeft, bottomTop - L.sectionGap, '月度规模 · PY / AC' + (metricUnit ? '（' + metricUnit + '）' : ''), { font: 'bold 12px sans-serif' }));
  }
  if (setting.showTotalDivider && monthCount > 0 && innerWidth > 0) {
    var sepX = gridLeft + innerWidth * monthCount / categories.length;
    [
      { start: topMargin - 6, end: topMargin + topHeight + 5 },
      { start: midTop - 6, end: midTop + midHeight + 5 },
      { start: bottomTop - 6, end: bottomTop + bottomHeight + 5 }
    ].forEach(function(r){
      g.push({ type: 'line', silent: true, shape: { x1: sepX, y1: r.start, x2: sepX, y2: r.end }, style: { stroke: C.gridLine, lineWidth: 1, lineDash: [4, 4] } });
    });
  }
  // 最后一列独立缩放双柱：显式注明与月度规模不是同一 Y 轴尺度。
  if (setting.showTotalSizeBars && setting.showTotalScaleHint && monthCount > 0 && innerWidth > 0) {
    var totalCenterX = gridLeft + innerWidth * (monthCount + 0.5) / categories.length;
    var totalHint = '独立尺度';
    g.push(graphicText(totalCenterX - 20, bottomTop + 8, totalHint,
      {font:'10px sans-serif',fill:C.secondary}));
  }
  if (setting.showPairLabels && monthCount > 0 && innerWidth > 0) {
    for (var li = 0; li < monthCount; li++) {
      var cx = gridLeft + innerWidth * (li + 0.5) / categories.length;
      var topVal = Math.max(Number(acMonthly[li] || 0), Number(pyMonthly[li] || 0));
      var yRatio = (topVal - barsAxis.min) / Math.max(eps, barsAxis.max - barsAxis.min);
      var labelY = bottomTop + bottomHeight * (1 - yRatio) - 30;
      var minY = bottomTop + 10;
      var maxY = bottomTop + bottomHeight - 38;
      labelY = Math.max(minY, Math.min(maxY, labelY));
      var acText = 'AC ' + fmt(acMonthly[li]);
      var pyText = 'PY ' + fmt(pyMonthly[li]);
      var approxAcW = (compact ? 6.4 : 7.0) * acText.length;
      var approxPyW = (compact ? 6.0 : 6.6) * pyText.length;
      g.push(graphicText(cx - approxAcW / 2, labelY, acText, { font: 'bold ' + (compact ? 10 : 11) + 'px sans-serif', fill: C.actual }));
      g.push(graphicText(cx - approxPyW / 2, labelY + 14, pyText, { font: (compact ? 10 : 11) + 'px sans-serif', fill: C.secondary }));
    }
  }
  if (setting.showFooter) {
    g.push(graphicText(L.footerLeft, H - L.footerBottom, '数据来源：' + sourceText, { font: '10px sans-serif', fill: C.secondary, width: Math.max(120, W - L.footerLeft * 2), overflow: 'truncate' }));

    g.push(graphicText(L.footerLeft, H - L.footerBottom + 18, note, { font: '10px sans-serif', fill: C.secondary, width: Math.max(120, W - L.footerLeft * 2), overflow: 'truncate' }));
  }
  return g;
}

      { left: gridLeft, right: gridRight, top: topMargin, bottom: H - topMargin - topHeight, containLabel: false },
      { left: gridLeft, right: gridRight, top: midTop, bottom: H - midTop - midHeight, containLabel: false },
      { left: gridLeft, right: gridRight, top: bottomTop, bottom: H - bottomTop - bottomHeight, containLabel: false }
    ],
    graphic: makeGraphic(),
    tooltip: {
      trigger: 'axis', confine: true, appendToBody: false,
      borderWidth: 1, borderColor: C.gridLine, backgroundColor: '#FFFFFF', padding: [11, 13],
      textStyle: { color: C.text, fontSize: 12 },
      extraCssText: 'box-shadow:0 5px 22px rgba(38,50,56,.11);border-radius:5px;',
      axisPointer: { type: 'shadow', shadowStyle: { color: 'rgba(174,183,193,.09)' } },
      formatter: function(params) {
        var p = Array.isArray(params) ? params[0] : params;
        if (!p || p.dataIndex === undefined || p.dataIndex === null) return '';
        var idx = p.dataIndex;
        if (idx === monthCount) return totalTooltip();
        if (idx >= 0 && idx < monthCount) return monthTooltip(idx);
        return '';
      }
    },
    axisPointer: { link: [{ xAxisIndex: [0, 1, 2] }] },
    xAxis: [
      { type: 'category', gridIndex: 0, data: categories, axisTick: { show: false }, axisLine: { show: false }, axisLabel: { show: false }, splitLine: { show: false } },
      { type: 'category', gridIndex: 1, data: categories, axisTick: { show: false }, axisLine: { show: false }, axisLabel: { show: false }, splitLine: { show: false } },
      { type: 'category', gridIndex: 2, data: categories, axisTick: { show: false }, axisLine: { show: true, lineStyle: { color: C.gridLine, width: 1 } }, axisLabel: { show: true, interval: 0, margin: 12, color: C.text, fontSize: compact ? 10 : 11 }, splitLine: { show: false } }
    ],
    yAxis: [
      { type: 'value', gridIndex: 0, min: bridgeAxis.min, max: bridgeAxis.max, interval: bridgeAxis.interval, axisLine: { show: false }, axisTick: { show: false }, axisLabel: { show: setting.showAxisNumbers, color: C.secondary, fontSize: 10 }, splitLine: { show: setting.showGridLines, lineStyle: { color: C.gridLine, type: 'dashed' } } },
      { type: 'value', gridIndex: 1, min: rateAxis.min, max: rateAxis.max, interval: rateAxis.interval, axisLine: { show: false }, axisTick: { show: false }, axisLabel: { show: setting.showAxisNumbers, color: C.secondary, fontSize: 10, formatter: '{value}%' }, splitLine: { show: setting.showGridLines, lineStyle: { color: C.gridLine, type: 'dashed' } } },
      { type: 'value', gridIndex: 2, min: barsAxis.min, max: barsAxis.max, interval: barsAxis.interval, axisLine: { show: false }, axisTick: { show: false }, axisLabel: { show: setting.showAxisNumbers, color: C.secondary, fontSize: 10 }, splitLine: { show: setting.showGridLines, lineStyle: { color: C.gridLine, type: 'dashed' } } }
    ],
    series: [
      {
        name: '月度差异贡献 · △PY', type: 'custom', coordinateSystem: 'cartesian2d',
        xAxisIndex: 0, yAxisIndex: 0, z: 5, data: rowsWaterfall.map(function(d){ return { value: d }; }), encode: { x: 0, y: [1, 2] },
        renderItem: function(params, api) {
          var index = api.value(0), start = api.value(1), end = api.value(2), delta = api.value(3), total = api.value(4) === 1;
          var center = api.coord([index, 0])[0];
          var yStart = api.coord([index, start])[1], yEnd = api.coord([index, end])[1];
          var topY = Math.min(yStart, yEnd), barH = Math.abs(yStart - yEnd);
          var lowH = Math.max(2, barH), barY = barH < 2 ? (yStart + yEnd) / 2 - 1 : topY;
          var col = directionColor(delta);
          var kids = [];
          if (setting.showBridgeConnectors && !total && index < monthCount - 1) {
            var nextCx = api.coord([index + 1, 0])[0];
            kids.push({ type: 'line', silent: true, shape: { x1: center + waterfallWidth / 2, y1: yEnd, x2: nextCx - waterfallWidth / 2, y2: yEnd }, style: { stroke: C.previous, lineWidth: 1, lineDash: [4, 4] } });
          }
          kids.push({ type: 'rect', shape: { x: center - waterfallWidth / 2, y: barY, width: waterfallWidth, height: lowH }, style: { fill: col, stroke: total ? col : col, lineWidth: total ? 1.1 : 0 } });
          var textY = delta >= -eps ? barY - L.labelDistance : barY + lowH + L.labelDistance;
          textY = Math.max(params.coordSys.y + 12, Math.min(params.coordSys.y + params.coordSys.height - 12, textY));
          kids.push({ type: 'text', silent: true, style: { x: center, y: textY, text: arrowText(delta, false), fill: col, font: (total ? 'bold ' : '') + (compact ? '11' : '12') + 'px sans-serif', align: 'center', verticalAlign: 'middle' } });
          return { type: 'group', children: kids };
        }
      },
      {
        name: '月度同比变化率 · △PY%', type: 'bar', xAxisIndex: 1, yAxisIndex: 1, z: 4,
        barWidth: rateBarWidth, barGap: '0%', barCategoryGap: '36%', clip: true,
        data: rateSeriesData, itemStyle: { color: C.positive }
      },
      {
        name: 'PY', type: 'bar', xAxisIndex: 2, yAxisIndex: 2, z: 2,
        barWidth: pyBarWidth, barGap: overlapGap, barCategoryGap: '35%',
        itemStyle: { color: C.previous }, data: buildPairLabel('PY')
      },
      {
        name: 'AC', type: 'bar', xAxisIndex: 2, yAxisIndex: 2, z: 3,
        barWidth: acBarWidth, barGap: overlapGap, barCategoryGap: '35%',
        itemStyle: { color: C.actual }, data: buildPairLabel('AC')
      },
      {
        // 累计列独立缩放，但仍置于与各月相同的横坐标单元。
        // 不把累计金额写进普通 bar data，以免 1—9 月的柱子被压扁。
        name: '累计规模 · AC/PY', type: 'custom', coordinateSystem: 'cartesian2d',
        xAxisIndex: 2, yAxisIndex: 2, z: 4,
        silent: false,
        encode: { x: 0 },
        data: setting.showTotalSizeBars && totalAC !== null && totalPY !== null
          ? [{ value: [monthCount, totalPY, totalAC] }] : [],
        renderItem: function(params, api) {
          var idx = api.value(0);
          var py = Number(api.value(1)), ac = Number(api.value(2));
          var maxTotal = Math.max(py, ac, eps);
          var centerX = api.coord([idx, 0])[0];
          // 独立尺度下 AC 与 PY 的真实相对比例不变。
          var fullHeight = params.coordSys.height * Math.max(0.20, Math.min(0.74, setting.totalBarHeightRatio));
          var baseY = params.coordSys.y + params.coordSys.height;
          var pyHeight = Math.max(1, py / maxTotal * fullHeight);
          var acHeight = Math.max(1, ac / maxTotal * fullHeight);
          var pyWidth = pyBarWidth, acWidth = acBarWidth;
          var actualOverlap = Math.min(pyWidth, pyWidth * Math.max(0, Math.min(1, setting.overlapRatio)));
          var groupWidth = pyWidth + acWidth - actualOverlap;
          var startX = centerX - groupWidth / 2;
          var acX = startX + pyWidth - actualOverlap;
          var kids = [
            {type:'rect',shape:{x:startX,y:baseY-pyHeight,width:pyWidth,height:pyHeight},
              style:{fill:C.previous}},
            {type:'rect',shape:{x:acX,y:baseY-acHeight,width:acWidth,height:acHeight},
              style:{fill:C.actual}}
          ];
          // 两行标签基于更高柱体固定排列，避免两根柱顶独立定位造成互相压盖。
          var topY = baseY - Math.max(pyHeight, acHeight) - 8;
          kids.push({type:'text',silent:true,style:{
            x:centerX,y:topY-13,text:'AC '+fmt(ac),fill:C.actual,
            font:'bold '+(compact?10:11)+'px sans-serif',align:'center',verticalAlign:'middle'
          }});
          kids.push({type:'text',silent:true,style:{
            x:centerX,y:topY,text:'PY '+fmt(py),fill:C.secondary,
            font:(compact?10:11)+'px sans-serif',align:'center',verticalAlign:'middle'
          }});
          return {type:'group',children:kids};
        }
      }
    ]
  };
