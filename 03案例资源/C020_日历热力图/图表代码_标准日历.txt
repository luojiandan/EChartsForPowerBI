// 三、数据校验与日期处理：从输入自动识别年份、月份、星期
function parseDate(value) {
  if (value instanceof Date && !isNaN(value.getTime())) {
    return [value.getFullYear(), String(value.getMonth()+1).padStart(2, '0'), String(value.getDate()).padStart(2, '0')].join('-');
  }
  const str = String(value == null ? '' : value).trim();
  const m = /^(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})(?:[ T].*)?$/.exec(str);
  if (!m) throw new Error('日期格式应为 YYYY-MM-DD：' + str);
  const y=+m[1], mo=+m[2], d=+m[3];
  const dt=new Date(y,mo-1,d);
  if (dt.getFullYear()!==y || dt.getMonth()+1!==mo || dt.getDate()!==d) throw new Error('无效日期：'+str);
  return y+'-'+String(mo).padStart(2,'0')+'-'+String(d).padStart(2,'0');
}
function formatNum(value, decimals) {
  return Number(value).toLocaleString('zh-CN', {minimumFractionDigits:decimals, maximumFractionDigits:decimals});
}
function escapeHtml(value) {
  return String(value == null ? '' : value).replace(/[&<>"']/g, function(c){
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
  });
}

// 四、日期预处理：按照真实日期生成每日记录，布局在渲染时随容器计算

    heatmapData.push({

    });

function countSelectedDays(range) {
  const monthly = Array(12).fill(0);
  let total = 0;
  let valid = 0;
  heatmapData.forEach(function (d) {
    const value = d.value[1];
    if (d.missing || !Number.isFinite(value)) return;
    valid++;
    if (value >= range[0] && value <= range[1]) {
      monthly[d.month - 1]++;
      total++;
    }
  });
  return { monthly: monthly, total: total, valid: valid };
}
function annualCountText() {
  const range = currentRange();
  const counts = countSelectedDays(range);
  return formatNum(range[0], 1) + setting.unit + '～' +
    formatNum(range[1], 1) + setting.unit + '｜全年 ' + counts.total + ' 天';
}

  // 选择不会越界且日期格尽量大的列数，兼容不同 Power BI 容器尺寸
  function candidateFor(columnCount) {
    const rows = Math.ceil(12 / columnCount);
    const eachWidth = (coord.width - (columnCount - 1) * colGap) / columnCount;
    const eachHeight = (coord.height - (rows - 1) * rowGap) / rows;
    const byWidth = (eachWidth - 6 * gap) / 7;
    const byHeight = (eachHeight - labelH - 5 * gap) / 6;
    return Math.floor(Math.min(byWidth, byHeight, maxSize));
  }

  if (setting.responsive) {
    // 优先考虑能整齐排满 12 个月的列数；同样大小时保留用户首选列数
    const candidates = [1, 2, 3, 4, 6, 12].filter(function(c) { return c <= maxCols; });
    if (!candidates.includes(preferredCols)) candidates.push(preferredCols);
    candidates.forEach(function(candidate) {
      const candidateSize = candidateFor(candidate);
      if (candidateSize > bestSize ||
        (candidateSize === bestSize && candidate === preferredCols)) {
        bestSize = candidateSize;
        cols = candidate;
      }
    });
  }
  const rows = Math.ceil(12 / cols);
  const size = Math.max(1, candidateFor(cols));
  const monthWidth = 7 * size + 6 * gap;
  const monthHeight = labelH + 6 * size + 5 * gap;
  const totalWidth = cols * monthWidth + (cols - 1) * colGap;
  const totalHeight = rows * monthHeight + (rows - 1) * rowGap;
  return {
    cols: cols, rows: rows, size: size, gap: gap, labelH: labelH,
    monthWidth: monthWidth, monthHeight: monthHeight,
    colGap: colGap, rowGap: rowGap,
    startX: coord.x + (coord.width - totalWidth) / 2,
    startY: coord.y + (coord.height - totalHeight) / 2
  };
}

// 八、交互联动：拖动图例更新持久选区，悬停图例临时预览附近数值的天数
// 依赖 ECharts 实例。ECharts 官方编辑器默认使用 myChart；Power BI 需在获取实例后调用绑定函数。
function bindCalendarRangeCounts(chartInstance) {
  if (!setting.showRangeCounts || !chartInstance || typeof chartInstance.on !== 'function') return;
  const zr = chartInstance.getZr && chartInstance.getZr();
  if (!zr) return;

  // 重新执行脚本时解绑旧监听，避免重复执行与内存泄漏
  const old = chartInstance.__c020InteractionHandlers;
  if (old) {
    chartInstance.off('datarangeselected', old.range);
    zr.off('mousemove', old.move);
    zr.off('globalout', old.leave);
  }

  function updateCounts() {
    chartInstance.setOption({
      graphic: [{id:'annualRangeCount', style:{text:annualCountText()}}],
      series: [{id:'calendarDays', data:heatmapData.slice()}]
    }, {lazyUpdate:false});
  }

  // 持久筛选：拖动两端手柄时立即统计选区天数
  const onRangeSelected = function (event) {
    if (!event || !Array.isArray(event.selected) || event.selected.length < 2) return;
    const low = Math.max(minValue, Math.min(Number(event.selected[0]), Number(event.selected[1])));
    const high = Math.min(maxValue, Math.max(Number(event.selected[0]), Number(event.selected[1])));
    if (!Number.isFinite(low) || !Number.isFinite(high) || low > high) return;
    selectedRange = [low, high];
    hoverRange = null;
    updateCounts();
  };

  // 悬停预览：连续型 visualMap 为横向、底部居中布局。
  // 使用与色阶一致的数值范围及悬停带宽，统计用户目前指向的温度附近天数。
  let lastHoverKey = '';
  const onMouseMove = function (event) {
    if (!setting.legendHoverStats || !setting.showLegend) return;
    const width = chartInstance.getWidth();
    const height = chartInstance.getHeight();
    const barLength = 190;
    const barCenterX = width / 2;
    // 连续型 visualMap bottom 为 53，横向色条高度为 12；给手柄适当余量。
    const barCenterY = height - 53 - 12;
    const x = Number(event.offsetX);
    const y = Number(event.offsetY);
    const inside = Number.isFinite(x) && Number.isFinite(y) &&
      x >= barCenterX - barLength / 2 - 3 && x <= barCenterX + barLength / 2 + 3 &&
      Math.abs(y - barCenterY) <= 19;
    if (!inside) {
      if (hoverRange !== null) {
        hoverRange = null;
        lastHoverKey = '';
        updateCounts();
      }
      return;
    }
    const position = Math.max(0, Math.min(1, (x - (barCenterX-barLength/2)) / barLength));
    const centerValue = minValue + position * (maxValue-minValue);
    const half = Math.max(0.001, Number(setting.legendHoverDataSize) || 3) / 2;
    const step = Math.max(0.01, Number(setting.legendHoverStep) || 0.2);
    const centerRounded = Math.round(centerValue/step)*step;
    const low = Math.max(minValue, centerRounded-half);
    const high = Math.min(maxValue, centerRounded+half);
    const key = low.toFixed(3)+'|'+high.toFixed(3);
    if (key === lastHoverKey) return;
    lastHoverKey = key;
    hoverRange = [low, high];
    updateCounts();
  };
  const onGlobalOut = function () {
    if (hoverRange === null) return;
    hoverRange = null;
    lastHoverKey = '';
    updateCounts();
  };

  chartInstance.on('datarangeselected', onRangeSelected);
  zr.on('mousemove', onMouseMove);
  zr.on('globalout', onGlobalOut);
  chartInstance.__c020InteractionHandlers = {range:onRangeSelected, move:onMouseMove, leave:onGlobalOut};
}
if (typeof myChart !== 'undefined') bindCalendarRangeCounts(myChart);
