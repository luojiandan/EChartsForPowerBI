// 三、数据校验与日期处理：从输入自动识别年份、月份、星期
function parseDate(value) {
  if (value instanceof Date && !isNaN(value.getTime())) {
    return [value.getFullYear(), String(value.getMonth()+1).padStart(2, '0'), String(value.getDate()).padStart(2, '0')].join('-');
  }
  const str = String(value == null ? '' : value).trim();
  const m = /^(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})(?:[ T].*)?$/.exec(str);
  if (!m) throw new Error('日期格式应为 YYYY-MM-DD：' + str);
  const y=+m[1], mo=+m[2], d=+m[3];
  const dt=new Date(y,mo-1,d);
  if (dt.getFullYear()!==y || dt.getMonth()+1!==mo || dt.getDate()!==d) throw new Error('无效日期：'+str);
  return y+'-'+String(mo).padStart(2,'0')+'-'+String(d).padStart(2,'0');
}
function formatNum(value, decimals) {
  return Number(value).toLocaleString('zh-CN', {minimumFractionDigits:decimals, maximumFractionDigits:decimals});
}
function escapeHtml(value) {
  return String(value == null ? '' : value).replace(/[&<>"']/g, function(c){
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
  });
}

// 四、日期预处理：根据输入日期生成真实星期与周序列
const heatmapData=[];

  const firstOffset=(new Date(selectedYear,mi,1).getDay()-setting.weekStart+7)%7;
  const days=new Date(selectedYear,month,0).getDate();
  for(let day=1;day<=days;day++) {
    const index=firstOffset+day-1;
    const date=selectedYear+'-'+String(month).padStart(2,'0')+'-'+String(day).padStart(2,'0');
    const val=dateMap.has(date) ? dateMap.get(date) : null;
    heatmapData.push({
      value:[heatmapData.length,val],date:date,day:day,month:month,
      weekday:index%7,week:Math.floor(index/7),missing:val===null
    });
  }
}

// 五、区间统计：按图例选中范围统计每月和全年有效数据天数

function currentRange() { return hoverRange || selectedRange; }
function countSelectedDays(range) {
  const monthly = Array(12).fill(0);
  let total = 0;
  let valid = 0;
  heatmapData.forEach(function (d) {
    const value = d.value[1];
    if (d.missing || !Number.isFinite(value)) return;
    valid++;
    if (value >= range[0] && value <= range[1]) {
      monthly[d.month - 1]++;
      total++;
    }
  });
  return { monthly: monthly, total: total, valid: valid };
}
function annualCountText() {
  const range = currentRange();
  const counts = countSelectedDays(range);
  return formatNum(range[0], 1) + setting.unit + '～' +
    formatNum(range[1], 1) + setting.unit + '｜全年 ' + counts.total + ' 天';
}

}

  return {type:'text',style:{x:x,y:y,text:String(text),fill:color,font:font,textAlign:align||'left'},silent:true};
}

// 八、交互联动：原生手柄控制持久选区，悬停图例临时预览统计天数
// 依赖 ECharts 实例。ECharts 官方编辑器默认使用 myChart；Power BI 需在获取实例后调用绑定函数。
function bindCalendarRangeCounts(chartInstance) {
  if (!chartInstance || typeof chartInstance.on !== 'function') return;
  if (!setting.showRangeCounts) return;
  const zr = chartInstance.getZr && chartInstance.getZr();
  if (!zr) return;

  // 重新执行脚本时解绑旧监听，避免重复执行与内存泄漏
  const old = chartInstance.__c020InteractionHandlers;
  if (old) {
    chartInstance.off('datarangeselected', old.range);
    zr.off('mousemove', old.move);
    zr.off('globalout', old.leave);
    if (old.down) zr.off('mousedown', old.down);
    if (old.up) zr.off('mouseup', old.up);
    if (old.hoverDay) chartInstance.off('mouseover', old.hoverDay);
    if (old.leaveDay) chartInstance.off('mouseout', old.leaveDay);
  }

  // 仅刷新统计文字；不能在鼠标悬停时修改 visualMap.range 或重建系列，
  // 否则会覆盖用户的正式筛选、打断手柄拖动并引起悬停事件循环。
  function updateCounts() {
    chartInstance.setOption({
      graphic: [{id:'annualRangeCount', style:{text:annualCountText()}}]
    }, {lazyUpdate:false});
  }

  // 持久筛选：拖动两端手柄时立即统计选区天数
  const onRangeSelected = function (event) {
    if (!event || !Array.isArray(event.selected) || event.selected.length < 2) return;
    const low = Math.max(minValue, Math.min(Number(event.selected[0]), Number(event.selected[1])));
    const high = Math.min(maxValue, Math.max(Number(event.selected[0]), Number(event.selected[1])));
    if (!Number.isFinite(low) || !Number.isFinite(high) || low > high) return;
    selectedRange = [low, high];
    hoverRange = null;
    updateCounts();
  };

  // 悬停只预览天数，不接管 ECharts visualMap 的着色与拖动事件。
  // 数值区间通过同一图例参数估算，且与拖动形成的持久选区相互独立。
  let lastHoverKey = '';
  let draggingLegend = false;
  function isOverLegend(x, y) {
    const width = chartInstance.getWidth();
    const height = chartInstance.getHeight();
    const length = setting.legendBarLength;
    const centerY = height - setting.legendBottom - setting.legendBarThickness;
    return Number.isFinite(x) && Number.isFinite(y) &&
      x >= width / 2 - length / 2 - 3 && x <= width / 2 + length / 2 + 3 &&
      Math.abs(y - centerY) <= 19;
  }
  const onMouseDown = function(event) {
    if (isOverLegend(Number(event.offsetX), Number(event.offsetY))) {
      draggingLegend = true;
      if (hoverRange !== null) {
        hoverRange = null;
        lastHoverKey = '';
        updateCounts();
      }
    }
  };
  const onMouseUp = function() { draggingLegend = false; };
  const onMouseMove = function (event) {
    if (!setting.legendHoverStats || !setting.showLegend || draggingLegend) return;
    const width = chartInstance.getWidth();
    const height = chartInstance.getHeight();
    const barLength = setting.legendBarLength;
    const barCenterX = width / 2;
    // 与图例配置共用位置参数；允许少量手柄操作的命中余量。
    const barCenterY = height - setting.legendBottom - setting.legendBarThickness;
    const x = Number(event.offsetX);
    const y = Number(event.offsetY);
    const inside = isOverLegend(x, y);
    if (!inside) {
      if (hoverRange !== null) {
        hoverRange = null;
        lastHoverKey = '';
        updateCounts();
      }
      return;
    }
    const position = Math.max(0, Math.min(1, (x - (barCenterX-barLength/2)) / barLength));
    const centerValue = minValue + position * (maxValue-minValue);
    const half = Math.max(0.001, Number(setting.legendHoverDataSize) || 3) / 2;
    const step = Math.max(0.01, Number(setting.legendHoverStep) || 0.2);
    const centerRounded = Math.round(centerValue/step)*step;
    const low = Math.max(minValue, centerRounded-half);
    const high = Math.min(maxValue, centerRounded+half);
    const key = low.toFixed(3)+'|'+high.toFixed(3);
    if (key === lastHoverKey) return;
    lastHoverKey = key;
    hoverRange = [low, high];
    updateCounts();
  };
  const onGlobalOut = function () {
    draggingLegend = false;
    if (hoverRange === null) return;
    hoverRange = null;
    lastHoverKey = '';
    updateCounts();
  };

  // 单日悬停：只为鼠标实际指向的色块绘制深色边框，图例预览不产生黑框。
  let activeDayRect = null;
  function resetDayOutline() {
    if (activeDayRect && typeof activeDayRect.setStyle === 'function') {
      activeDayRect.setStyle({stroke:'#FFFFFF',lineWidth:1,shadowBlur:0});
    }
    activeDayRect = null;
  }
  const onDayOver = function(params) {
    if (!params || params.seriesId !== 'calendarDays') return;
    resetDayOutline();
    const target = params.event && params.event.target;
    if (!target) return;
    const rect = target.type === 'rect' ? target :
      (target.parent && target.parent.children && target.parent.children().find(function(c){return c.type === 'rect';}));
    if (rect && typeof rect.setStyle === 'function') {
      rect.setStyle({stroke:'#263238',lineWidth:1.4,shadowBlur:3,shadowColor:'rgba(38,50,56,0.2)'});
      activeDayRect = rect;
    }
  };
  const onDayOut = function() { resetDayOutline(); };

  chartInstance.on('datarangeselected', onRangeSelected);
  zr.on('mousemove', onMouseMove);
  zr.on('globalout', onGlobalOut);
  zr.on('mousedown', onMouseDown);
  zr.on('mouseup', onMouseUp);
  chartInstance.on('mouseover', onDayOver);
  chartInstance.on('mouseout', onDayOut);
  chartInstance.__c020InteractionHandlers = {
    range:onRangeSelected,move:onMouseMove,leave:onGlobalOut,down:onMouseDown,up:onMouseUp,
    hoverDay:onDayOver,leaveDay:onDayOut
  };
}
if (typeof myChart !== 'undefined') bindCalendarRangeCounts(myChart);
