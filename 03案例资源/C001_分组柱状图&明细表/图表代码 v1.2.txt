// ======================================================
// 8. 颜色辅助函数
// ======================================================

function hexToRgb(hex) {
  const value = hex.replace("#", "");

  return {
    r: parseInt(value.substring(0, 2), 16),

    g: parseInt(value.substring(2, 4), 16),

    b: parseInt(value.substring(4, 6), 16),
  };
}

function rgbToHex(r, g, b) {
  const toHex = (n) => {
    const h = n.toString(16);

    return h.length === 1 ? "0" + h : h;
  };

  return "#" + toHex(r) + toHex(g) + toHex(b);
}

function mixColor(color1, color2, t) {
  const c1 = hexToRgb(color1);

  const c2 = hexToRgb(color2);

  const r = Math.round(c1.r + (c2.r - c1.r) * t);

  const g = Math.round(c1.g + (c2.g - c1.g) * t);

  const b = Math.round(c1.b + (c2.b - c1.b) * t);

  return rgbToHex(r, g, b);
}

// ======================================================
// 9. 区域浅背景
// ======================================================

function getAreaColor(region) {
  return mixColor("#FFFFFF", barColors[region], 0.07);
}

// ======================================================
// 10. 柱体渐变色
// ======================================================

function getBarShadeColor(region, order, count) {
  const baseColor = barColors[region];

  if (count <= 1) {
    return baseColor;
  }

  const t = order / (count - 1);

  const topColor = mixColor(baseColor, "#FFFFFF", 0.03 + t * 0.18);

  const bottomColor = mixColor(baseColor, "#FFFFFF", 0.16 + t * 0.3);

  return new echarts.graphic.LinearGradient(0, 0, 0, 1, [
    {
      offset: 0,
      color: topColor,
    },
    {
      offset: 1,
      color: bottomColor,
    },
  ]);
}

// ======================================================
// 11. 表格单元格颜色
// ======================================================

function getCellColor(value, min, max, region) {
  const baseColor = barColors[region];

  if (max === min) {
    return mixColor("#FFFFFF", baseColor, 0.42);
  }

  const t = (value - min) / (max - min);

  const strength = 0.18 + t * 0.54;

  return mixColor("#FFFFFF", baseColor, strength);
}

function getTextColor(bgColor) {
  const c = hexToRgb(bgColor);

  const luminance = (0.299 * c.r + 0.587 * c.g + 0.114 * c.b) / 255;

  return luminance > 0.62 ? "#36404A" : "#FFFFFF";
}

const salesMax = Math.max(...salesValues);

const profitMax = Math.max(...profitValues);

// ======================================================
// 13. Heatmap 表格数据
// ======================================================

function createHeatData(metric, rowIndex, min, max) {
  return barData.map((d, i) => {
    const value = d[metric];

    const bgColor = getCellColor(value, min, max, d.region);

    return {
      value: [i, rowIndex, value],

      region: d.region,

      product: d.product,

      itemStyle: {
        color: bgColor,
        borderColor: "#FFFFFF",
        borderWidth: 1,
      },

      label: {
        color: getTextColor(bgColor),
      },
    };
  });
}

const qtyHeatData = createHeatData("qty", 0, qtyMin, qtyMax);

const salesHeatData = createHeatData("sales", 1, salesMin, salesMax);

const profitHeatData = createHeatData("profit", 2, profitMin, profitMax);

// 15. 第二层X轴：区域中心
//
// 用 value 轴，使偶数个产品时也能真正居中
// ======================================================

const regionCenterMap = {};

regionRanges.forEach((item) => {
  const center = (item.start + item.end) / 2;

  regionCenterMap[center.toFixed(1)] = item.region;
});

function getRegionAxisLabel(value) {
  const key = Number(value).toFixed(1);

  return regionCenterMap[key] || "";
}

// ======================================================
// 16. 表格区域分隔线位置
// ======================================================

function pctToNum(value) {
  return parseFloat(String(value).replace("%", ""));
}

const tableLeftPercent = pctToNum(layout.tableGridLeft);

const tableRightPercent = pctToNum(layout.tableGridRight);

const tableWidthPercent = 100 - tableLeftPercent - tableRightPercent;

function getRegionDividerPercent(index) {
  const beforeCount = regionRanges
    .slice(0, index)
    .reduce((sum, d) => sum + d.count, 0);

  return tableLeftPercent + (beforeCount / colCount) * tableWidthPercent;
}

// 18. 根据 product 更新视觉状态
// ======================================================

function setHighlight(product) {

  const getOpacity = d =>
    !product || d.product === product ? 1 : 0.03;


  function updateHeatData(sourceData) {
    return sourceData.map(d => ({
      ...d,

      itemStyle: {
        ...d.itemStyle,
        opacity: getOpacity(d)
      },

      label: {
        ...d.label,
        opacity: getOpacity(d)
      }
    }));
  }


  myChart.setOption({
    series: [
      {
        name: "销额",

        data: barData.map(d => ({
          ...d,

          itemStyle: {
            opacity: getOpacity(d)
          },

          label: {
            opacity: getOpacity(d)
          }
        }))
      },

      {
        name: "销量",
        data: updateHeatData(qtyHeatData)
      },

      {
        name: "销额表格",
        data: updateHeatData(salesHeatData)
      },

      {
        name: "利润",
        data: updateHeatData(profitHeatData)
      }
    ]
  });
}


// ======================================================
// 6. 鼠标事件
// 先清掉旧的事件，再绑定当前这一版事件，防止重复注册。
// ======================================================
myChart.off('mouseover');
myChart.off('globalout');

myChart.on('mouseover', params => {
  if (
    params.seriesType === 'bar' &&
    params.data &&
    params.data.product
  ) {
    setHighlight(params.data.product);
  }
});

myChart.on('globalout', function () {
  setHighlight(null);
});
