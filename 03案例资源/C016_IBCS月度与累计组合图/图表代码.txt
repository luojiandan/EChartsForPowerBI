function fmt(v, decimals) {
  if (v === null || v === undefined || !isFinite(Number(v))) return '—';
  return Number(v).toLocaleString('zh-CN', {
    minimumFractionDigits: decimals === undefined ? setting.unitDecimals : decimals,
    maximumFractionDigits: decimals === undefined ? setting.unitDecimals : decimals
  });
}
function esc(s) {
  return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')
    .replace(/"/g,'&quot;').replace(/'/g,'&#39;');
}
function numeric(v) {
  return v !== null && v !== undefined && String(v).trim() !== '' && isFinite(Number(v));
}
var invalidInput = 年份.length !== 月份.length || 年份.length !== 度量值.length;
var rows = [];
if (!invalidInput) {
  for (var i=0; i<年份.length; i++) {
    var yy = Number(String(年份[i]).trim());
    var mm = Number(String(月份[i]).replace('月','').trim());
    if (!Number.isInteger(yy) || !Number.isInteger(mm) || mm<1 || mm>12 || !numeric(度量值[i])) continue;
    rows.push({year:yy, month:mm, value:Number(度量值[i])});
  }
}
var years = Array.from(new Set(rows.map(function(r){return r.year;}))).sort(function(a,b){return a-b;});
var acYear = setting.actualYear==='auto' ? years[years.length-1] : Number(setting.actualYear);
var pyYear = setting.previousYear==='auto' ? acYear-1 : Number(setting.previousYear);
var mapAC={}, mapPY={};
rows.forEach(function(r){
  var map = r.year===acYear ? mapAC : r.year===pyYear ? mapPY : null;
  if (map) map[r.month]=(map[r.month]===undefined?0:map[r.month])+r.value;
});
var now = new Date();

if (setting.cutoffMonth==='previousMonth') {

} else if (setting.cutoffMonth!=='all') {
  cutoff=Math.max(1,Math.min(12,Math.floor(Number(setting.cutoffMonth)||12)));
}

var pyMonth=months.map(function(m){return mapPY[m]===undefined?null:mapPY[m];});
function cumsum(arr){
  var sum=0, complete=true;
  return arr.map(function(v){
    if(v===null) complete=false;
    if (v!==null) sum+=v;
    return (!complete && setting.requireCompleteMonths) ? null : sum;
  });
}
var acYTD=cumsum(acMonth), pyYTD=cumsum(pyMonth);
function lastIndex(arr){for(var q=arr.length-1;q>=0;q--)if(arr[q]!==null)return q;return -1;}
var lastAC=lastIndex(acYTD), lastPY=lastIndex(pyYTD);
var lastBoth=-1;
for(var k=0;k<months.length;k++)if(acYTD[k]!==null && pyYTD[k]!==null)lastBoth=k;
var latestMonth=lastBoth>=0?months[lastBoth]:(months.length?months[months.length-1]:null);

var totalPY=lastBoth>=0?pyYTD[lastBoth]:null;
var diff=totalAC!==null && totalPY!==null?totalAC-totalPY:null;
var rate=diff!==null && totalPY!==0?diff/Math.abs(totalPY)*100:null;
function diffColor(v){
  if(v===null || !isFinite(v) || Math.abs(v)<1e-9)return C.secondary;
  return (v>0)===setting.positiveIsGood?C.positive:C.negative;
}
function diffText(v,pct){
  if(v===null)return '—';
  return (v>0?'▲ ':v<0?'▼ ':'')+fmt(Math.abs(v),pct?setting.percentDecimals:setting.unitDecimals)+(pct?'%':'');
}
function niceMax(value, steps) {
  var target=Math.max(value,0.00001)/steps;
  var magnitude=Math.pow(10,Math.floor(Math.log(target)/Math.LN10));
  var cs=[1,2,2.5,3,5,10];
  var step=10*magnitude;
  for(var i=0;i<cs.length;i++)if(cs[i]*magnitude>=target-1e-10){step=cs[i]*magnitude;break;}
  return {max:+(step*steps).toPrecision(12),interval:+step.toPrecision(12)};
}

var allYTD=acYTD.concat(pyYTD).filter(function(v){return v!==null && isFinite(v);});

var leftFraction=maxMonthly/leftAxis.max;

// 为通用模板保护：如果数据有负数或两图形空间太挤，提示换用其他可视化类型。

  !years.includes(acYear)?'缺少 AC 年度数据。':

// 文字元素：文字定位全部使用明确的左边界，不通过右对齐覆盖容器边界。
function textAt(x,y,t,style){
  return {type:'text',silent:true,left:x,top:y,
    style:Object.assign({text:String(t),fill:C.text,font:'12px sans-serif'},style||{})};
}

// 垂直空间：上半部分显示累计折线，下半部分显示月度柱；

function thinLineAt(y, opacity) {
  return {type:'rect',silent:true,left:plotLeft,top:Math.round(y),
    shape:{width:plotWidth,height:1},style:{fill:C.gridLine,opacity:opacity}};
}
// 参考线只画在月度柱所在的低位区间，不延伸到累计折线区。

  for(var gv=leftAxis.interval;gv<=visibleMonthlyTick+1e-8;gv+=leftAxis.interval){
    var gy=plotTop+plotHeight*(1-gv/leftAxis.max);
    if(gy>bandDividerY+5) graphics.push(thinLineAt(gy,.42));
  }
}

  graphics.push(textAt(plotLeft+2,bandDividerY+10,'月度 · 左轴'+(metricUnit?'（'+metricUnit+'）':''),
    {font:'11px sans-serif',fill:C.secondary}));

function barValues(arr,labelColor){
  return arr.map(function(v,i){
    if(!setting.showMonthlyLastLabels || i!==arr.length-1 || v===null)return v;
    return {value:v,label:{show:true,position:'top',distance:5,fontSize:10,
      color:labelColor,formatter:fmt(v)}};
  });
}
function lineValues(arr,last,shortName,labelColor,loc){
  return arr.map(function(v,i){
    if(v===null || !setting.showLastYTDLabels || i!==last)return v;
    return {value:v,label:{show:true,position:loc,distance:9,color:labelColor,
      fontSize:11,fontWeight:600,backgroundColor:'rgba(255,255,255,.92)',padding:[3,4],
      formatter:shortName+' YTD '+fmt(v)}};
  });
}

function tooltipFn(params){
  var idx=params && params.length?params[0].dataIndex:-1;
  if(idx<0||idx>=months.length)return '';
  var unitHeader=metricUnit?'（'+esc(metricUnit)+'）':'';
  var ac=acMonth[idx],py=pyMonth[idx],ay=acYTD[idx],pyy=pyYTD[idx];
  var d=(ac!==null&&py!==null)?ac-py:null;
  var yd=(ay!==null&&pyy!==null)?ay-pyy:null;
  var p=d!==null&&py!==0?d/Math.abs(py)*100:null;
  var yp=yd!==null&&pyy!==0?yd/Math.abs(pyy)*100:null;
  function cellRow(n,a,b,color){
    return '<tr><td style="padding:6px 9px 6px 0;color:'+color+'">'+esc(n)+
      '</td><td style="text-align:right;padding:6px 8px">'+fmt(a)+
      '</td><td style="text-align:right;padding:6px 0 6px 9px">'+fmt(b)+'</td></tr>';
  }
  function diffRow(n,a,b){
    return '<tr><td style="padding:6px 9px 6px 0;color:'+C.secondary+'">'+esc(n)+
      '</td><td style="text-align:right;padding:6px 8px;font-weight:600;color:'+diffColor(a)+'">'+diffText(a,false)+
      '</td><td style="text-align:right;padding:6px 0 6px 9px;font-weight:600;color:'+diffColor(a)+'">'+diffText(b,true)+'</td></tr>';
  }
  return '<div style="min-width:295px;max-width:360px;font:12px/1.4 sans-serif;color:'+C.text+'">'+
    '<div style="font-size:14px;font-weight:700;padding:0 0 9px;border-bottom:1px solid '+C.gridLine+'">'+
    acYear+'年'+months[idx]+'月 · '+esc(metricName)+'</div>'+
    '<table style="border-collapse:collapse;width:100%;margin-top:5px"><thead><tr style="color:'+C.secondary+'">'+
    '<th style="text-align:left;font-weight:400">情景</th><th style="text-align:right;font-weight:400">当月（'+esc(metricUnit)+
    '）</th><th style="text-align:right;font-weight:400">1—'+months[idx]+'月累计</th></tr></thead>'+
    cellRow('● AC · '+acYear,ac,ay,C.actual)+cellRow('● PY · '+pyYear,py,pyy,C.previous)+
    '</table><div style="border-top:1px solid '+C.gridLine+';margin-top:7px"></div>'+
    '<table style="border-collapse:collapse;width:100%"><thead><tr style="color:'+C.secondary+'">'+
    '<th style="text-align:left;font-weight:400">差异</th><th style="text-align:right;font-weight:400">增减额（'+esc(metricUnit)+
    '）</th><th style="text-align:right;font-weight:400">增减率</th></tr></thead>'+
    diffRow('当月 △PY',d,p)+diffRow('累计 △PY',yd,yp)+
    '</table></div>';
}
