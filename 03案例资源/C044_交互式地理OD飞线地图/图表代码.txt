function c044DecodeOutline(encoded) {
  var i=0, x=0, y=0, pts=[];
  function nextDelta() {
    var shift=0, result=0, b;
    do { b=encoded.charCodeAt(i++)-63; result|=(b&31)<<shift; shift+=5; } while (b>=32 && i<=encoded.length);
    return result&1 ? ~(result>>1) : result>>1;
  }
  while(i<encoded.length) {
    x += nextDelta(); y += nextDelta();
    pts.push([x/1000,y/1000]);
  }
  if (pts.length > 2 && (pts[0][0]!==pts[pts.length-1][0] || pts[0][1]!==pts[pts.length-1][1])) {
    pts.push([pts[0][0],pts[0][1]]);
  }
  return pts;
}

    var lng=Number(point[0]), lat=Number(point[1]);
    if (!Number.isFinite(lng)||!Number.isFinite(lat)) return;
    minLng=Math.min(minLng,lng);maxLng=Math.max(maxLng,lng);
    minLat=Math.min(minLat,lat);maxLat=Math.max(maxLat,lat);
  }
  c044MapGeoJSON.features.forEach(function(feature){
    var g=feature.geometry;
    var polygons=g.type==='MultiPolygon' ? g.coordinates : [g.coordinates];
    polygons.forEach(function(rings){rings.forEach(function(ring){ring.forEach(addPoint);});});
  });
  c044Airports.forEach(function(a){addPoint(a.coord);});
  var pad=Math.max(0,Number(setting.mapExtentPadding)||0);
  return [[minLng-pad,maxLat+pad],[maxLng+pad,minLat-pad]];
}

function c044SelectionDetails() {
  function m(label, value, tone) { return {label:String(label),value:String(value),tone:tone||'normal'}; }
  if (c044Focus.kind==='route') {
    var row=c044Rows.filter(function(r){return r.originalIndex===c044Focus.key;})[0];
    if (row) {
      var tone=Number.isFinite(row.yoy) ? row.yoy>0?'positive':row.yoy<0?'negative':'normal' : 'normal';
      var yoyText=Number.isFinite(row.yoy) ? (row.yoy>0?'▲ ':row.yoy<0?'▼ ':'')+Math.abs(row.yoy).toFixed(2)+'%' : '—';
      return {
        heading:row.from+' → '+row.to+'  ·  第'+row.rank+'名',
        metrics:[
          m(setting.metricName,c044Format(row.value)+' '+setting.metricUnit),
          m('当前样本占比',(c044Total?100*row.value/c044Total:0).toFixed(2)+'%'),
          m(setting.yoyLabel||'同比变化',yoyText,tone),
          m(setting.distanceLabel,Number.isFinite(row.distance)?c044Format(row.distance)+' '+setting.distanceUnit:'—')
        ]
      };
    }
  }
  if (c044Focus.kind==='airport') {
    var node=c044AirportMap[c044Focus.key];
    if (node) {
      var sum=node.departures+node.arrivals;
      return {
        heading:node.name+'  ·  当前样本关联'+setting.routeLabel,
        metrics:[
          m('关联'+setting.routeLabel,node.links+' 条'),
          m('关联'+setting.metricName,c044Format(sum)+' '+setting.metricUnit),
          m('当前样本占比',(c044Total?100*sum/c044Total:0).toFixed(2)+'%'),
          m(setting.originLabel+' / '+setting.destinationLabel,c044Format(node.departures)+' / '+c044Format(node.arrivals))
        ]
      };
    }
  }
  // 取消选择后展示当前 Top N 样本概览。这里的全部数字也由输入数据自动计算。
  return {
    heading:'当前样本概览',
    metrics:[
      m('单向'+setting.routeLabel,c044Rows.length+' 条'),
      m(setting.nodeLabel,c044Airports.length+' 个'),
      m('样本'+setting.metricName,c044Format(c044Total)+' '+setting.metricUnit),
      m('最大单向'+setting.metricName,c044Rows.length?c044Format(c044Rows[0].value)+' '+setting.metricUnit:'—')
    ]
  };
}

function c044BindEvents(inst){
  if (!inst || typeof inst.on!=='function') return;
  // 重跑编辑器代码时只移除本案例自身的旧监听器，不影响宿主其他事件。
  if (inst.__c044ClickCallback && typeof inst.off==='function') inst.off('click',inst.__c044ClickCallback);
  var zr=typeof inst.getZr==='function' ? inst.getZr() : null;
  if (zr && inst.__c044BlankCallback && typeof zr.off==='function') zr.off('click',inst.__c044BlankCallback);
  // ECharts 先转发 series / geo click，再由 ZRender 通知底层 click。
  // 同一鼠标事件内避免 ZRender 的空白识别抵消刚做出的选择。
  inst.__c044HandledShapeClick=false;
  inst.__c044ClickCallback=function(params){
    if(params && params.seriesId==='c044-airports' && params.data){
      inst.__c044HandledShapeClick=true;
      c044Focus.kind='airport';c044Focus.key=params.data.name;
      c044Update(inst);
    } else if(params && params.seriesId==='c044-route-lines' && params.data){
      inst.__c044HandledShapeClick=true;
      c044Focus.kind='route';c044Focus.key=params.data.rowId;
      c044Update(inst);
    } else if(params && params.componentType==='geo') {
      // 单击灰色省区背景也是“空白处”：取消聚焦，不把省份当成航线选中。
      inst.__c044HandledShapeClick=true;
      c044ClearFocus(inst);
    }
  };
  inst.on('click',inst.__c044ClickCallback);
  // 图表由 Power BI 或官方编辑器 resize 后，根据新宽度自动切换 4列/2列详情。
  // rendered 只检查尺寸变化；宽度不变时不再 setOption，避免形成渲染循环。
  if(inst.__c044ResizeCheck && typeof inst.off==='function') inst.off('rendered',inst.__c044ResizeCheck);
  inst.__c044LastLayoutWidth=c044ChartWidth(inst);
  inst.__c044ResizeCheck=function(){
    if(!setting.showSelectionDetails)return;
    var width=c044ChartWidth(inst);
    if(Math.abs(width-inst.__c044LastLayoutWidth)<1)return;
    inst.__c044LastLayoutWidth=width;
    c044ApplyDetailLayout(inst,true);
  };
  inst.on('rendered',inst.__c044ResizeCheck);
  if (zr && typeof zr.on==='function') {
    inst.__c044BlankCallback=function(event){
      if (inst.__c044HandledShapeClick){inst.__c044HandledShapeClick=false;return;}
      // 白色空白画布没有 ZRender 图形命中目标。
      if (!event || !event.target) c044ClearFocus(inst);
    };
    zr.on('click',inst.__c044BlankCallback);
  }
}
