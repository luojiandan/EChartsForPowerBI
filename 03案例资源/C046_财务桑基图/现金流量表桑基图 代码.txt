// ======================= 3. 数据校验与现金流计算 =======================
var C = setting.colors;
var groups = { '经营活动': [], '投资活动': [], '筹资活动': [] };
var openingCash = null;
var reportedEndingCash = null;
if (accountNames.length !== activityTypes.length || activityTypes.length !== cashFlowValues.length) {
  throw new Error('accountNames、activityTypes、cashFlowValues 三个数组长度必须一致');
}
accountNames.forEach(function (name, i) {
  var type = activityTypes[i];
  var value = Number(cashFlowValues[i]);
  if (!Number.isFinite(value) || String(name).trim() === '') {
    throw new Error('第 ' + (i + 1) + ' 行名称或金额不合法');
  }
  if (type === '期初现金') openingCash = value;
  else if (type === '期末现金') reportedEndingCash = value;
  else if (groups[type]) groups[type].push({ name: String(name), value: value, index: i });
  else throw new Error('未知活动类型：' + type);
});
if (openingCash === null || reportedEndingCash === null) {
  throw new Error('必须提供期初现金及期末现金余额');
}
function sumRows(rows) {
  return rows.reduce(function (total, row) { return total + row.value; }, 0);
}
function positiveRows(rows) { return rows.filter(function (row) { return row.value > 0; }); }
function negativeRows(rows) { return rows.filter(function (row) { return row.value < 0; }); }
function sortRowsDesc(rows) {
  return rows.slice().sort(function (a, b) {
    return Math.abs(b.value) - Math.abs(a.value) || b.value - a.value || a.index - b.index;
  });
}
function positiveTotal(rows) { return positiveRows(rows).reduce(function (s, row) { return s + row.value; }, 0); }
function negativeAbsTotal(rows) { return negativeRows(rows).reduce(function (s, row) { return s + Math.abs(row.value); }, 0); }
function fmt(n, showPlus) {
  var factor = Math.pow(10, setting.decimals);
  var scaled = Math.abs(Number(n)) / setting.divisor * factor;
  var rounded = Math.round(scaled + 1e-9) / factor;
  var text = rounded.toLocaleString('en-US', {
    minimumFractionDigits: setting.decimals,
    maximumFractionDigits: setting.decimals
  });
  if (n < 0) return '-' + text;
  if (showPlus && n > 0) return '+' + text;
  return text;
}
function pct(n, d) { return d ? (Math.abs(n) / d * 100).toFixed(1) + '%' : '—'; }
function htmlSafe(s) {
  return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;')
    .replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}

var invNegative = sortRowsDesc(negativeRows(groups['投资活动']));

var opPositiveTotal = positiveTotal(groups['经营活动']);
var opNegativeTotal = negativeAbsTotal(groups['经营活动']);

var invNegativeTotal = negativeAbsTotal(groups['投资活动']);
var finPositiveTotal = positiveTotal(groups['筹资活动']);
var finNegativeTotal = negativeAbsTotal(groups['筹资活动']);
var operatingCashFlow = opPositiveTotal - opNegativeTotal;
var investingCashFlow = invPositiveTotal - invNegativeTotal;
var financingCashFlow = finPositiveTotal - finNegativeTotal;
var netCashIncrease = operatingCashFlow + investingCashFlow + financingCashFlow;
var calculatedEndingCash = openingCash + netCashIncrease;

// 现金来源合计：期初现金 + 经营活动净现金流 + 投资/筹资活动正向现金流。
// 这是为了展示现金收支平衡而计算的中间节点，并非报表单列科目。

  throw new Error('现金余额不平：期初现金 + 现金净增加额 ≠ 报表期末现金');
}
if (Math.abs(cashSourceTotal - cashUsesTotal) > 0.00001) {
  throw new Error('桑基图现金收支不平：现金来源合计 ≠ 现金用途合计');
}

var nodes = [];

var nodeIndex = {};
function addNode(id, label, value, signedValue, color, position, depth, extra) {
  var n = {
    name: id,
    displayName: label,
    value: value,
    signedValue: signedValue,
    depth: depth,
    itemStyle: { color: color, borderWidth: 0 },
    label: {
      show: setting.showNodeLabels,
      position: position || 'right',
      distance: 6,
      align: position === 'left' ? 'right' : 'left',
      formatter: function (p) {
        return '{nm|' + p.data.displayName + '}\n{val|' + fmt(p.data.signedValue, false) + '}';
      }
    }
  };
  if (extra) Object.keys(extra).forEach(function (key) { n[key] = extra[key]; });
  nodes.push(n);
  nodeIndex[id] = n;
}
function addLink(source, target, value, displayName, signedValue) {
  if (value > 0) links.push({
    source: source,
    target: target,
    value: value,
    displayName: displayName || (nodeIndex[source].displayName + ' → ' + nodeIndex[target].displayName),
    signedValue: signedValue === undefined ? value : signedValue
  });
}

var sankeyLayoutScale = 0;
// 4.4 手动定位：严格复刻 ECharts Sankey 内部的节点高度比例尺。
// ECharts computeNodeValues 使用 max(节点原始值, 入边总和, 出边总和)。
// ECharts initializeNodeDepth 使用 min((height-(同列节点数-1)*nodeGap)/同列节点值总和)。
// 之前的错误是用“标签间距约束”另算了更小的比例尺，但 ECharts 仍按自己的比例尺绘制节点，
// 导致 localY 预留位置小于实际节点高度，进而出现节点相互覆盖。本版统一使用同一个 ky。
function assignManualNodePositions() {
  var plotHeight = setting.layout.height;
  var columns = {};
  var layoutValues = {};

  nodes.forEach(function (n) {
    var incoming = links.reduce(function (sum, e) {
      return sum + (e.target === n.name ? e.value : 0);
    }, 0);
    var outgoing = links.reduce(function (sum, e) {
      return sum + (e.source === n.name ? e.value : 0);
    }, 0);
    // 与 ECharts computeNodeValues 保持一致。
    layoutValues[n.name] = Math.max(Number(n.value) || 0, incoming, outgoing);
    var key = String(n.depth);
    if (!columns[key]) columns[key] = [];
    columns[key].push(n);
  });

  // 与 ECharts initializeNodeDepth 完全一致：全图所有深度列共用最小 ky。
  var minKy = Infinity;
  Object.keys(columns).forEach(function (key) {
    var column = columns[key];
    var total = column.reduce(function (sum, n) { return sum + layoutValues[n.name]; }, 0);
    var ky = (plotHeight - (column.length - 1) * setting.nodeGap) / total;
    if (!(ky > 0)) throw new Error('桑基图绘图区高度不足，请降低 nodeGap 或增加 layout.height');
    minKy = Math.min(minKy, ky);
  });
  sankeyLayoutScale = minKy;

  Object.keys(columns).forEach(function (key) {
    var column = columns[key];
    var nodeHeightTotal = column.reduce(function (sum, n) {
      return sum + layoutValues[n.name] * minKy;
    }, 0);
    var baseGapTotal = Math.max(0, column.length - 1) * setting.nodeGap;
    var extraGapTotal = column.reduce(function (sum, n, i) {
      return sum + (i < column.length - 1 ? (n.gapAfter || 0) : 0);
    }, 0);
    var usedHeight = nodeHeightTotal + baseGapTotal + extraGapTotal;
    if (usedHeight > plotHeight + 0.01) {
      throw new Error('第 ' + key + ' 层自定义间距超出绘图区，请减少该层 gapAfter');
    }
    var cursor = (plotHeight - usedHeight) / 2;
    column.forEach(function (n, i) {
      n.localY = cursor / plotHeight;
      cursor += layoutValues[n.name] * minKy;
      if (i < column.length - 1) cursor += setting.nodeGap + (n.gapAfter || 0);
    });
  });
}

  if (!nodeIndex[id]) return;

  var outgoing = links.filter(function (e) { return e.source === id; }).reduce(function (s, e) { return s + e.value; }, 0);
  if (Math.abs(incoming - outgoing) > 0.00001) {
    throw new Error('桑基图金额不平：' + nodeIndex[id].displayName + '，流入=' + incoming + '，流出=' + outgoing);
  }
});

// 仅设置 left/right 百分比不够：Sankey 的节点高度按画布高度重新计算，
// 而 V6 的 localY 已固定为基准布局的比例。若 nodeGap 仍为固定 23px，缩放后
// 实际节点高度和 localY 的位置比例会失配，导致节点或金额标签重叠。
//
// 解决方案：本例所有 localY 都来自基准高度 580px 的严格勾稽计算；
// 各 media 高度档同步缩放 nodeGap，使节点高度/空白比例和 localY 一起同比缩放。
// media 是 ECharts 内置机制，图表 resize() 时自动切换档位，不依赖外部监听。
// 每 50px 一档；取该档上限计算 nodeGap，确保相邻节点不发生矩形碰撞。
// 说明：文字有最低物理像素尺寸，小画布会有选择地隐藏次要明细标签，
//       但保留全部流线和悬停 Tooltip，不合并或删改财务数据。

function cloneNodeWithLabel(n, widthMode, chartH) {
  var n2 = Object.assign({}, n);
  var label2 = Object.assign({}, n.label);
  var magnitude = Math.abs(n.signedValue);
  var important = n.name.charAt(0) === '@' ||
    n.name === 'fin-buyback' || n.name === 'inv-out#11';
  if (chartH < 630) {
    // 高度不足时隐藏小额叶节点，仅保留核心汇总与大额流线标签。
    label2.show = important || magnitude >= 24000;
  } else if (chartH < 750) {
    label2.show = important || magnitude >= 9500;
  }
  if (widthMode === 'mini') {
    // 画布非常窄时，只保留五个财务主节点以及期初/期末现金。
    label2.show = n.name === '@cash-source' || n.name === '@opening-cash' ||
      n.name === '@ending-cash' || n.name === '@operating-positive' ||
      n.name === '@investing-negative' || n.name === '@financing-negative';
  } else if (widthMode === 'small') {
    label2.show = label2.show && (important || magnitude >= 16000);
  }
  // 保持节点完整值与原始名称。标签简化不影响 Tooltip 或财务计算。
  label2.show = Boolean(setting.showNodeLabels && label2.show);
  n2.label = label2;
  return n2;
}
