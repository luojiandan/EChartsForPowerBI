var C = setting.colors;
var TYPES = ['流动资产', '非流动资产', '流动负债', '非流动负债', '股东权益'];
var group = {};
TYPES.forEach(function (t) { group[t] = []; });
if (accountNames.length !== accountTypes.length || accountTypes.length !== accountValues.length) {
  throw new Error('accountNames、accountTypes、accountValues 数组长度不一致');
}
accountNames.forEach(function (label, i) {
  var t = accountTypes[i];
  var v = Number(accountValues[i]);
  if (!group[t] || !Number.isFinite(v)) throw new Error('分类或金额不合法：第 ' + (i + 1) + ' 行');
  if (t !== '股东权益' && v < 0) throw new Error('资产/负债明细应非负；股东权益明细可为负数');
  if (String(label).trim() === '') throw new Error('科目名称不能为空：第 ' + (i + 1) + ' 行');
  group[t].push({ name: String(label), value: v, index: i });
});
function sum(data) { return data.reduce(function (s, v) { return s + v.value; }, 0); }
function fmt(n) {
  // 使用十进制四舍五入，避免 645.25 等金额因浮点格式化显示为 645.2。
  var factor = Math.pow(10, setting.decimals);
  var scaled = Number(n) / setting.divisor * factor;
  var rounded = Math.sign(scaled) * Math.round(Math.abs(scaled) + 1e-10) / factor;
  return rounded.toLocaleString('en-US', {
    minimumFractionDigits: setting.decimals,
    maximumFractionDigits: setting.decimals
  });
}
function pct(a, b) { return b ? (a / b * 100).toFixed(1) + '%' : '—'; }
function htmlSafe(s) {
  return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;')
    .replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}
var ca = sum(group['流动资产']);
var na = sum(group['非流动资产']);
var cl = sum(group['流动负债']);
var nl = sum(group['非流动负债']);
var eq = sum(group['股东权益']);
var totalAsset = ca + na;
var totalLiability = cl + nl;
if (totalAsset <= 0 || ca < 0 || na < 0 || cl < 0 || nl < 0 || eq < 0) {
  throw new Error('模板要求资产、负债及净权益非负；净权益为负的企业需改用专门的负权益结构');
}
if (Math.abs(totalAsset - totalLiability - eq) > 0.00001) {
  throw new Error('资产负债表不平衡：总资产(' + totalAsset + ') ≠ 总负债(' + totalLiability + ') + 股东权益(' + eq + ')');
}

// ======================= 4. 桑基图：左资产 → 总资产 → 右负债与权益 =======================
var nodes = [];
var links = [];
var nodeIndex = {};
function addNode(id, label, value, color, position, depth, extra) {
  var n = {
    name: id,
    displayName: label,
    value: value,
    depth: depth,
    itemStyle: { color: color, borderWidth: 0 },
    label: {
      show: setting.showNodeLabels,
      position: position || 'right',
      distance: 7,
      align: position === 'left' ? 'right' : 'left',
      formatter: function (p) {
        return '{nm|' + p.data.displayName + '}\n{val|' + fmt(p.data.value) + '}';
      }
    }
  };
  if (extra) Object.keys(extra).forEach(function (key) { n[key] = extra[key]; });
  nodes.push(n);
  nodeIndex[id] = n;
}
function addLink(source, target, value) {
  if (value > 0) links.push({ source: source, target: target, value: value });
}

// 股东权益中有负科目（例如累计其他综合损失），不可将其画成负宽度流线。
// 正确做法：保留全部原始有符号科目，对可抵消部分显示净额，并在 Tooltip 中展开。
// 无负科目时，各权益科目原样拆分；有负科目时优先单独展示最大正项，其余按净额展示。
var equityRows = group['股东权益'];
var negativeEquityRows = equityRows.filter(function (r) { return r.value < 0; });
if (negativeEquityRows.length === 0) {
  equityRows.forEach(function (r) {
    var id = 'equity#' + r.index;
    addNode(id, r.name, r.value, C.positive, 'right', 4);
    addLink(ID.equity, id, r.value);
  });
} else {
  var positiveEquityRows = equityRows.filter(function (r) { return r.value > 0; });
  positiveEquityRows.sort(function (a, b) { return b.value - a.value; });
  var main = positiveEquityRows[0];
  var remainder = main ? eq - main.value : eq;
  if (main && remainder >= 0) {
    var mainId = 'equity#' + main.index;
    addNode(mainId, main.name, main.value, C.positive, 'right', 4);
    addLink(ID.equity, mainId, main.value);
    var details = equityRows.filter(function (r) { return r.index !== main.index; });
    if (remainder > 0) {
      addNode('@equity-net', setting.names.equityNet, remainder, C.positive, 'right', 4,
        { detailRows: details, note: '有符号权益科目的净额，已扣除累计其他综合损失' });
      addLink(ID.equity, '@equity-net', remainder);
    }
  } else {
    // 少数特殊情形：权益存在负项且单个最大正项超过权益净额时，整体显示净权益。
    addNode('@equity-net', setting.names.equityNet, eq, C.positive, 'right', 4,
      { detailRows: equityRows, note: '股东权益全部科目合计净额' });
    addLink(ID.equity, '@equity-net', eq);
  }
}

// ======================= 4.1 手动垂直排布（保持语义分组与自然曲线） =======================
// ECharts 默认会根据连线关系自动调整同层节点顺序，可能把业务分组打散。
// 这里按 data 数组顺序计算各层节点的 localY：
// 左侧：流动资产在上、非流动资产在下；右侧：流动负债、非流动负债、股东权益。
// 通过 layoutIterations: 0 固定内部边缘顺序，再用 localY 居中摆放节点。
// 连线仍由 lineStyle.curveness 的原生 Bezier 曲线绘制，避免关闭迭代后布局挤在顶部。
function assignManualNodePositions() {
  var plotHeight = setting.layout.height;
  var columns = {};
  nodes.forEach(function (n) {
    var key = String(n.depth);
    if (!columns[key]) columns[key] = [];
    columns[key].push(n);
  });

  // 统一计算桑基图节点高度比例，确保每一层的流量宽度口径一致。
  var scale = Infinity;
  Object.keys(columns).forEach(function (key) {
    var column = columns[key];
    var total = column.reduce(function (s, n) { return s + n.value; }, 0);
    var available = plotHeight - Math.max(0, column.length - 1) * setting.nodeGap;
    if (available <= 0 || total <= 0) throw new Error('桑基图绘图区高度或节点数据不合法');
    scale = Math.min(scale, available / total);
  });

  // 各层独立垂直居中，节点顺序严格遵循 nodes 数组顺序。
  Object.keys(columns).forEach(function (key) {
    var column = columns[key];
    var usedHeight = column.reduce(function (s, n) { return s + n.value * scale; }, 0)
      + Math.max(0, column.length - 1) * setting.nodeGap;
    var cursor = (plotHeight - usedHeight) / 2;
    column.forEach(function (n) {
      n.localY = cursor / plotHeight;
      cursor += n.value * scale + setting.nodeGap;
    });
  });
}
assignManualNodePositions();

// 中间节点的每层流入、流出必须相等。
[ID.ca, ID.na, ID.assets, ID.cl, ID.nl, ID.equity].forEach(function (id) {
  var incoming = links.filter(function (e) { return e.target === id; })
    .reduce(function (s, e) { return s + e.value; }, 0);
  var outgoing = links.filter(function (e) { return e.source === id; })
    .reduce(function (s, e) { return s + e.value; }, 0);
  if (Math.abs(incoming - outgoing) > 0.00001) {
    throw new Error('桑基图金额不平：' + nodeIndex[id].displayName);
  }
});

function responsiveNode(n, widthMode, chartHeight) {
  var cloned = Object.assign({}, n);
  var label = Object.assign({}, n.label);
  var isLeaf = n.depth === 0 || n.depth === 4;
  var isSummary = n.depth === 1 || n.depth === 2 || n.depth === 3;
  var isMajorLeaf = isLeaf && n.value >= 40000;
  if (widthMode === 'mini') {
    label.show = n.name === '@total-assets' || n.name === '@current-liabilities' ||
      n.name === '@equity';
  } else if (widthMode === 'small') {
    label.show = isSummary || (isMajorLeaf && chartHeight >= 700);
  } else if (widthMode === 'medium') {
    label.show = isSummary || (isLeaf && n.value >= 15000 && chartHeight >= 620);
  } else if (chartHeight < 600) {
    label.show = isSummary || isMajorLeaf;
  } else if (chartHeight < 750) {
    label.show = isSummary || n.value >= 22000;
  }
  label.show = !!setting.showNodeLabels && !!label.show;
  cloned.label = label;
  return cloned;
}

            '<span>' + htmlSafe(r.name) + '</span><b style="color:' + vColor + ';">' + htmlSafe(valueText) + '</b></div>';
        });
      }
      if (!isEdge && d.note) {
        block += '<div style="margin-top:8px;font-size:10px;color:' + C.muted + ';">' + htmlSafe(d.note) + '</div>';
      }
      return block;
    }
  },

// ======================= 7. 自适应布局：ECharts 原生 media =======================
// 横向使用百分比，纵向图高 = 实际画布高度 - top - bottom。
// 原始节点 localY 是按基准绘图区 514px 和 nodeGap=23px 计算的。
// 因此，图高变化时同步按比例缩放 nodeGap，避免节点高度失配再次造成重叠。
// 每 50px 建立一个尺寸档；取高度区间上限作为保守 gap。
// 小画布会隐藏次要明细标签，但不删除数据或流线，可通过 Tooltip 查看金额。
var responsiveBaseOption = option;
var responsiveMedia = [];
var widthBands = [
  { min: 0, max: 519, mode: 'mini', ref: 300, left: '13%', right: '12%' },
  { min: 520, max: 759, mode: 'small', ref: 520, left: '16%', right: '14%' },
  { min: 760, max: 999, mode: 'medium', ref: 760, left: '18.5%', right: '17.5%' },
  { min: 1000, max: 1199, mode: 'wide', ref: 1000, left: setting.responsive.left, right: setting.responsive.right },
  { min: 1200, max: 99999, mode: 'wide', ref: 1200, left: setting.responsive.left, right: setting.responsive.right }
];
var heightBands = [{ min: 0, max: 350, tooShort: true }];
for (var startHeight = 351; startHeight <= 2351; startHeight += 50) {
  heightBands.push({ min: startHeight, max: startHeight + 49 });
}
heightBands.push({ min: 2401, max: 99999, effectiveMax: 2800 });
widthBands.forEach(function (wb) {
  heightBands.forEach(function (hb) {
    var refHeight = hb.effectiveMax || hb.max;
    var compactHeight = refHeight < 700;
    var shortHeight = refHeight < 480;
    var top = shortHeight ? setting.responsive.shortTop :
      compactHeight ? setting.responsive.compactTop : setting.responsive.top;
    var bottom = shortHeight ? setting.responsive.shortBottom :
      compactHeight ? setting.responsive.compactBottom : setting.responsive.bottom;
    var refPlotHeight = Math.max(1, refHeight - top - bottom);
    var gap = Math.max(0.5, setting.responsive.referenceGap * refPlotHeight /
      setting.responsive.referenceHeight);
    var data = nodes.map(function (n) { return responsiveNode(n, wb.mode, refHeight); });
    var f = responsiveGraphics(wb.mode, compactHeight, !!hb.tooShort, wb.ref);
    responsiveMedia.push({
      query: { minWidth: wb.min, maxWidth: wb.max, minHeight: hb.min, maxHeight: hb.max },
      option: {
        graphic: f,
        series: [{
          left: wb.left,
          right: wb.right,
          top: top,
          bottom: bottom,
          nodeGap: gap,
          nodeWidth: wb.mode === 'mini' ? 9 : wb.mode === 'small' ? 11 : setting.nodeWidth,
          data: hb.tooShort ? [] : data,
          links: hb.tooShort ? [] : links
        }]
      }
    });
  });
});

option = { baseOption: responsiveBaseOption, media: responsiveMedia };
