const periodList = [];
const categoryList = [];
const seenPeriods = Object.create(null);
const seenCategories = Object.create(null);
const dataByPeriod = Object.create(null);
let observedMaxRank = 1;
function parseRank(value, label, row) {
  if (value === null || value === undefined || String(value).trim() === '') return null;
  const n = Number(value);
  if (!Number.isInteger(n) || n < 1) throw new Error('C022：第' + row + '条记录' + label + '应为正整数。');
  return n;
}
function parseMetric(value, label, row) {
  if (value === null || value === undefined || String(value).trim() === '') return null;
  const n = Number(value);
  if (!Number.isFinite(n)) throw new Error('C022：第' + row + '条记录' + label + '应为数字。');
  return n;
}
for (let i = 0; i < periods.length; i++) {
  const period = String(periods[i] == null ? '' : periods[i]).trim();
  const category = String(categories[i] == null ? '' : categories[i]).trim();
  if (!period || !category) throw new Error('C022：第' + (i + 1) + '条记录周期或对象为空。');
  if (!seenPeriods[period]) {
    seenPeriods[period] = true;
    periodList.push(period);
    dataByPeriod[period] = Object.create(null);
  }
  if (!seenCategories[category]) {
    seenCategories[category] = true;
    categoryList.push(category);
  }
  if (Object.prototype.hasOwnProperty.call(dataByPeriod[period], category)) {
    throw new Error('C022：周期' + period + '与对象' + category + '重复。');
  }
  const rank = parseRank(ranks[i], '排名', i + 1);
  const metrics = metricFieldArrays.map(function (values, j) {
    return parseMetric(values[i], '指标' + (j + 1), i + 1);
  });
  if (rank !== null) observedMaxRank = Math.max(observedMaxRank, rank);
  dataByPeriod[period][category] = { rank: rank, metrics: metrics };
}

// 四、排名显示与名次变化；不以子集重新计算全球名次
const latestPeriod = periodList[periodList.length - 1];
function getRecord(period, category) {
  return dataByPeriod[period] && dataByPeriod[period][category];
}
function rankText(value) { return value == null ? '无排名' : '第' + value + '名'; }
function metricText(value) {
  if (value == null) return '—';
  const digits = Math.max(0, Math.min(8, Math.floor(Number(setting.metricDecimalPlaces) || 0)));
  return Number(value).toLocaleString('en-US', { maximumFractionDigits: digits });
}
function metricsText(record) {
  return record && record.metrics
    ? record.metrics.map(metricText).join(' / ') : '—';
}
function escapeHtml(value) {
  return String(value == null ? '' : value)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}
function rankChange(period, category) {
  const index = periodList.indexOf(period);
  if (index <= 0) return { text: '首个周期', color: setting.colors.textSecondary };
  const previous = getRecord(periodList[index - 1], category);
  const current = getRecord(period, category);
  if (!previous || !current || previous.rank == null || current.rank == null) {
    return { text: '无可比排名', color: setting.colors.textSecondary };
  }
  const delta = previous.rank - current.rank;
  if (delta === 0) return { text: '名次持平', color: setting.colors.textSecondary };
  return delta > 0
    ? { text: '▲ 提升 ' + delta + ' 位', color: setting.colors.positive }
    : { text: '▼ 下降 ' + Math.abs(delta) + ' 位', color: setting.colors.negative };
}
// 首次出现的类别序号就是色板序号；未知类别回退为中性参考色。
const colorByCategory = Object.create(null);
categoryList.forEach(function (category, index) {
  colorByCategory[category] = setting.seriesColors[index % setting.seriesColors.length];
});
function colorOf(category) {
  return colorByCategory[category] || setting.colors.barReference;
}
const axisMax = setting.rankMax == null

// 五、系列：同一个 custom 系列包含全部类别；每个类别是独立数据项。
// 关键：ECharts 6 CustomView 从 renderItem 返回图形的【根级 focus、blurScope】读取悬停策略。
//       写在 series.emphasis 或 data[i].emphasis 对自定义图形的相互淡化不会生效。
//       同系列的其他国家数据项进入 blur，由各子图形 blur.style.opacity 控制透明度。
// 原生透明 line 系列仅负责按周期的 axis Tooltip，不参与绘制或类别高亮。
// 统一使用圆形节点：颜色区分类别，不再使用不同符号形状编码类别。
function markerGraphic(x, y, size, color) {
  return {
    type: 'circle', silent: true,
    shape: { cx: x, cy: y, r: size / 2 },
    style: { fill: color, stroke: '#FFFFFF', lineWidth: 1.1, opacity: 1 },
    emphasis: { style: { opacity: 1 } },
    blur: { style: { opacity: setting.blurOpacity } }
  };
}

function categoryTooltip(category) {
  const color = colorOf(category);
  let html = '<div style="font-size:14px;font-weight:700;margin-bottom:8px;color:' + color + '">' +
    escapeHtml(category) + ' · ' + escapeHtml(setting.historyTitleSuffix) + '</div>' +
    '<table style="border-collapse:collapse;font-size:11px;white-space:nowrap;line-height:1.65;">' +
    '<thead><tr style="color:' + setting.colors.textSecondary + ';border-bottom:1px solid ' + setting.colors.gridLine + '">' +
    '<th style="text-align:left;padding:3px 12px 5px 0;">' + escapeHtml(setting.periodColumnLabel) + '</th>' +
    '<th style="text-align:left;padding:3px 12px 5px 0;">' + escapeHtml(setting.rankColumnLabel) + '</th>' +
    '<th style="text-align:left;padding:3px 12px 5px 0;">名次变化</th>' +
    '<th style="text-align:right;padding:3px 0 5px 0;">' + escapeHtml(setting.metricLabels.join(' / ')) + '</th></tr></thead><tbody>';
  periodList.forEach(function (period) {
    const record = getRecord(period, category);
    const change = rankChange(period, category);
    const rank = record && record.rank != null ? rankText(record.rank) : '—';
    const metricSummary = metricsText(record);
    html += '<tr style="border-bottom:1px solid #F0F2F4;">' +
      '<td style="padding:5px 12px 5px 0;">' + escapeHtml(period) + '</td>' +
      '<td style="padding:5px 12px 5px 0;font-weight:600;">' + escapeHtml(rank) + '</td>' +
      '<td style="padding:5px 12px 5px 0;color:' + change.color + ';">' +
      (setting.showRankChange ? escapeHtml(change.text) : '—') + '</td>' +
      '<td style="padding:5px 0;text-align:right;">' + escapeHtml(metricSummary) + '</td></tr>';
  });
  const first = getRecord(periodList[0], category);
  const last = getRecord(latestPeriod, category);
  html += '</tbody></table>' +
    '<div style="display:flex;justify-content:space-between;gap:14px;margin-top:9px;font-size:11px;">' +
    '<span style="color:' + setting.colors.textSecondary + ';">' + escapeHtml(periodList[0]) + '—' + escapeHtml(latestPeriod) + ' · ' + escapeHtml(setting.rankScopeLabel) + '</span>' +
    '<strong>' + escapeHtml(rankText(first && first.rank)) + ' → ' + escapeHtml(rankText(last && last.rank)) + '</strong></div>';
  return html;
}

// 单个 custom 系列中每个类别为一个数据项：每项包含完整线段、节点及末端标签。
// 每个类别的根图形组设置 focus=self；blurScope=series 淡化其他类别。
const series = [{
  name: '排名轨迹',
  type: 'custom',
  coordinateSystem: 'cartesian2d',
  clip: false,
  z: 10,
  data: categoryList.map(function (category) {
    const latest = getRecord(latestPeriod, category);
    return {
      name: category,
      value: [periodList.length - 1, latest && latest.rank != null ? latest.rank : setting.rankMin],
    };
  }),
  encode: { x: 0, y: 1 },
  tooltip: { trigger: 'item' },
  renderItem: function (params, api) {
    const category = categoryList[params.dataIndex];
    const color = colorOf(category);
    const latest = getRecord(latestPeriod, category);
    const children = [];
    let prevPoint = null;
    let lastPoint = null;
    periodList.forEach(function (period, index) {
      const record = getRecord(period, category);
      if (!record || record.rank == null) { prevPoint = null; return; }
      const point = api.coord([index, record.rank]);
      if (!Number.isFinite(point[0]) || !Number.isFinite(point[1])) { prevPoint = null; return; }
      if (prevPoint) {
        children.push({
          type: 'line', silent: true,
          shape: { x1: prevPoint[0], y1: prevPoint[1], x2: point[0], y2: point[1] },
          style: { stroke: color, lineWidth: setting.lineWidth, opacity: 0.95 },
          emphasis: { style: { lineWidth: setting.hoverLineWidth, opacity: 1 } },
          blur: { style: { opacity: setting.blurOpacity } }
        });
      }
      if (setting.showSymbols) {
        children.push(markerGraphic(point[0], point[1], setting.symbolSize, color));
      }
      prevPoint = point;
      if (index === periodList.length - 1) lastPoint = point;
    });
    if (setting.showEndLabels && lastPoint && latest) {
      const x = lastPoint[0] + 8;
      const y = lastPoint[1] - setting.labelHitHeight / 2;
      // 仅标签矩形接收鼠标命中，其他图形 silent 避免触发无关悬停。
      children.push({
        type: 'rect',
        shape: { x: x, y: y, width: setting.labelHitWidth, height: setting.labelHitHeight, r: 3 },
        style: { fill: 'rgba(255,255,255,0.92)', opacity: 1 },
        emphasis: { style: { fill: 'rgba(244,247,249,0.99)', opacity: 1 } },
        blur: { style: { opacity: setting.blurOpacity } }
      });
      children.push({
        type: 'text', silent: true,
        style: {
          x: x + 5, y: lastPoint[1],
          text: category + '  ' + rankText(latest.rank),
          fill: color, font: '600 11px sans-serif',
          textVerticalAlign: 'middle', textAlign: 'left', opacity: 1
        },
        emphasis: { style: { opacity: 1 } },
        blur: { style: { opacity: setting.blurOpacity } }
      });
    }
    // 根级配置！CustomView.createOrUpdateItem 调用 toggleHoverEmphasis(el, elOption.focus, elOption.blurScope)。
    // 放在 series.emphasis / data.emphasis 中，标签 hover 仅会加粗当前线条，不会触发其他国家 blur。
    return {
      type: 'group',
      focus: 'self',
      blurScope: 'series',
      children: children
    };
  }
}];

// 透明原生折线负责按年份触发轴 Tooltip。
categoryList.forEach(function (category) {
  const values = periodList.map(function (period) {
    const record = getRecord(period, category);
    return record && record.rank != null ? record.rank : null;

  series.push({
    name: category,
    type: 'line',
    data: values,
    showSymbol: false,
    symbolSize: 0,
    smooth: false,
    connectNulls: false,
    silent: true,
    lineStyle: { opacity: 0, width: 0 },
    itemStyle: { opacity: 0 },
    tooltip: { trigger: 'axis' },
    z: 0
  });
});
