function stableBusinessCompare(a,b,order){
  var ia=order.indexOf(String(a)),ib=order.indexOf(String(b));
  if(ia>=0 && ib>=0)return ia-ib;
  if(ia>=0)return -1;
  if(ib>=0)return 1;
  return String(a).localeCompare(String(b),'zh-CN');
}

function drawVariance(i,cy,delta,effect,mode,bold){
  var x=lefts[i]+2, w=widths[i]-4;

    var ex=delta/valueMax*span, col=effectColor(effect);

      graphics.push(graphicLine(center,cy,center+ex,cy,col,2));
      graphics.push(graphicLine(center+ex,cy-thickness/2-2,center+ex,cy+thickness/2+2,col,2));

      graphics.push(graphicRect(ex>=0?center:center+ex,cy-thickness/2,Math.abs(ex),thickness,col,2));
    }

  var label=mode==='percent'?percent(delta):signed(delta);
  graphics.push(graphicText(x+w-1,cy,label,valSize,bold,effectColor(effect),'right',w*0.44,6));
}

        '<td style="text-align:right;font-weight:600;color:'+(color||C.text)+'">'+esc(v)+'</td></tr>';
    }
    return '<div style="min-width:245px;font-family:'+setting.font+';line-height:1.5;color:'+C.text+'"><b>'+esc(r.group)+' · '+esc(r.name)+'</b><table style="margin-top:8px;width:100%;border-collapse:collapse">'+
      tr(ac+' (AC)',unitVal(r.ac)+' '+setting.unit)+tr(py+' (PY)',unitVal(r.py)+' '+setting.unit)+
      (showPLColumns?tr(setting.planIsSimulated?'PL（教学模拟）':'PL',r.pl===null?'未提供':unitVal(r.pl)+' '+setting.unit):'')+
      tr('△PY 金额',signed(r.delta)+' '+setting.unit,effectColor(r.effect))+
      tr('△PY%',percent(r.pct),effectColor(r.effect))+
      (showPLColumns?
        tr('△PL 金额',r.planDelta===null?'—':signed(r.planDelta)+' '+setting.unit,
           r.planEffect===null?C.secondary:effectColor(r.planEffect))+
        tr('△PL%',percent(r.planPct),
           r.planEffect===null?C.secondary:effectColor(r.planEffect))
      :'')+
      '</table></div>';
  }},
  series:[tooltipSeries]
};
