function normalizePeriod(value) {
  // 支持 YYYY-MM、YYYY/MM、YYYYMM、YYYY-MM-DD、ISO 日期及 Date 对象。
  if (Object.prototype.toString.call(value)==='[object Date]') {
    if (isNaN(value.getTime())) return null;
    return String(value.getFullYear())+'-'+('0'+(value.getMonth()+1)).slice(-2);
  }
  var s = String(value == null ? '' : value).trim();
  var m = /^(\d{4})[-\/]?(\d{1,2})(?:[-\/]\d{1,2}(?:[ T].*)?)?$/.exec(s);
  if (!m) return null;
  var year = Number(m[1]);
  var month = Number(m[2]);
  return month >= 1 && month <= 12 ? String(year) + '-' + ('0' + month).slice(-2) : null;
}
function monthSerial(period) {
  var p = normalizePeriod(period);
  if (!p) return null;
  return Number(p.slice(0,4)) * 12 + Number(p.slice(5,7)) - 1;
}
function serialToPeriod(serial) {
  var year = Math.floor(serial / 12);
  var month = serial - year * 12 + 1;
  return String(year) + '-' + ('0' + month).slice(-2);
}
function cleanNumber(value) {
  if (value === null || value === '' || typeof value === 'undefined') return null;
  var n = Number(value);
  return Number.isFinite(n) && n >= 0 ? n : null;
}
function pctText(value) {
  return Number(value).toFixed(setting.percentDecimals) + '%';
}
function observationLabel(step) {
  // 列是相对首期的观察月序：相同列不一定是相同的自然月。
  if (setting.periodLabelMode === 'code') return String(setting.periodPrefix || 'M') + step;
  return step === 0 ? setting.firstPeriodText : String(step) + setting.afterPeriodSuffix;
}
function formatCount(value) {
  return Number(value).toLocaleString('zh-CN', {maximumFractionDigits: 0});
}
function escapeHtml(textValue) {
  return String(textValue).replace(/[&<>"']/g, function(c) {
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
  });
}
function hexToRgb(hex) {
  var h = String(hex).replace('#','');
  if (h.length === 3) h = h.split('').map(function(x){return x+x;}).join('');
  return [parseInt(h.slice(0,2),16),parseInt(h.slice(2,4),16),parseInt(h.slice(4,6),16)];
}
function blendHex(from,to,position) {
  var t = Math.max(0, Math.min(1, position));
  var a=hexToRgb(from), b=hexToRgb(to);
  return '#' + a.map(function(x,i){return ('0'+Math.round(x+(b[i]-x)*t).toString(16)).slice(-2);}).join('');
}

  var period = normalizePeriod(同期群月份[i]);

  var active = cleanNumber(活跃数量[i]);

    console.warn('[C024] 同一同期群与月序出现重复记录：'+period+' '+String(setting.periodPrefix)+step+'；保留最后一条。');
  }
  rowData[period][step] = active;
  allPeriodSerials.push(monthSerial(period) + step);
}
var allCohorts = Object.keys(rowData).filter(function(x){return rowData[x][0] > 0;}).sort();

var retentionForScale = [];
for (var r=0;r<cohorts.length;r++) {
  var c = cohorts[r];
  var base = rowData[c][0];
  for (var j=0;j<monthCount;j++) {
    var observed = latestObserved !== null && monthSerial(c) + j <= latestObserved;
    var activeCount = observed ? (Object.prototype.hasOwnProperty.call(rowData[c],j) ? rowData[c][j] : 0) : null;
    // 留存按同批初始对象口径计算。活跃数量超出初始规模，提醒用户校验去重口径。
    if (observed && activeCount > base) {
      console.warn('[C024] 观察期活跃数量超过初始数量，请检查数据去重和同期群口径：'+c+' '+j);
    }
    var rate = observed ? 100 * activeCount / base : null;
    if (observed && j>0) retentionForScale.push(rate);
    cells.push({row:r,col:j,cohort:c,base:base,active:activeCount,rate:rate,observed:observed,calendarPeriod:serialToPeriod(monthSerial(c)+j)});
  }
}
var actualMax = retentionForScale.length ? Math.max.apply(null,retentionForScale) : 0;
// “好读”的动态色阶刻度（1/2/5 × 10^n）；无任何业务数据上限常量。
function niceColorStep(maxValue) {
  if (!(maxValue>0)) return 1;
  var raw = maxValue / 7;
  var base = Math.pow(10, Math.floor(Math.log10(raw)));
  var ratio = raw / base;
  return (ratio<=1?1:ratio<=2?2:ratio<=5?5:10)*base;
}
var colorStep = niceColorStep(actualMax);
var autoMax = actualMax > 0 ? Math.ceil(actualMax / colorStep - 1e-10) * colorStep : colorStep;
autoMax = Math.round(autoMax*1e9)/1e9;
var scaleMax = Number.isFinite(Number(setting.maxColorPercent)) && setting.maxColorPercent !== null && Number(setting.maxColorPercent) > 0 ? Number(setting.maxColorPercent) : autoMax;
function cellFill(cell) {
  if (!cell.observed) return setting.palette.future;
  if (cell.col===0) return setting.palette.m0;
  return blendHex(setting.palette.low,setting.palette.high,Math.pow(Math.min(1,cell.rate/scaleMax),0.75));
}
function cellLabelColor(cell) {
  if (!cell.observed) return setting.palette.textSecondary;
  if (cell.col===0) return setting.palette.text;
  return cell.rate/scaleMax >= 0.60 ? setting.palette.white : setting.palette.text;
}

  };
});

// ==================== 04 富文本 Tooltip ====================
function tableLine(label,value,emphasisColor) {
  var c=emphasisColor||setting.palette.text;
  return '<div style="display:flex;justify-content:space-between;gap:28px;margin-top:8px;white-space:nowrap">'
    + '<span style="color:'+setting.palette.textSecondary+'">'+escapeHtml(label)+'</span>'
    + '<strong style="font-weight:600;color:'+c+'">'+escapeHtml(value)+'</strong></div>';
}
function tooltipFormatter(params) {
  var d=params.data || {};
  if (!d.cohort) return '';
  var title = '<div style="font-size:13px;font-weight:700;color:'+setting.palette.text+';padding-bottom:8px;border-bottom:1px solid '+setting.palette.gridLine+'">'
    + escapeHtml(d.cohort)+' · '+escapeHtml(observationLabel(d.index))+' '+escapeHtml(setting.rateLabel)+'明细</div>';
  if (!d.observed) return '<div style="min-width:230px">'+title+'<div style="margin-top:10px;color:'+setting.palette.textSecondary+'">尚未到观察期，不等于 0% 留存。</div></div>';
  var changeLine = '';
  if (setting.showChangeInTooltip && d.index>0) {
    var prev = rowData[d.cohort][d.index-1];
    if (typeof prev==='undefined') prev=0;
    var pp=(d.active-prev)*100/d.base;
    var textValue = (pp>0?'▲ +':pp<0?'▼ ':'')+pp.toFixed(1)+' 个百分点';
    var clr=pp>0?setting.palette.positive:pp<0?setting.palette.negative:setting.palette.textSecondary;
    changeLine=tableLine('较上一观察期变化',textValue,clr);
  }
  return '<div style="min-width:244px;font-size:12px;line-height:1.5">'+title
    + tableLine(setting.periodLabel,d.cohort)
    + tableLine(setting.activityLabel,d.calendarPeriod)
    + tableLine(setting.baseLabel,formatCount(d.base)+(setting.countUnit?' '+setting.countUnit:''))
    + tableLine(setting.activeLabel,formatCount(d.active)+(setting.countUnit?' '+setting.countUnit:''))
    + tableLine(setting.rateLabel,pctText(d.rate),setting.palette.positive)
    + changeLine
    + '<div style="font-size:10px;border-top:1px solid '+setting.palette.gridLine+';margin-top:11px;padding-top:7px;color:'+setting.palette.textSecondary+'">'
    + (d.index===0?'首期为同期群初始规模，按定义为 100%。':'以初始规模为分母；各观察月份独立计算。')
    + '</div></div>';
}
function placeTooltip(point,params,dom,rect,size) {
  var x=point[0]+17,y=point[1]+15;
  if (x+size.contentSize[0]>size.viewSize[0]-8) x=point[0]-size.contentSize[0]-16;
  if (y+size.contentSize[1]>size.viewSize[1]-7) y=size.viewSize[1]-size.contentSize[1]-7;
  return [Math.max(5,x),Math.max(5,y)];
}
