var getQuantile = function (sorted, ratio) {
  var k = (sorted.length - 1) * ratio;
  var left = Math.floor(k), right = Math.ceil(k);
  return sorted[left] + (sorted[right] - sorted[left]) * (k - left);
};
var summaries = pointGroups.map(function (points, groupIndex) {

  var iqr = q3 - q1, lowLimit = q1 - 1.5 * iqr, highLimit = q3 + 1.5 * iqr;
  var within = sorted.filter(function (n) { return n >= lowLimit && n <= highLimit; });
  points.forEach(function (p) { p.outlier = p.value < lowLimit || p.value > highLimit; });
  return { group: groupNames[groupIndex], index: groupIndex, count: sorted.length,
    min: sorted[0], q1: q1, median: median, q3: q3, max: sorted[sorted.length - 1],
    low: within[0], high: within[within.length-1],
    outliers: sorted.filter(function (n) { return n < lowLimit || n > highLimit; }).length };

};

var axisPadding = span > 0 ? span * 0.055 : Math.max(Math.abs(maxValue) * 0.08, 0.08);
// 根据数值量级选取整齐的通用刻度步长（1、2、2.5、5 × 10^n），不写死业务范围。
var axisLo = minValue - axisPadding;
var axisHi = maxValue + axisPadding;
var targetTicks = Math.max(2, Number(setting.axisSplitNumber) || 8);
var rawStep = (axisHi - axisLo) / targetTicks;
var magnitude = Math.pow(10, Math.floor(Math.log(rawStep) / Math.LN10));
var multiples = [1, 2, 2.5, 5, 10];
var axisStep = magnitude;
var bestDistance = Infinity;
multiples.forEach(function (m) {
  var candidate = m * magnitude;
  var distance = Math.abs(Math.log(candidate / rawStep));
  if (distance < bestDistance) { bestDistance = distance; axisStep = candidate; }
});
var xMin = setting.xMin === null ? Math.floor(axisLo / axisStep + 1e-10) * axisStep : Number(setting.xMin);
var xMax = setting.xMax === null ? Math.ceil(axisHi / axisStep - 1e-10) * axisStep : Number(setting.xMax);
if (!isFinite(xMin) || !isFinite(xMax) || xMax <= xMin) throw new Error('数值坐标轴范围无效，请检查 xMin 和 xMax。');
var subtitleText = setting.subtitle || (String(targetPeriod) + '｜' + groupNames.length + ' 类别｜' + input.length + ' 条样本');
// 轴标题与单位均从用户设置获取；不在图形计算或 Tooltip 中写死业务信息。
var axisName = setting.metricName
  ? setting.metricName + (setting.valueUnit ? '（' + setting.valueUnit + '）' : '')
  : (setting.valueUnit || '');
var formatValueWithUnit = function (v) { return fmt(v) + (setting.valueUnit ? ' ' + setting.valueUnit : ''); };
var axisPrecision = setting.axisLabelPrecision === null ? setting.valuePrecision : setting.axisLabelPrecision;
var fmtAxis = function(v){return Number(v).toLocaleString('zh-CN',{minimumFractionDigits:0, maximumFractionDigits:Math.min(15, Math.max(0, Number(axisPrecision) || 0))});};

// ==================== 4. 富文本 Tooltip（分布 / 单个样本） ====================
var tooltipShell = function (heading, color, body, footer) {
  return '<div style="min-width:238px;font-family:Arial,Microsoft YaHei,sans-serif;color:#263238;">'
    + '<div style="font-size:13px;font-weight:700;padding:1px 0 9px;border-bottom:1px solid #E5E9ED;color:'+color+'">'
    + escapeHtml(heading) + '</div>' + body
    + '<div style="color:#8493A2;font-size:10px;margin-top:9px;padding-top:7px;border-top:1px solid #E5E9ED">'
    + escapeHtml(footer) + '</div></div>';
};
var tooltipRow = function (label, value, color) {
  return '<div style="display:flex;justify-content:space-between;gap:18px;padding:4px 0;font-size:11px">'
    + '<span style="color:#8493A2">'+escapeHtml(label)+'</span>'
    + '<b style="color:'+(color || '#263238')+'">'+escapeHtml(value)+'</b></div>';
};

// ==================== 5. 图形与同类别聚焦 ====================
// 箱体和散点作为同一个 custom 系列的数据项。聚焦同类别的完整索引集合，
// 避免跨系列淡化时把当前散点所属的箱体一起变灰。
var drawItems = [];
var focusIndexGroups = groupNames.map(function () { return []; });
var rgba = function(hex, alpha) {
  var r=parseInt(hex.slice(1,3),16),g=parseInt(hex.slice(3,5),16),b=parseInt(hex.slice(5,7),16);
  return 'rgba('+r+','+g+','+b+','+alpha+')';
};
// 每组先加入箱体，再加入本组散点：这样散点自身的 tooltip 能单独识别。
summaries.forEach(function (s, gi) {
  var bIndex=drawItems.length;
  drawItems.push({kind:'box',groupIndex:gi,
    value:[gi,s.median,gi,s.low,s.q1,s.q3,s.high]});
  focusIndexGroups[gi].push(bIndex);
  if (setting.showPoints) {
    pointGroups[gi].forEach(function (p) {
      var jitterY=gi+deterministicJitter(p.sourceIndex,gi);
      var pointIndex=drawItems.length;
      drawItems.push({kind:'point',groupIndex:gi,record:p,
        value:[gi,p.value,jitterY]});
      focusIndexGroups[gi].push(pointIndex);
    });
  }
});

var distributionSeries = {
  name:'分布及样本',type:'custom',coordinateSystem:'cartesian2d',z:7,
  // x/y 坐标及类目范围仍由原数据与统计计算；不依赖任何示例类别名称。
  data:drawItems,
  encode:{x:1,y:2},
  renderItem:function(params,api) {
    var item=drawItems[params.dataIndex];
    if (!item) return;
    var gi=item.groupIndex, color=getColor(gi);
    // ECharts 6 通过返回图元的根节点读取 focus 与 blurScope，
    // 数组索引让该类别所有箱体和散点保留，其他类别进入 blur。
    var focus=focusIndexGroups[gi];
    if (item.kind==='point') {
      var p=item.record;
      var coord=api.coord([p.value,gi+deterministicJitter(p.sourceIndex,gi)]);
      var radius=(p.outlier && setting.showOutlierRing ? setting.pointSize+3 : setting.pointSize)/2;
      return {
        type:'circle',
        focus:focus,blurScope:'series',
        shape:{cx:coord[0],cy:coord[1],r:radius},
        style:{fill:color,opacity:setting.pointOpacity,
          stroke:p.outlier && setting.showOutlierRing ? color:'none',
          lineWidth:p.outlier && setting.showOutlierRing ? 1.7:0},
        emphasis:{shape:{r:radius*1.6},style:{opacity:setting.hoverOpacity,
          stroke:'#FFFFFF',lineWidth:1.1}},
        blur:{style:{opacity:setting.blurOpacity}}
      };
    }
    var s=summaries[gi];
    var pos=function(v){return api.coord([v,gi])[0];};
    var midY=api.coord([s.median,gi])[1];
    var low=pos(s.low),q1=pos(s.q1),med=pos(s.median),q3=pos(s.q3),high=pos(s.high);
    var hh=setting.boxHalfHeight;
    var children=[
      {type:'line',shape:{x1:low,y1:midY,x2:high,y2:midY},style:{stroke:color,lineWidth:2}},
      {type:'line',shape:{x1:low,y1:midY-hh*0.58,x2:low,y2:midY+hh*0.58},
        style:{stroke:color,lineWidth:setting.boxLineWidth}},
      {type:'line',shape:{x1:high,y1:midY-hh*0.58,x2:high,y2:midY+hh*0.58},
        style:{stroke:color,lineWidth:setting.boxLineWidth}},
      {type:'rect',shape:{x:q1,y:midY-hh,width:Math.max(1,q3-q1),height:hh*2},
        style:{fill:rgba(color,setting.boxFillOpacity),stroke:color,lineWidth:setting.boxLineWidth}},
      {type:'line',shape:{x1:med,y1:midY-hh,x2:med,y2:midY+hh},
        style:{stroke:color,lineWidth:setting.medianWidth}}
    ];
    if (setting.showMedianLabels) children.push({
      type:'text',style:{x:med,y:midY-hh-12,text:fmt(s.median),fill:color,
        align:'center',verticalAlign:'middle',font:'600 11px Microsoft YaHei, sans-serif'}
    });
    return {type:'group',focus:focus,blurScope:'series',children:children,
      emphasis:{style:{opacity:1}},blur:{style:{opacity:setting.blurOpacity}}};
  },
  tooltip:{show:true}
};

// Tooltip 优先出现在鼠标右侧；空间不够时自动翻转至左侧，并限制上下边界。
// 使用 ECharts 提供的容器和内容尺寸，不读取 DOM，不依赖固定图表宽高。
var tooltipPosition=function(pos,params,dom,rect,size) {
  var w=size.viewSize[0],h=size.viewSize[1];
  var tw=size.contentSize[0],th=size.contentSize[1];
  var gap=Math.max(0,Number(setting.tooltipGap)||0);
  var edge=Math.max(0,Number(setting.tooltipEdge)||0);
  var rightX=pos[0]+gap, leftX=pos[0]-tw-gap;
  var x=rightX+tw+edge<=w?rightX:(leftX>=edge?leftX:Math.max(edge,w-tw-edge));
  var topLimit=setting.showTitle?104:edge;
  var bottomLimit=h-(setting.chartMargin.bottom||0)+22;
  var maxY=Math.max(topLimit,Math.min(h-th-edge,bottomLimit-th));
  var y=Math.min(maxY,Math.max(topLimit,pos[1]-th/2));
  return [x,y];
};
