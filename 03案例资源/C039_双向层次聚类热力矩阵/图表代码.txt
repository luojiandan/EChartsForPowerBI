// ========== 3. 数据校验与算法：不使用外部聚类库 ==========
function c039Number(v) {
  if (typeof v === 'number') return Number.isFinite(v) ? v : NaN;
  if (typeof v === 'string' && v.trim() !== '') {
    var s = v.replace(/,/g,'').trim();
    return /^[-+]?(?:\d+\.?\d*|\.\d+)(?:[eE][-+]?\d+)?$/.test(s) ? Number(s) : NaN;
  }
  return NaN;
}
function c039Fmt(n, digits) {
  if (!Number.isFinite(n)) return '—';
  return n.toLocaleString('zh-CN',{maximumFractionDigits:digits===undefined?2:digits});
}
function c039Escape(s) {
  return String(s).replace(/[&<>"']/g,function(ch){return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[ch];});
}
function c039Prepare(names, metrics, values) {
  if (!Array.isArray(names) || !Array.isArray(metrics) || !Array.isArray(values)) throw Error('输入字段必须为三个平行的一维数组。');
  if (names.length !== metrics.length || names.length !== values.length) throw Error('对象、指标、数值三列长度不一致，请检查 Power BI 字段绑定与聚合粒度。');
  if (names.length === 0) throw Error('当前没有有效数据，请提供“对象、指标、数值”三列。');
  var rowMap=new Map(), colMap=new Map(), pairs=new Map();
  for (var k=0; k<names.length; k++) {
    var n = names[k] === null || names[k] === undefined ? '' : String(names[k]).trim();
    var m = metrics[k] === null || metrics[k] === undefined ? '' : String(metrics[k]).trim();
    if (!n || !m) throw Error('第 '+(k+1)+' 行对象或指标为空。');
    if (/\u0000/.test(n+m)) throw Error('对象/指标名称不能包含特殊分隔符。');
    var v=c039Number(values[k]);
    if (!Number.isFinite(v)) throw Error('第 '+(k+1)+' 行数值不是有效数字：'+ n+' / '+m+'。');
    var key=JSON.stringify([n,m]);
    if (pairs.has(key)) throw Error('出现重复的对象×指标：'+n+' / '+m+'。请在 Power BI 中先按两列汇总。');
    pairs.set(key,v);
    if (!rowMap.has(n)) rowMap.set(n,rowMap.size);
    if (!colMap.has(m)) colMap.set(m,colMap.size);
  }
  var rows=Array.from(rowMap.keys()).sort(function(a,b){return a.localeCompare(b,'zh-CN');});
  var cols=Array.from(colMap.keys()).sort(function(a,b){return a.localeCompare(b,'zh-CN');});
  if(rows.length<3||cols.length<3) throw Error('双向聚类至少需要 3 个对象和 3 项指标。');
  if(rows.length>setting.maxRows||cols.length>setting.maxColumns) throw Error('当前为 '+rows.length+'×'+cols.length+'，超过模板安全上限 '+setting.maxRows+'×'+setting.maxColumns+'；请先筛选对象/指标。');
  var matrix=[], imputed=[];
  for(var r=0;r<rows.length;r++) {
    var record=[], rowFlags=[];
    for(var c=0;c<cols.length;c++) {
      var key2=JSON.stringify([rows[r],cols[c]]);
      record.push(pairs.has(key2)?pairs.get(key2):null);
      rowFlags.push(false);
    }
    matrix.push(record);imputed.push(rowFlags);
  }
  var missing=[];
  for(var ri=0;ri<rows.length;ri++)for(var ci=0;ci<cols.length;ci++)if(matrix[ri][ci]===null)missing.push(rows[ri]+' / '+cols[ci]);
  if(missing.length) {
    if(setting.missingCellMode!=='columnMean') throw Error('发现 '+missing.length+' 个缺失单元格（例如 '+missing.slice(0,2).join('、')+'），默认不插补以免影响聚类；可补齐数据或设置 missingCellMode=columnMean。');
    for(var j=0;j<cols.length;j++){
      var observed=matrix.map(function(row){return row[j];}).filter(function(v){return v!==null;});
      if(!observed.length)throw Error('指标“'+cols[j]+'”整列缺失，无法插补。');
      var mean=observed.reduce(function(a,b){return a+b;},0)/observed.length;
      for(var i=0;i<rows.length;i++)if(matrix[i][j]===null){matrix[i][j]=mean;imputed[i][j]=true;}
    }
  }
  var normal=rows.map(function(){return new Array(cols.length);}), means=[], sds=[], constant=[];
  for(var col=0;col<cols.length;col++){
    var x=matrix.map(function(a){return a[col];});
    var mu=x.reduce(function(a,b){return a+b;},0)/x.length;
    var variance=x.reduce(function(s,a){return s+Math.pow(a-mu,2);},0)/x.length;
    var sd=Math.sqrt(variance);
    means.push(mu);sds.push(sd);constant.push(sd<1e-12);
    for(var row=0;row<rows.length;row++)normal[row][col]=sd<1e-12?0:(matrix[row][col]-mu)/sd;
  }
  return {rows:rows,cols:cols,matrix:matrix,normal:normal,means:means,sds:sds,constant:constant,imputed:imputed,missingCount:missing.length};
}
function c039Euclidean(a,b) {
  var s=0;
  for(var i=0;i<a.length;i++) s+=Math.pow(a[i]-b[i],2);
  return Math.sqrt(s/a.length);
}
function c039Cluster(vectors, linkage, minBranchGapPx, treePixelWidth) {
  var n=vectors.length, nodes=[], active=[], distances=new Map();
  function pair(a,b){return a<b?a+'|'+b:b+'|'+a;}
  for(var i=0;i<n;i++){nodes.push({id:i,leaf:i,size:1,height:0});active.push(i);}
  for(var i=0;i<n;i++)for(var j=i+1;j<n;j++)distances.set(pair(i,j),c039Euclidean(vectors[i],vectors[j]));
  var nextId=n;
  while(active.length>1){
    var best=Infinity,aId=-1,bId=-1;
    for(var a=0;a<active.length;a++)for(var b=a+1;b<active.length;b++){
      var ai=active[a],bi=active[b],d=distances.get(pair(ai,bi));
      if(d<best-1e-12 || (Math.abs(d-best)<1e-12&&(ai<aId||aId<0||ai===aId&&bi<bId))){best=d;aId=ai;bId=bi;}
    }
    var left=nodes[aId],right=nodes[bId];
    var parent={id:nextId,left:left,right:right,size:left.size+right.size,height:best};
    nodes[nextId]=parent;
    for(var z=0;z<active.length;z++){
      var other=active[z];if(other===aId||other===bId)continue;
      var da=distances.get(pair(aId,other)), db=distances.get(pair(bId,other));
      var dist=linkage==='single'?Math.min(da,db):linkage==='complete'?Math.max(da,db):(left.size*da+right.size*db)/(left.size+right.size);
      distances.set(pair(nextId,other),dist);
    }
    active=active.filter(function(q){return q!==aId&&q!==bId;});active.push(nextId);nextId++;
  }
  var root=nodes[active[0]],order=[];
  function visit(node){if(node.leaf!==undefined){order.push(node.leaf);return;}visit(node.left);visit(node.right);}
  visit(root);
  var leafPos=new Map();order.forEach(function(idx,i){leafPos.set(idx,i);});
  function locate(node){
    if(node.leaf!==undefined){node.pos=leafPos.get(node.leaf);return node.pos;}
    node.pos=(locate(node.left)+locate(node.right))/2;
    return node.pos;
  }
  locate(root);
  // V0.8.0：聚类的真实 height 保持不变；另外生成显示坐标 displayHeight。
  // 原始距离非常接近时，固定像素宽度会把两个内部节点压缩到 1px 以内。
  // 使用树形拓扑约束，给每一代父子合并节点预留可辨识的最小横向间距。
  // 这里的横向展示非严格距离等比例缩放：距离的真实值仍保存在 height / branchMeta 中。
  var gapPx=Number(minBranchGapPx);
  var spanPx=Number(treePixelWidth);
  var displayStep=(Number.isFinite(gapPx)&&gapPx>0&&Number.isFinite(spanPx)&&spanPx>0)
    ? gapPx/Math.max(1,spanPx)*Math.max(root.height,0.05)*1.06 : 0;
  function layoutHeight(node){
    if(node.leaf!==undefined){node.displayHeight=0;return 0;}
    var lh=layoutHeight(node.left),rh=layoutHeight(node.right);
    node.displayHeight=Math.max(node.height,lh+displayStep,rh+displayStep);
    return node.displayHeight;
  }
  layoutHeight(root);
  function cut(k){
    var pending=[root];
    while(pending.length<Math.min(n,k)){
      var split=-1;
      for(var p=0;p<pending.length;p++)if(pending[p].left && (split<0||pending[p].height>pending[split].height))split=p;
      if(split<0)break;
      var part=pending.splice(split,1)[0];pending.push(part.left,part.right);
    }
    function smallest(node){return node.leaf!==undefined?node.pos:Math.min(smallest(node.left),smallest(node.right));}
    pending.sort(function(a,b){return smallest(a)-smallest(b);});
    var groups=new Array(n);
    function color(node,id){if(node.leaf!==undefined){groups[node.leaf]=id;return;}color(node.left,id);color(node.right,id);}
    pending.forEach(function(node,i){color(node,i);});return groups;
  }
  var branches=[],branchMeta=[];
  function leavesOf(node){
    if(node.leaf!==undefined)return [node.leaf];
    return leavesOf(node.left).concat(leavesOf(node.right));
  }
  function collect(node){
    if(!node.left)return;
    branches.push([node.left.pos,node.left.displayHeight,node.right.pos,node.right.displayHeight,node.displayHeight]);
    branchMeta.push({id:node.id,leaves:leavesOf(node),height:node.height,size:node.size});
    collect(node.left);collect(node.right);
  }
  collect(root);
  return {order:order,branches:branches,branchMeta:branchMeta,groups:cut(1),getGroups:cut,
    maxHeight:Math.max(root.displayHeight,0.05),rawMaxHeight:Math.max(root.height,0.05)};
}

function c039FindChart(element, event) {
  // 优先从点击元素所在 ZRender 的原生画布向上查找 ECharts 容器。
  // 只访问 ECharts 已创建的容器，不创建 HTML 元素，也不绑定额外全局监听。
  var zr = element && element.__zr;
  if(!zr && event && event.target) zr = event.target.__zr;
  if(!zr && event && event.topTarget) zr = event.topTarget.__zr;
  if(typeof echarts!=='undefined' && echarts && typeof echarts.getInstanceByDom==='function' && zr && zr.painter) {
    var root=zr.painter.getViewportRoot && zr.painter.getViewportRoot();
    var dom=root,depth=0;
    while(dom && depth<9){
      try{
        var found=echarts.getInstanceByDom(dom);
        if(found && typeof found.getZr==='function' && found.getZr()===zr) return found;
      }catch(err){}
      dom=dom.parentNode || dom.host;
      depth++;
    }
  }
  // 与某些官方编辑器、宿主的全局实例命名兼容；不依赖其存在。
  if(typeof myChart!=='undefined' && myChart && typeof myChart.setOption==='function') return myChart;
  if(typeof chart!=='undefined' && chart && typeof chart.setOption==='function') return chart;
  if(typeof echartsInstance!=='undefined' && echartsInstance && typeof echartsInstance.setOption==='function') return echartsInstance;
  return null;
}

function c039RefreshFocus(chart, changedAxis){
  if(!chart || typeof chart.setOption!=='function') return false;
  // V0.9.0：row 和 col 是互不覆盖的独立选择状态。
  // 仅更新本次改变的树系列，绝不因对象树操作重建指标树（反之亦然）。
  var selectedRows=c039FocusState.row?new Set(c039FocusState.row):null;
  var selectedCols=c039FocusState.col?new Set(c039FocusState.col):null;
  var n=Number(setting.focusOpacity);
  var faded=Number.isFinite(n)?Math.max(0.08,Math.min(0.85,n)):0.35;
  var nextHeat=c039FocusState.heat.map(function(item){
    var visible=(!selectedRows||selectedRows.has(item.rowName)) && (!selectedCols||selectedCols.has(item.colName));
    return {
      value:item.value.slice(),raw:item.raw,rowName:item.rowName,colName:item.colName,
      rowGroup:item.rowGroup,colGroup:item.colGroup,imputed:item.imputed,constant:item.constant,
      itemStyle:{opacity:visible?1:faded,borderColor:'#FFFFFF',borderWidth:1},
      label:{show:c039ShouldShowCellLabel(visible)}
    };
  });
  var groups=Array.from({length:c039FocusState.rowGroupCount},function(){return [];});
  nextHeat.forEach(function(cell){groups[cell.rowGroup-1].push(cell);});
  var series=groups.map(function(data,i){return {id:'c039-heatmap-group-'+(i+1),data:data};});
  // 禁用 custom series 自带 selectedMode：聚焦完全由 rowNode/colNode 状态控制。
  // 局部更新避免两种点击顺序触发另一棵树无关重绘，且分支 ID 始终稳定。
  if(!changedAxis || changedAxis==='row')
    series.push({id:'c039-row-tree',data:c039FocusState.rowTreeData});
  if(!changedAxis || changedAxis==='col')
    series.push({id:'c039-col-tree',data:c039FocusState.colTreeData});
  series.push({id:'c039-cross-outline',data:c039CrossData()});
  var xLabel=c039FocusAxisLabels('col');
  var yLabel=c039FocusAxisLabels('row');
  var patch={
    series:series,
    graphic:[{id:'c039-focus-caption',style:{text:c039FocusCaption()}}]
  };
  // 轴标签也只更新被点击方向，避免另一方向的格式器和选中样式被覆盖。
  if(!changedAxis||changedAxis==='col')patch.xAxis=[{axisLabel:{formatter:xLabel.formatter,rich:xLabel.rich}}];
  if(!changedAxis||changedAxis==='row')patch.yAxis=[{axisLabel:{formatter:yLabel.formatter,rich:yLabel.rich}}];
  chart.setOption(patch, {lazyUpdate:false});
  return true;
}

  if(!chart){
    if(typeof console!=='undefined' && console.warn)console.warn('C039: 无法定位 ECharts 实例；请检查宿主是否允许 graphic 交互。');

    // 使用相邻整数类目中心的像素中点，而不是小数类目索引。
    // 从 params.coordSys 获取精确矩阵边框，确保线条不会延伸到标签/树状图中。
    var type=api.value(0), afterIndex=Math.round(api.value(1));
    var box=params.coordSys, start=box.x, top=box.y;
    var endX=start+box.width, endY=top+box.height;
    var first, second, middle;
    if(type===0){
      // 横向行聚类边界：在 afterIndex-1 与 afterIndex 两行之间。
      if(afterIndex<=0 || afterIndex>=rowOrder.length)return;
      first=api.coord([0,afterIndex-1])[1];
      second=api.coord([0,afterIndex])[1];
      middle=(first+second)/2;
      return {type:'line',shape:{x1:start,y1:middle,x2:endX,y2:middle},
        style:{stroke:setting.clusterBoundaryColor,opacity:setting.clusterBoundaryOpacity,lineWidth:setting.clusterBoundaryWidth}};
    }
    // 纵向列聚类边界：在 afterIndex-1 与 afterIndex 两列之间。
    if(afterIndex<=0 || afterIndex>=colOrder.length)return;
    first=api.coord([afterIndex-1,0])[0];
    second=api.coord([afterIndex,0])[0];
    middle=(first+second)/2;
    return {type:'line',shape:{x1:middle,y1:top,x2:middle,y2:endY},
      style:{stroke:setting.clusterBoundaryColor,opacity:setting.clusterBoundaryOpacity,lineWidth:setting.clusterBoundaryWidth}};
  }
  var footer='图表说明：按指标做 Z-score 标准化；双向平均连接层次聚类；暖色=高于该指标均值，冷色=低于均值（不代表好坏）。';
  if(setting.linkage!=='average')footer='图表说明：按指标做 Z-score 标准化；'+setting.linkage+' 连接层次聚类；颜色表示相对指标均值的高低，并非业务好坏。';
  if(d.missingCount)footer+=' 缺失值采用列均值插补 '+d.missingCount+' 项。';
  var subtitle=setting.subTitlePrefix+'  |  '+rowOrder.length+' 个对象 × '+colOrder.length+' 项指标  |  行 '+new Set(rowGroup).size+' 组 / 列 '+new Set(colGroup).size+' 组';
  return {
    backgroundColor:'transparent',

    graphic:[
      {type:'rect',left:43,top:17,shape:{width:19,height:4},style:{fill:setting.colors.negative},silent:true},
      {type:'text',left:lay.mainLeft,top:129,style:{text:'指标聚类树 · '+(setting.distance==='euclidean'?'欧氏距离':setting.distance)+' / '+(setting.linkage==='average'?'平均连接':setting.linkage)+' · 行 '+new Set(rowGroup).size+' 组 / 列 '+new Set(colGroup).size+' 组（组数可配置）',fill:setting.colors.secondary,font:'11px sans-serif'},silent:true},
      {id:'c039-focus-caption',type:'text',right:lay.mainRight,top:109,
        style:{text:c039FocusCaption(),font:'11px sans-serif',fill:setting.colors.secondary},silent:true},
      // V6 修复：group 本身没有可命中的填充区域，必须将事件放到真实的 rect/text 图形上。
      {id:'c039-reset-focus',type:'group',right:46,top:33,z:50,silent:false,
        children:[
          {type:'rect',shape:{x:0,y:0,width:94,height:27,r:5},silent:false,cursor:'pointer',
            onclick:function(evt){c039ResetFocus(this,evt);},
            style:{fill:'#F4F7F9',stroke:'#D9E2E8',lineWidth:1}},
          {type:'text',silent:false,cursor:'pointer',
            onclick:function(evt){c039ResetFocus(this,evt);},

// ========== 6. 可选聚类树事件桥接：仅当宿主确实暴露 chart 实例时启用 ==========
// 注意：ECharts 官方 custom series 文档要求通过 chart.on('click') 注册子节点事件。
// Power BI 的仅 option 沙箱可能不暴露实例，此时应使用顶部原生图例筛选行组。
function c039BindTreeEvents(chartInstance){
  if(!chartInstance || typeof chartInstance.on!=='function')return false;
  // 不再仅依赖 V4 的 __c039Bound 标志：宿主可能复用同一个 ECharts 实例。
  // V5 后续重复运行时只移除自身注册的监听，不影响 Power BI 或其他组件的监听。
  var previous=chartInstance.__c039V5Handlers;
  if(previous && typeof chartInstance.off==='function'){
    chartInstance.off('click',previous.row);
    chartInstance.off('click',previous.col);
  }
  c039FocusState.chart=chartInstance;
  function onRow(p){
    var id=p && p.data && p.data.branchId;
    var meta=id!==undefined?c039FocusState.rowBranchById[String(id)]:c039FocusState.rowBranchMeta[p.dataIndex];
    if(meta)c039ToggleBranch('row',meta,null,null);
  }
  function onCol(p){
    var id=p && p.data && p.data.branchId;
    var meta=id!==undefined?c039FocusState.colBranchById[String(id)]:c039FocusState.colBranchMeta[p.dataIndex];
    if(meta)c039ToggleBranch('col',meta,null,null);
  }
  chartInstance.on('click',{seriesId:'c039-row-tree'},onRow);
  chartInstance.on('click',{seriesId:'c039-col-tree'},onCol);
  chartInstance.__c039V5Handlers={row:onRow,col:onCol};
  chartInstance.__c039Bound=true;
  return true;
}
