/*
 * C028｜柱顶环形份额组合图（Column Chart with Mini Donuts）
 * ECharts 官方编辑器 / ECharts for Power BI 通用模板
 * 示例：全球智能手机市场，2026 年第二季度
 * 来源：IDC Worldwide Quarterly Mobile Phone Tracker，2026 Q2 Final Historical Results
 *
 * 核心输入仅有【类别】和【数值】两个等长的一维数组；总量、占比、坐标位置自动计算。
 * 环形图不是固定 left/top 的 graphic；由 custom series 的 api.coord() 锚定柱顶。
 * 如仅传入总量中的部分类别，请在 setting.totalValue 指定完整总体量。
 */

// 一、原始业务输入：一字段、一数组。替换成 Power BI 字段时只修改这两行。
// 示例：var 类别 = dataArr.dataObjects[0].values;
//       var 数值 = dataArr.dataObjects[1].values;
var 类别 = ['三星', '苹果', '小米', 'OPPO', 'vivo','其他'];
var 数值 = [62.7, 55.7, 31.2, 28.9, 21.2,76.6];

// 二、样式与全部用户可修改的设置
const COLORS = {
  positive: '#239263',
  negative: '#D9544D',
  bar: '#BFC6CE',
  barSecondary: '#D9DEE3',
  text: '#263238',
  textSecondary: '#8493A2',
  gridLine: '#E5E9ED',
  ringTrack: '#E5E9ED',
  ringProgress: '#263238',
  ringMuted: '#AEB7C1',
  white: '#FFFFFF'
};

var setting = {
  title: '全球手机市场：品牌出货量与市场份额',
  period: '2026 第二季度',
  subtitle: '',           // 留空则自动生成：周期｜规模名称 + 占比名称
  categoryName: '品牌',
  valueName: '出货量',
  shareName: '市场份额',
  unit: '百万部',
  valueDecimals: 1,
  shareDecimals: 1,
  compactLargeNumbers: true, // 大数标签自动使用“万/亿”，Tooltip 仍显示精确原值
  totalLabel: '总量',
  totalValue: null,        // null=全部类别的数值之和；若只展示部分类别，此处传入完整总量
  mergeSameCategory: true,// 将重复类别合并求和（Power BI 仍建议在数据模型中先聚合）
  mutedCategories: [],      // 通用模板默认不区分类别；若需弱化特定类别，可填写 ['其他']
  sourceText: '数据来源：IDC Worldwide Quarterly Mobile Phone Tracker，2026-Q2 最终历史结果',
  noteText: '',             // 留空时自动生成与当前数据分母一致的说明；也可自行填写

  grid: { left: 78, right: 55, top: 146, bottom: 105 }, // 四边均为像素
  yAxisMax: null,       // null=自动估算；固定上限需为最大数值保留至少约 20% 的顶部空间
  ySplitNumber: 5,
  autoAxisHeadroom: 1.25,
  barWidth: 34,        // 最大柱宽；类目变多/容器变窄时自动收窄
  ringOuterRadius: 20,
  ringInnerRadius: 15,
  ringOffsetFromBar: 51,
  valueOffsetFromBar: 14,
  ringTextFontSize: 10.5,
  valueTextFontSize: 12,
  maxVisibleCategories: 9,     // 类别多时启用图内横向缩放，避免柱和标签拥挤
  enableCategoryZoom: true,
  categoryLabelMaxWidth: 104,
  titleFontSize: 20,
  fontFamily: 'sans-serif'
};

// 三、数据校验、聚合与派生计算（以下全部为通用逻辑，不依赖类别内容）
function c028ToNumber(v) {
  if (v == null || String(v).trim() === '') return NaN;
  var n = Number(v);
  return Number.isFinite(n) ? n : NaN;
}
function c028Round(v, digits) {
  var n = Number(v);
  var rounded = n.toFixed(digits);
  // 数值非零却被设定的小数位舍入为“0.0”时，改用有效数字，避免误导。
  return n !== 0 && Number(rounded) === 0 ? n.toPrecision(2) : rounded;
}
function c028ShowValue(v) {
  var n = Number(v), a = Math.abs(n);
  if (setting.compactLargeNumbers && a >= 1e8) return c028Round(n / 1e8, 1) + '亿';
  if (setting.compactLargeNumbers && a >= 1e4) return c028Round(n / 1e4, 1) + '万';
  return c028Round(n, setting.valueDecimals);
}
function c028ShowShare(share) {
  var rounded = c028Round(share, setting.shareDecimals);
  if (share > 0 && share < Math.pow(10, -setting.shareDecimals) / 2) {
    return '<' + Math.pow(10, -setting.shareDecimals).toFixed(setting.shareDecimals) + '%';
  }
  return rounded + '%';
}
function c028EscapeHtml(s) {
  return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;')
    .replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}
function c028NiceMaximum(maximum, splitNumber) {
  var rawStep = maximum / Math.max(1, splitNumber);
  var exponent = Math.pow(10, Math.floor(Math.log(rawStep) / Math.LN10));
  var ratio = rawStep / exponent;
  var scale = ratio <= 1 ? 1 : ratio <= 2 ? 2 : ratio <= 2.5 ? 2.5 : ratio <= 5 ? 5 : 10;
  return scale * exponent * splitNumber;
}
function c028Prepare(categoryArray, valueArray) {
  if (!Array.isArray(categoryArray) || !Array.isArray(valueArray) ||
      categoryArray.length !== valueArray.length) {
    throw new Error('C028：类别与数值必须是长度相等的两个一维数组。');
  }
  // Power BI 筛选后可能返回零行：正常显示“暂无数据”，而不是让图表报错。
  if (categoryArray.length === 0) {
    return { names: [], values: [], total: 0, denominator: 0, shares: [], empty: true };
  }
  var names = [];
  var values = [];
  var lookup = Object.create(null);
  for (var i = 0; i < categoryArray.length; i++) {
    var name = String(categoryArray[i] == null ? '' : categoryArray[i]).trim();
    var value = c028ToNumber(valueArray[i]);
    if (!name || !Number.isFinite(value) || value < 0) {
      throw new Error('C028：第 ' + (i + 1) + ' 行的类别为空，或数值不是非负数。');
    }
    var key = '$' + name;
    if (Object.prototype.hasOwnProperty.call(lookup, key)) {
      if (!setting.mergeSameCategory) {
        throw new Error('C028：存在重复类别“' + name + '”，请先汇总或开启 mergeSameCategory。');
      }
      values[lookup[key]] += value;
    } else {
      lookup[key] = names.length;
      names.push(name);
      values.push(value);
    }
  }
  var selectedTotal = values.reduce(function (acc, v) { return acc + v; }, 0);
  if (!Number.isFinite(selectedTotal)) {
    throw new Error('C028：全部类别数值之和必须是有限数值。');
  }
  if (selectedTotal === 0 && setting.totalValue == null) {
    return { names: names, values: values, total: 0, denominator: 0,
      shares: values.map(function () { return 0; }), empty: true };
  }
  var denominator = setting.totalValue == null ? selectedTotal : c028ToNumber(setting.totalValue);
  if (!Number.isFinite(denominator) || denominator <= 0 ||
      denominator < selectedTotal - Math.max(1e-8, selectedTotal * 1e-9)) {
    throw new Error('C028：totalValue 必须不小于已展示类别的数值之和。');
  }
  return {
    names: names,
    values: values,
    total: selectedTotal,
    denominator: denominator,
    shares: values.map(function (v) { return v / denominator * 100; })
  };
}

// 参数合法性校验：避免更换业务数据或配置时出现不可解释的标签/刻度错误。
if (!Number.isInteger(setting.valueDecimals) || setting.valueDecimals < 0 || setting.valueDecimals > 8 ||
    !Number.isInteger(setting.shareDecimals) || setting.shareDecimals < 0 || setting.shareDecimals > 3) {
  throw new Error('C028：valueDecimals 应为 0–8，shareDecimals 应为 0–3 的整数。');
}
if (!Number.isInteger(setting.ySplitNumber) || setting.ySplitNumber < 1 ||
    !Number.isInteger(setting.maxVisibleCategories) || setting.maxVisibleCategories < 1 ||
    !Number.isFinite(setting.autoAxisHeadroom) || setting.autoAxisHeadroom < 1.2 ||
    !Number.isFinite(setting.barWidth) || setting.barWidth <= 0 ||
    !Number.isFinite(setting.ringOuterRadius) || setting.ringOuterRadius <= 5 ||
    !Number.isFinite(setting.ringInnerRadius) || setting.ringInnerRadius <= 0 ||
    setting.ringInnerRadius >= setting.ringOuterRadius ||
    !Array.isArray(setting.mutedCategories)) {
  throw new Error('C028：刻度数量、类别窗口、柱宽、环形半径或淡化类别配置不合法。');
}
var c028Data = c028Prepare(类别, 数值);
var c028CategoryCount = c028Data.names.length;
var c028MaxValue = c028Data.empty ? 1 : Math.max.apply(null, c028Data.values);
var c028YMax = setting.yAxisMax == null
  ? c028NiceMaximum(c028MaxValue * setting.autoAxisHeadroom, setting.ySplitNumber)
  : c028ToNumber(setting.yAxisMax);
if (!Number.isFinite(c028YMax) || !(c028YMax >= c028MaxValue * 1.2)) {
  throw new Error('C028：yAxisMax 必须比最大柱体数值至少高 20%，为柱顶环形图预留空间。');
}
var c028MutedSet = new Set(setting.mutedCategories.map(function (v) { return String(v); }));
var c028NoteText = setting.noteText || (
  '图表说明：柱体表示' + setting.valueName + '；柱顶环形图表示' + setting.shareName +
  '（该类别数值 ÷ ' + (setting.totalValue == null ? '当前输入总量' : '指定完整总体量') + '）。'
);
var c028SubTitle = setting.subtitle || [
  setting.period,
  setting.valueName + '看规模，' + setting.shareName + '看构成'
].filter(Boolean).join('｜');

function c028Tooltip(params) {
  var i = params.dataIndex;
  if (!(i >= 0 && i < c028CategoryCount)) return '';
  return '<div style="min-width:172px;padding:3px 4px;line-height:1.85">' +
    '<div style="font-size:14px;font-weight:700;color:' + COLORS.text +
    ';border-bottom:1px solid ' + COLORS.gridLine +
    ';padding-bottom:5px;margin-bottom:6px;">' + c028EscapeHtml(c028Data.names[i]) + '</div>' +
    '<div><span style="color:' + COLORS.textSecondary + ';">' + c028EscapeHtml(setting.valueName) +
    '　</span><b>' + c028Round(c028Data.values[i], setting.valueDecimals) +
    ' ' + c028EscapeHtml(setting.unit) + '</b></div>' +
    '<div><span style="color:' + COLORS.textSecondary + ';">' + c028EscapeHtml(setting.shareName) +
    '　</span><b>' + c028ShowShare(c028Data.shares[i]) + '</b></div>' +
    '</div>';
}

// 四、组合图：柱体 + 自定义环形标签。使用 api.coord() 跟随数值和尺寸重新定位。
option = {
  backgroundColor: 'transparent',
  animation: true,
  animationDuration: 450,
  grid: {
    left: setting.grid.left, right: setting.grid.right,
    top: setting.grid.top, bottom: setting.grid.bottom
  },
  tooltip: {
    trigger: 'item', confine: true,
    backgroundColor: 'rgba(255,255,255,0.98)',
    borderColor: '#D9DEE3', borderWidth: 1,
    textStyle: { color: COLORS.text, fontSize: 12 },
    extraCssText: 'border-radius:7px;box-shadow:0 6px 18px rgba(38,50,56,0.13);',
    formatter: c028Tooltip
  },
  xAxis: {
    type: 'category', data: c028Data.names,
    axisLine: { show: false }, axisTick: { show: false },
    axisLabel: {
      color: COLORS.text, fontSize: 12, margin: 13, interval: 0,
      width: setting.categoryLabelMaxWidth, overflow: 'truncate'
    }
  },
  yAxis: {
    type: 'value', min: 0, max: c028YMax, splitNumber: setting.ySplitNumber,
    axisLine: { show: false }, axisTick: { show: false },
    axisLabel: { color: COLORS.textSecondary, fontSize: 10, formatter: function (v) { return c028ShowValue(v); } },
    splitLine: { show: true, lineStyle: { color: COLORS.gridLine, width: 1 } }
  },
  dataZoom: setting.enableCategoryZoom && c028CategoryCount > setting.maxVisibleCategories ? [
    {
      type: 'inside', xAxisIndex: 0, filterMode: 'none',
      startValue: c028Data.names[0],
      endValue: c028Data.names[setting.maxVisibleCategories - 1],
      zoomOnMouseWheel: true, moveOnMouseWheel: false, moveOnMouseMove: true
    }
  ] : [],
  series: [
    {
      name: setting.valueName, type: 'bar', z: 2,
      barMaxWidth: setting.barWidth, // 非固定柱宽：窄容器/大量类别自动收缩
      barWidth: '48%',
      data: c028Data.values.map(function (v, i) {
        return {
          value: v,
          itemStyle: { color: c028MutedSet.has(c028Data.names[i]) ? COLORS.barSecondary : COLORS.bar }
        };
      }),
      emphasis: { itemStyle: { opacity: 1 } }, // 悬停仅显示提示，不让灰色柱体突然变浅
      label: { show: false }
    },
    {
      name: setting.shareName, type: 'custom',
      coordinateSystem: 'cartesian2d', xAxisIndex: 0, yAxisIndex: 0,
      encode: { x: 0, y: 1 }, z: 5, silent: false,
      data: c028Data.values.map(function (v, i) {
        return [i, v, c028Data.shares[i]];
      }),
      renderItem: function (params, api) {
        var i = params.dataIndex;
        var value = Number(api.value(1));
        var share = Number(api.value(2));
        var pt = api.coord([api.value(0), value]);
        var x = pt[0], barTop = pt[1];
        // 按数据类目间距自动缩小环形图；不会为每个类别写死坐标。
        var categoryWidth = params.coordSys.width / Math.max(1, c028CategoryCount);
        // dataZoom 只显示一部分类别时，使用两侧数据坐标实际间距。
        if (c028CategoryCount > 1) {
          var next = api.coord([api.value(0) + 1, value]);
          if (next && Number.isFinite(next[0])) categoryWidth = Math.abs(next[0] - x);
        }
        var plotTop = params.coordSys.y;
        var freeHeight = barTop - plotTop;
        var numberFontSize = Math.min(setting.valueTextFontSize, Math.max(9, categoryWidth * 0.27));
        var numberY = barTop - setting.valueOffsetFromBar;
        // 为环形图、百分比、柱顶数值同时留空间；缩窄容器时优先缩小环，而不是压盖。
        var numberClearance = setting.valueOffsetFromBar + numberFontSize / 2 + 7;
        var outer = Math.min(
          setting.ringOuterRadius,
          (categoryWidth - 10) / 2,
          (freeHeight - numberClearance - 5) / 2
        );
        var drawRing = outer >= 11 && categoryWidth >= 48;
        var shareText = c028ShowShare(share);
        var children = [];
        if (drawRing) {
          var inner = Math.min(outer - 2, outer * (setting.ringInnerRadius / setting.ringOuterRadius));
          var minOffset = outer + numberClearance;
          var maxOffset = freeHeight - outer - 5;
          var ringOffset = Math.max(minOffset, Math.min(setting.ringOffsetFromBar, maxOffset));
          var cy = barTop - ringOffset;
          var arcColor = c028MutedSet.has(c028Data.names[i]) ? COLORS.ringMuted : COLORS.ringProgress;
          // 100.0% 比 7.7% 更长，字号按文本真实长度缩小，避免超出内圆。
          var shareFontSize = Math.max(6, Math.min(
            setting.ringTextFontSize, outer * 0.53,
            (2 * inner - 2) / Math.max(1, shareText.length * 0.58)
          ));
          children.push(
            { type: 'circle', shape: { cx: x, cy: cy, r: outer }, style: { fill: COLORS.ringTrack } },
            { type: 'sector', shape: {
              cx: x, cy: cy, r: outer, r0: inner,
              startAngle: -Math.PI / 2,
              endAngle: -Math.PI / 2 + Math.PI * 2 * (share / 100),
              clockwise: true
            }, style: { fill: arcColor } },
            { type: 'circle', shape: { cx: x, cy: cy, r: inner }, style: { fill: COLORS.white } },
            { type: 'text', style: {
              x: x, y: cy, text: shareText,
              textAlign: 'center', textVerticalAlign: 'middle',
              font: '600 ' + shareFontSize + 'px ' + setting.fontFamily,
              fill: COLORS.text
            } }
          );
        } else if (freeHeight >= 42 && categoryWidth >= 45) {
          // 极窄或极矮的容器，以文字份额代替小环，防止百分比与柱顶数值压盖。
          children.push({ type: 'text', style: {
            x: x, y: barTop - 37, text: shareText,
            textAlign: 'center', textVerticalAlign: 'middle',
            font: '600 9px ' + setting.fontFamily, fill: COLORS.text
          } });
        }
        children.push({ type: 'text', style: {
          x: x, y: numberY, text: c028ShowValue(value),
          textAlign: 'center', textVerticalAlign: 'middle',
          font: '700 ' + numberFontSize + 'px ' + setting.fontFamily,
          fill: COLORS.text
        } });
        return { type: 'group', children: children };
      }
    }
  ],
  graphic: [
    { type: 'rect', left: 29, top: 16, shape: { width: 20, height: 4 }, style: { fill: COLORS.negative } },
    { type: 'text', left: 29, top: 32, style: {
      text: setting.title, fill: COLORS.text, font: '700 ' + setting.titleFontSize + 'px ' + setting.fontFamily
    } },
    { type: 'text', left: 29, top: 62, style: {
      text: c028SubTitle, fill: COLORS.textSecondary, font: '12px ' + setting.fontFamily
    } },
    { type: 'text', left: 29, top: 98, style: {
      text: setting.totalLabel, fill: COLORS.textSecondary, font: '12px ' + setting.fontFamily
    } },
    { type: 'text', left: 29 + Math.max(55, String(setting.totalLabel).length * 12 + 17), top: 94, style: {
      text: c028ShowValue(c028Data.denominator) + ' ' + setting.unit,
      fill: COLORS.text, font: '700 16px ' + setting.fontFamily
    } },
    { type: 'text', left: setting.grid.left - 24, top: setting.grid.top - 17, style: {
      text: setting.unit, fill: COLORS.textSecondary, font: '10px ' + setting.fontFamily
    } },
    { type: 'text', left: 29, bottom: 38, style: {
      text: setting.sourceText, fill: COLORS.textSecondary, font: '10px ' + setting.fontFamily
    } },
    { type: 'text', left: 29, bottom: 20, style: {
      text: c028NoteText, fill: COLORS.textSecondary, font: '10px ' + setting.fontFamily
    } }
  ]
};


// 筛选至零行或所有值均为零时，保留标题/来源，显示友好的空状态。
if (c028Data.empty) {
  option.series = [];
  option.dataZoom = [];
  option.xAxis.show = false;
  option.yAxis.show = false;
  option.graphic.push({
    type: 'text', left: 'center', top: 'middle',
    style: { text: '暂无可展示的数据', fill: COLORS.textSecondary,
      font: '13px ' + setting.fontFamily }
  });
}
