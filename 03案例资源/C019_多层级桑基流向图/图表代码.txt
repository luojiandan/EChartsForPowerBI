// 四、辅助方法：数值格式化、节点与连线生成、流量汇总
function roundTo(value, decimals) {
  const factor = Math.pow(10, decimals);
  return Math.round(value * factor) / factor;
}

function formatNumber(value, decimals) {
  return echarts.format.addCommas(roundTo(value, decimals).toFixed(decimals));
}

function escapeHtml(value) {
  return String(value == null ? '' : value).replace(/[&<>"']/g, function (char) {
    return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[char];
  });
}

function buildNodes(links) {
  const names = [];
  const nameSet = new Set();
  const incoming = new Map();
  const outgoing = new Map();
  const depth = new Map();
  const inputOrder = [];
  links.forEach(function (link) {
    [link.source, link.target].forEach(function (name) {
      if (!nameSet.has(name)) {
        nameSet.add(name);
        names.push(name);
        incoming.set(name, 0);
        outgoing.set(name, []);
      }
    });
    incoming.set(link.target, incoming.get(link.target) + 1);
    outgoing.get(link.source).push(link.target);
  });

  // 从无入边的根节点出发，以最长路径自动计算深度；不写死三层结构。
  names.forEach(function (name) {
    if (incoming.get(name) === 0) {
      depth.set(name, 0);
      inputOrder.push(name);
    }
  });
  if (names.length && inputOrder.length === 0) {
    throw new Error('桑基图存在环路或不存在起点：请检查来源节点和目标节点');
  }
  let processed = 0;
  for (let head = 0; head < inputOrder.length; head++) {
    const name = inputOrder[head];
    processed++;
    outgoing.get(name).forEach(function (target) {
      depth.set(target, Math.max(depth.get(target) || 0, depth.get(name) + 1));
      incoming.set(target, incoming.get(target) - 1);
      if (incoming.get(target) === 0) inputOrder.push(target);
    });
  }
  if (processed !== names.length) {
    throw new Error('桑基图节点之间存在环路，请使用从起点到终点的有向无环流向数据');
  }

  const maxDepth = Math.max(0, ...Array.from(depth.values()));
  const colorIndexByDepth = {};
  const nodeMap = {};
  names.forEach(function (name) {
    const level = depth.get(name);
    colorIndexByDepth[level] = colorIndexByDepth[level] || 0;
    const colorIndex = colorIndexByDepth[level]++;
    const palette = level === maxDepth ? setting.destinationColors : setting.intermediateColors;
    const color = level === 0 ? setting.sourceColor :
      palette[setting.intermediaryColorMode === 'depth' && level < maxDepth ? (level - 1) % palette.length : colorIndex % palette.length];
    nodeMap[name] = { depth: level, color: color };
  });
  return {
    nodes: names.map(function (name) {
      return { name: name, depth: nodeMap[name].depth, itemStyle: { color: nodeMap[name].color } };
    }),
    nodeMap: nodeMap
  };
}

function buildLinks(sources, targets, values) {
  return sources.map(function (source, i) {
    return {
      source: source,
      target: targets[i],
      value: values[i],
      lineStyle: {
        opacity: setting.lineOpacity,
        curveness: setting.lineCurveness,
        color: 'source'
      }
    };
  });
}

function computeNodeTotals(links) {
  const inMap = {};
  const outMap = {};
  links.forEach(function (link) {
    outMap[link.source] = (outMap[link.source] || 0) + link.value;
    inMap[link.target] = (inMap[link.target] || 0) + link.value;
  });
  return { inMap: inMap, outMap: outMap };
}

// 五、预处理：检查数据、自动推导节点层级、构建连线和总量
if (sourceNodes.length !== targetNodes.length || sourceNodes.length !== flowValues.length) {
  throw new Error('三个原始数据数组的长度必须一致');
}
for (let i = 0; i < flowValues.length; i++) {
  if (!String(sourceNodes[i]).trim() || !String(targetNodes[i]).trim()) {
    throw new Error('流向第 ' + (i + 1) + ' 行存在空节点名称');
  }
  if (typeof flowValues[i] !== 'number' || !Number.isFinite(flowValues[i]) || flowValues[i] < 0) {
    throw new Error('流向第 ' + (i + 1) + ' 行的流量必须是非负有限数值');
  }
}
const links = buildLinks(sourceNodes, targetNodes, flowValues);
const graph = buildNodes(links);
const nodes = graph.nodes;
const nodeInfo = graph.nodeMap;
const nodeTotals = computeNodeTotals(links);
// 只统计所有起始节点的出流量；多层连接不重复计入总量。
const totalFlow = links.filter(function (d) { return nodeInfo[d.source].depth === 0; })
  .reduce(function (sum, item) { return sum + item.value; }, 0);
