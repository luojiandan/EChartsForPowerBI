// ===================================================================
// 03 数据与数据语义处理（不依赖指标名称、年份、指标数量）

var C = setting.colors, L = setting.layout;
function isNil(v) { return v === null || v === undefined || String(v).trim() === ''; }
function toNum(v) { return isNil(v) ? null : Number(String(v).replace(/,/g,'')); }
function closeNum(a,b) { return Math.abs(a-b) <= Math.max(1e-7,Math.abs(b)*1e-9); }
function roundZero(v) { return Math.abs(v) < 1e-9 ? 0 : v; }
function display(v){ return v / setting.valueDivisor; }
function nfmt(v,digits) {
  if (v === null || !isFinite(v)) return '—';
  return roundZero(v).toLocaleString('en-US',{minimumFractionDigits:digits, maximumFractionDigits:digits});
}
function money(v) { return nfmt(display(v),setting.decimals); }
function signedMoney(v) { return (v>1e-9?'+':'')+money(v); }
function signedPct(v) { return v===null ? '—' : (v>1e-9?'+':'')+nfmt(v,setting.percentDecimals)+'%'; }
function dir(v) { return v>1e-9?'▲':v< -1e-9?'▼':'●'; }
function effectColor(v){return v>1e-9?C.positive:v< -1e-9?C.negative:C.secondaryText;}
function html(v){return String(v).replace(/[&<>"']/g,function(s){return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[s];});}
function metricKind(v) {
  var t=String(v).trim().toLowerCase();
  if (['income','add','+','加项','增项'].indexOf(t)>=0) return 'income';
  if (['expense','subtract','-','减项','支出','费用'].indexOf(t)>=0) return 'expense';
  if (['subtotal','total','sum','=','小计','合计'].indexOf(t)>=0) return 'subtotal';
  throw new Error('未知指标类型：'+v+'（请使用 income/expense/subtotal）');
}
if (!(setting.valueDivisor>0)) throw new Error('setting.valueDivisor 必须大于 0');
if ([指标.length,度量值.length,指标类型.length].some(function(k){return k!==年份.length;}))
  throw new Error('年份、指标、度量值、指标类型四个数组必须等长，所有字段按相同索引对应。');
if (!年份.length) throw new Error('没有输入数据。');
var periodList=Array.from(new Set(年份.map(String)));
periodList.sort(function(a,b){var x=Number(a),y=Number(b);return isFinite(x)&&isFinite(y)?x-y:a.localeCompare(b,'zh-CN');});
var selectedPeriods=setting.periods&&setting.periods.length===2?setting.periods.map(String):periodList.slice(-2);
if(selectedPeriods.length!==2 || selectedPeriods[0]===selectedPeriods[1])throw new Error('必须至少有两个不同的期间。');
var py=selectedPeriods[0], ac=selectedPeriods[1];
var entries=Object.create(null),kindMap=Object.create(null),metricOrder=[];
for(var i=0;i<年份.length;i++) {
  var period=String(年份[i]),name=String(指标[i]).trim(),kind=metricKind(指标类型[i]);
  var value=toNum(度量值[i]);
  if(!name)throw new Error('第 '+(i+1)+' 条记录缺少指标名称。');
  if(value!==null&&!isFinite(value))throw new Error('非数字的度量值：'+period+'/'+name);
  if(kind!=='subtotal'&&value===null)throw new Error('收入/费用项目不能缺少金额：'+period+'/'+name);
  var key=JSON.stringify([period,name]);
  if(Object.prototype.hasOwnProperty.call(entries,key))throw new Error('重复的期间+指标：'+period+'/'+name);
  entries[key]=value;
  if(kindMap[name]&&kindMap[name]!==kind)throw new Error('同一指标不能跨期间修改类型：'+name);
  kindMap[name]=kind;
  if(period===py)metricOrder.push(name);
}
var rows=[];
for(var i=0;i<metricOrder.length;i++) {
  var name=metricOrder[i],k0=JSON.stringify([py,name]),k1=JSON.stringify([ac,name]);
  if(!Object.prototype.hasOwnProperty.call(entries,k1))throw new Error('两期必须拥有相同指标；当期缺少：'+name);
  rows.push({name:name,kind:kindMap[name],pyInput:entries[k0],acInput:entries[k1]});
}
Object.keys(entries).forEach(function(key){var pair=JSON.parse(key);if(pair[0]===ac&&metricOrder.indexOf(pair[1])<0)throw new Error('两期必须拥有相同指标；同期缺少：'+pair[1]);});
if(!rows.length) throw new Error('请提供至少一个财务指标。');
if(rows[rows.length-1].kind!=='subtotal') rows.push({name:setting.resultLabel,kind:'subtotal',pyInput:null,acInput:null,autoFinal:true});
function calcSteps(field){
  var running=0,steps=[];
  rows.forEach(function(row,i){
    var source=row[field+'Input'],before=running,after;
    if(row.kind==='subtotal'){
      after=running;
      if(source!==null && setting.strictSubtotal && !closeNum(source,after))
        throw new Error('小计数值不匹配：'+row.name+' / '+field+'，填入 '+source+'，自动计算 '+after+'。修改指标后，可把 subtotal 对应的度量值改为 null。');
      row[field]=source===null?after:source;
      steps.push({from:0,to:row[field],kind:'subtotal',index:i});
    } else {
      var change=(row.kind==='expense'?-source:source);
      running+=change;
      row[field]=source;
      steps.push({from:before,to:running,kind:row.kind,index:i});
    }
  });
  return steps;
}
var stepsPY=calcSteps('py'),stepsAC=calcSteps('ac');
rows.forEach(function(r,i){
  r.index=i;
  r.rawChange=r.ac-r.py;
  r.contribution=r.kind==='expense'?-r.rawChange:r.rawChange;
  r.percent=closeNum(r.py,0)?null:100*r.rawChange/Math.abs(r.py);
});
var final=rows[rows.length-1];
var contributions=rows.filter(function(r){return r.kind!=='subtotal';}).reduce(function(acc,r){return acc+r.contribution;},0);
if(!closeNum(contributions,final.ac-final.py))throw new Error('明细贡献与期末结果没有勾稽，请检查输入科目。');

// 图表主体顶部位置完全由 layout.top 控制
var top=L.top;
var bottom=Math.min(L.bottom,Math.max(66,Math.round(H*0.13)));
var plotHeight=H-top-bottom,rowHeight=plotHeight/rows.length;
var canLabel=rowHeight>=L.minRowLabelHeight;
var gap=compact?9:L.panelGap;
var startX=L.left+nameWidth;
var totalWidth=W-startX-L.right-gap*3;
var ratioSum=L.ratios.reduce(function(a,b){return a+b;},0);
var widths=L.ratios.map(function(r){return totalWidth*r/ratioSum;});
var lefts=[startX];for(var j=1;j<4;j++)lefts.push(lefts[j-1]+widths[j-1]+gap);
var domains=[];
var financeValues=[0];stepsPY.concat(stepsAC).forEach(function(s){financeValues.push(s.from,s.to);});
var financeMin=Math.min.apply(null,financeValues),financeMax=Math.max.apply(null,financeValues);
var financeSpread=Math.max(1,financeMax-financeMin);
var xFinancialMin=(financeMin<0?financeMin-financeSpread*0.07:0);
var xFinancialMax=financeMax+financeSpread*0.24;
var maxDelta=Math.max(1,Math.max.apply(null,rows.map(function(r){return Math.abs(display(r.contribution));})));
var maxPercent=Math.max(1,Math.max.apply(null,rows.map(function(r){return Math.abs(r.percent===null?0:r.percent);})));
domains=[{min:display(xFinancialMin),max:display(xFinancialMax)},
 {min:display(xFinancialMin),max:display(xFinancialMax)},
 {min:-maxDelta*1.20,max:maxDelta*1.85},
 {min:-maxPercent*1.20,max:maxPercent*2.30}];
function grid(left,width){return {left:Math.round(left),right:Math.round(W-left-width),top:Math.round(top),bottom:Math.round(bottom),containLabel:false};}
function xAxis(ix,domain){return {gridIndex:ix,type:'value',min:domain.min,max:domain.max,show:false,scale:true};}
function yAxis(ix,names){return {
  gridIndex:ix,type:'category',data:names,inverse:true,boundaryGap:true,
  axisTick:{show:false},axisLine:{show:false},splitLine:{show:false},
  axisLabel: ix===0?{
    show:true,interval:0,width:nameWidth-12,overflow:'truncate',align:'right',margin:16,
    color:C.text,fontSize:fontPx,fontFamily:setting.font,
    rich:{subtotal:{color:C.text,fontSize:fontPx,fontWeight:700,fontFamily:setting.font}},
    formatter:function(value,i){var r=rows[i];return r.kind==='subtotal'?'{subtotal|=  '+value+'}':(r.kind==='expense'?'−  ':'+  ')+value;}
  }:{show:false}
};}
function clip(value,min,max){return Math.max(min,Math.min(max,value));}
function textGraphic(x,y,str,size,bold,color,align,width){return {type:'text',silent:true,z:8,
  style:{x:x,y:y,text:String(str),font:(bold?'bold ':'')+size+'px '+setting.font,fill:color||C.text,
    align:align||'left',verticalAlign:'middle',width:width,overflow:'truncate'}};}
function lineGraphic(x1,y1,x2,y2,color,width){return {type:'line',silent:true,z:1,
  shape:{x1:x1,y1:y1,x2:x2,y2:y2},style:{stroke:color,lineWidth:width||1}};}
function backgroundHeader(){
  var g=[{type:'rect',silent:true,left:L.left,top:22,z:9,shape:{width:20,height:4},style:{fill:C.negative}}];
  g.push(textGraphic(L.left,L.titleTop,setting.title,W<850?16:20,true,C.text,'left',W-2*L.left));
  g.push(textGraphic(L.left,L.subtitleTop,setting.entity+' · '+py+' → '+ac+' · '+setting.subtitle+' · 单位：'+setting.displayUnit,
    compact?10:12,false,C.secondaryText,'left',W-2*L.left));
  g.push(lineGraphic(L.left,top-22,W-L.right,top-22,C.grid,1));
  g.push(textGraphic(L.left,H-48,setting.source,W<900?9:10,false,C.secondaryText,'left',W-L.left-L.right));
  g.push(textGraphic(L.left,H-29,setting.description,W<900?9:10,false,C.secondaryText,'left',W-L.left-L.right));
  return g;
}
function blankTooltipStyle(){return {
  trigger:'item',confine:true,show:setting.showTooltips,backgroundColor:C.white,borderColor:C.grid,borderWidth:1,
  padding:[10,12],textStyle:{color:C.text,fontSize:12},extraCssText:'box-shadow:0 4px 20px rgba(0,0,0,.10);'
};}
function tooltipFor(r){
  var tr=function(a,b){return '<tr><td style="padding:5px 14px 5px 0;color:#8493A2">'+html(a)+
    '</td><td style="padding:5px 0;text-align:right;font-weight:600">'+html(b)+'</td></tr>';};
  var color=effectColor(r.contribution);
  return '<div style="font-family:'+setting.font+';min-width:225px;color:'+C.text+';line-height:1.5">'+
    '<div style="font-size:14px;font-weight:bold;padding-bottom:7px">'+html(r.name)+'</div><table style="border-collapse:collapse;width:100%">'+
    tr(py+' (PY)',money(r.py)+' '+setting.displayUnit)+
    tr(ac+' (AC)',money(r.ac)+' '+setting.displayUnit)+
    tr('指标自身变化',signedMoney(r.rawChange)+' '+setting.displayUnit)+
    tr('同比变化率',signedPct(r.percent))+'</table><div style="border-top:1px solid '+C.grid+';margin-top:7px;padding-top:7px">'+
    '<span style="font-weight:bold;color:'+color+'">'+html(dir(r.contribution)+' '+signedMoney(r.contribution))+' '+html(setting.displayUnit)+'</span>'+
    '<span style="color:'+C.secondaryText+';margin-left:8px">对最终结果的贡献</span></div>'+
    '<div style="font-size:11px;color:'+C.secondaryText+';margin-top:5px">'+
    html(r.kind==='expense'?'费用金额增加对利润通常是不利的；此处对利润的贡献取费用变动的相反数。':
    r.kind==='subtotal'?'小计或合计由前面科目累计计算，不重复计入差异桥。':'收入/收益的变动按原值计入利润贡献。')+'</div></div>';
}
function shapeRect(x1,x2,cy,height,fill,lo,hi){
  var a=clip(Math.min(x1,x2),lo,hi),b=clip(Math.max(x1,x2),lo,hi);
  return {type:'rect',shape:{x:a,y:cy-height/2,width:Math.max(0.8,b-a),height:height},style:{fill:fill}};
}
function shapeText(str,x,y,align,color,fontSize,bold){return {type:'text',silent:true,
  style:{text:String(str),x:x,y:y,textAlign:align,textVerticalAlign:'middle',fill:color,
    font:(bold?'bold ':'')+fontSize+'px '+setting.font}};}
function tooltipHitbox(left,width,cy,height){return {type:'rect',shape:{x:left,y:cy-height/2,width:width,height:height},
  style:{fill:'rgba(255,255,255,0)'},z:5};}
function financePainter(steps,ix){
  return function(params,api){
    var i=params.dataIndex,r=rows[i],s=steps[i];
    var y=api.coord([0,i])[1],from=api.coord([display(s.from),i])[0],to=api.coord([display(s.to),i])[0];
    var plotL=lefts[ix]+2,plotR=lefts[ix]+widths[ix]-2;
    var height=Math.max(2.5,Math.min(L.maxBarHeight,rowHeight*L.barHeightRatio));
    var isTotal=r.kind==='subtotal',fill=isTotal?(ix===0?C.py:C.ac):(ix===0?C.barSecondary:C.bar);
    var children=[];
    if(setting.showConnectors&&i<rows.length-1&&!isTotal&&rows[i+1].kind!=='subtotal'){
      var end=clip(to,plotL,plotR),nextY=api.coord([0,i+1])[1];
      children.push({type:'line',shape:{x1:end,y1:y+height/2,x2:end,y2:nextY-height/2},
        style:{stroke:C.secondaryText,lineWidth:0.8,opacity:0.45}});
    }
    var rectFrom=isTotal?api.coord([0,i])[0]:from;
    children.push(shapeRect(rectFrom,to,y,height,fill,plotL,plotR));
    if(setting.showValueLabels&&canLabel){
      // 数值强制放在当前栏最右端的文字槽，任何指标数值均不进入相邻栏。
      var label=money(r[ix===0?'py':'ac']);
      var labelX=plotR-3;
      // 深色合计柱覆盖最右端文字槽时切成白字。
      var onDark=isTotal&&ix===1&&Math.max(rectFrom,to)>labelX-label.length*valuePx*0.65;
      children.push(shapeText(label,labelX,y,'right',onDark?C.white:C.text,valuePx,isTotal));
    }
    children.push(tooltipHitbox(plotL,plotR-plotL,y,Math.max(12,rowHeight*0.85)));
    return {type:'group',children:children};
  };
}
function variancePainter(ix,percent){
  return function(params,api){
    var i=params.dataIndex,r=rows[i],v=percent?r.percent:display(r.contribution);
    if(v===null)return {type:'group',children:[tooltipHitbox(lefts[ix],widths[ix],api.coord([0,i])[1],Math.max(12,rowHeight*0.8))]};
    var y=api.coord([0,i])[1],zero=api.coord([0,i])[0],end=api.coord([v,i])[0];
    var pL=lefts[ix]+2,pR=lefts[ix]+widths[ix]-2;
    var fill=effectColor(r.contribution),height=Math.max(3,Math.min(12,rowHeight*0.30));
    var children=[shapeRect(zero,end,y,height,fill,pL,pR)];
    var endClip=clip(end,pL,pR);
    children.push({type:'rect',shape:{x:endClip-1,y:y-height/2-2,width:2,height:height+4},style:{fill:fill}});
    if(canLabel&&(percent?setting.showPercentLabels:setting.showDeltaLabels)){
      var label=percent?(r.percent===null?'—':dir(r.percent)+' '+nfmt(Math.abs(r.percent),setting.percentDecimals)+'%'):
        dir(r.contribution)+' '+nfmt(Math.abs(display(r.contribution)),setting.decimals);
      children.push(shapeText(label,pR-2,y,'right',fill,valuePx,r.kind==='subtotal'));
    }
    children.push(tooltipHitbox(pL,pR-pL,y,Math.max(12,rowHeight*0.84)));
    return {type:'group',children:children};
  };
}
function makeCustom(name,ix,values,painter,tooltip) {return {
  name:name,type:'custom',coordinateSystem:'cartesian2d',xAxisIndex:ix,yAxisIndex:ix,
  encode:{x:0,y:1},
  data:values.map(function(v,i){return {value:[v,i],rowIndex:i};}),
  renderItem:painter,clip:true,z:3,animationDuration:setting.animationDuration,
  emphasis:{disabled:true},tooltip:{formatter:tooltip}
};}
