function validNumber(v) {
  if (typeof v === 'number') return isFinite(v) ? v : NaN;
  var x = String(v == null ? '' : v).replace(/[,，\s]/g,'').trim();
  return x === '' ? NaN : Number(x);
}
function htmlEscape(v) {
  return String(v == null ? '' : v).replace(/[&<>"']/g,function(x){
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[x];
  });
}
function rgba(hex,a) {
  var h=hex.replace('#','');
  return 'rgba('+parseInt(h.slice(0,2),16)+','+parseInt(h.slice(2,4),16)+','+parseInt(h.slice(4,6),16)+','+a+')';
}
function niceNum(v,d) { return Number(v).toLocaleString('zh-CN',{maximumFractionDigits:d == null ? 1 : d}); }
function comparePeriod(a,b) {
  var x=Number(a), y=Number(b);
  if (isFinite(x) && isFinite(y)) return x-y;
  return String(a).localeCompare(String(b),'zh-CN',{numeric:true});
}
function band(v) {
  return v >= B[1] ? 2 : v >= B[0] ? 1 : 0;
}
function bandLabel(b) { return ['低','中','高'][b]; }
function strategyIndex(strength, attractiveness) {
  var sum=band(strength)+band(attractiveness);
  return sum >= 3 ? 0 : sum === 2 ? 1 : 2;
}
function cellName(strength, attractiveness) {
  return '行业吸引力'+bandLabel(band(attractiveness))+' / 竞争实力'+bandLabel(band(strength));
}
function tooltipPosition(point, params, dom, rect, size) {
  var vw = size.viewSize[0], vh = size.viewSize[1];
  var cw = size.contentSize[0], ch = size.contentSize[1];
  var x = point[0] + 18, y = point[1] + 15;
  if (x + cw > vw - 10) x = point[0] - cw - 18;
  if (y + ch > vh - 20) y = point[1] - ch - 18;
  return [Math.max(10, Math.min(x, Math.max(10, vw - cw - 10))),
          Math.max(10, Math.min(y, Math.max(10, vh - ch - 10)))];
}

function bubbleDiameter(v) {
  var range = sizeMax - sizeMin;
  var frac = range > 0 ? Math.max(0,Math.min(1,(v-sizeMin)/range)) : 0.5;
  var low=setting.bubble.minDiameter, high=setting.bubble.maxDiameter;
  return Math.sqrt(low*low+frac*(high*high-low*low)); // 面积线性映射
}

// markArea 的文字与 scatter 标签是两套布局，单靠 labelLayout 不会相互避让。
// 对九个子区逐一比较「顶/底 × 左/中/右」六种注记位置，选择离散点及其标签较远的位置。
// 使用评分坐标做近似碰撞检测，保持与容器/画布无关，不依赖 myChart、DOM 或事件钩子。
function chooseCellLabelPosition(xi, yi, caption) {
  if (!setting.bubble.labelAvoidance || !points.length) return 'insideTop';
  var x0=edges[xi], x1=edges[xi+1], y0=edges[yi], y1=edges[yi+1];
  // 参考横纵坐标同为 0～100 的典型绘图区比例，将像素尺寸近似换算到评分空间。
  // ECharts 的 labelLayout.hideOverlap 还会依据实际像素再做一次气泡文字避让。
  var pxX=920/(setting.score.max-setting.score.min);
  var pxY=510/(setting.score.max-setting.score.min);
  var w=Math.max(15,Array.from(caption).length*11.5), h=15;
  var pad=7;
  var candidates=[
    {name:'insideTop',anchor:'center',x:(x0+x1)/2,y:y1-(pad+h/2)/pxY},
    {name:'insideTopLeft',anchor:'left',x:x0+pad/pxX,y:y1-(pad+h/2)/pxY},
    {name:'insideTopRight',anchor:'right',x:x1-pad/pxX,y:y1-(pad+h/2)/pxY},
    {name:'insideBottom',anchor:'center',x:(x0+x1)/2,y:y0+(pad+h/2)/pxY},
    {name:'insideBottomLeft',anchor:'left',x:x0+pad/pxX,y:y0+(pad+h/2)/pxY},
    {name:'insideBottomRight',anchor:'right',x:x1-pad/pxX,y:y0+(pad+h/2)/pxY}
  ];
  function intersects(a,b){
    var ow=Math.min(a[2],b[2])-Math.max(a[0],b[0]);
    var oh=Math.min(a[3],b[3])-Math.max(a[1],b[1]);
    return Math.max(0,ow)*Math.max(0,oh);
  }
  var best=candidates[0], bestPenalty=Infinity;
  candidates.forEach(function(c,index){
    var left=c.x-(c.anchor==='center'?w/2/pxX:(c.anchor==='right'?w/pxX:0));
    var r=[left,c.y-h/2/pxY,left+w/pxX,c.y+h/2/pxY];
    var penalty=index*0.03; // 无冲突时仍优先顶部居中，保持原版整齐的九格标注
    points.forEach(function(v){
      var radius=bubbleDiameter(v.size)/2;
      var dot=[v.s-radius/pxX,v.a-radius/pxY,v.s+radius/pxX,v.a+radius/pxY];
      penalty+=intersects(r,dot)*32;
      if(setting.bubble.showLabels && visibleNames[v.name]){
        var textW=Array.from(v.name).length*11+6;
        var nameLeft=v.s+(radius+7)/pxX;
        var nameBox=[nameLeft,v.a-9/pxY,nameLeft+textW/pxX,v.a+9/pxY];
        penalty+=intersects(r,nameBox)*42;
      }
    });
    if(penalty<bestPenalty){bestPenalty=penalty;best=c;}
  });
  return best.name;
}

// 辅助视觉编码：使用真实的 ECharts graphic 图形，而非字体符号「▪」，
// 图形与绘图区的历史方块/箭头同形同色；年份始终来自数据数组。
function visualLegendGraphic(){
  var children=[], x=0, cy=8;
  var ink=COLORS.secondary;
  function textWidth(s){
    return Array.from(String(s)).reduce(function(sum,ch){return sum+(ch.charCodeAt(0)>255?10.5:6.1);},0);
  }
  function addLabel(s,space){
    children.push({type:'text',silent:true,style:{x:x,y:cy,text:s,font:'10px sans-serif',fill:ink,verticalAlign:'middle'}});
    x+=textWidth(s)+(space==null?24:space);
  }
  GROUPS.forEach(function(g,i){
    children.push({type:'circle',silent:true,shape:{cx:x+4+i*7,cy:cy,r:3.3},style:{fill:g.color}});
  });
  x+=29;
  addLabel(currentPeriod+'年业务（颜色=策略、大小≈规模）',28);
  var hasMoves=setting.history.show && !!previousPeriod && points.some(function(v){return !!v.prev;});
  if(hasMoves){
    children.push({type:'rect',silent:true,shape:{x:x,y:cy-4,width:8,height:8},style:{fill:COLORS.history}});
    x+=17;
    addLabel(previousPeriod+'年位置',26);
    children.push({type:'line',silent:true,shape:{x1:x,y1:cy,x2:x+24,y2:cy},style:{stroke:COLORS.history,lineWidth:1.5}});
    if(setting.history.showArrow){
      children.push({type:'polygon',silent:true,shape:{points:[[x+26,cy],[x+19,cy-4],[x+19,cy+4]]},style:{fill:COLORS.history}});
    }
    x+=36;
    addLabel(previousPeriod+'—'+currentPeriod+'年迁移方向',0);
  }
  return {id:'c038-visual-legend',type:'group',left:'center',top:139,silent:true,children:children};
}
