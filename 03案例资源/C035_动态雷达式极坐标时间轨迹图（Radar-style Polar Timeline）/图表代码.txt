// ==================== 通用数据校验及处理 ====================
function c035ParseYear(value, row) {
  var n = Number(String(value).trim());
  if (!Number.isInteger(n) || n < 1 || n > 9999) {
    throw new Error('第' + (row + 1) + '行：年份应为有效的整数年份。');
  }
  return n;
}
function c035ParseMonth(value, row) {
  var m = String(value).trim().match(/^(\d{1,2})(?:月)?$/);
  var n = m ? Number(m[1]) : NaN;
  if (!Number.isInteger(n) || n < 1 || n > 12) {
    throw new Error('第' + (row + 1) + '行：月份必须为1—12，也接受“3月”。');
  }
  return n;
}
function c035ParseValue(value, row) {
  if (value == null || String(value).trim() === '') {
    throw new Error('第' + (row + 1) + '行：指标值为空，请检查Power BI字段粒度和聚合方式。');
  }
  var n = typeof value === 'number' ? value : Number(String(value).replace(/,/g, '').trim());
  if (!Number.isFinite(n)) throw new Error('第' + (row + 1) + '行：指标值不是有效数字。');
  return n;
}
function c035BuildRows(years, months, vals, config) {
  if (!Array.isArray(years) || !Array.isArray(months) || !Array.isArray(vals)) {
    throw new Error('原始输入必须是三条平行的一维数组（年份、月份、指标值）。');
  }
  if (!years.length || years.length !== months.length || years.length !== vals.length) {
    throw new Error('三个字段必须非空且长度一致（当前：' + [years.length, months.length, vals.length].join('/') + '）。');
  }
  var rows = years.map(function (y, i) {
    return {year:c035ParseYear(y,i), month:c035ParseMonth(months[i],i), value:c035ParseValue(vals[i],i)};
  });
  rows.sort(function(a,b){return a.year-b.year || a.month-b.month;});
  for (var i=1; i<rows.length; i++) {
    var a=rows[i-1], b=rows[i];
    var diff=(b.year-a.year)*12+(b.month-a.month);
    if (diff===0) throw new Error('重复的年份+月份：'+b.year+'-'+b.month+'；请检查Power BI数据粒度。');
    if (diff!==1) throw new Error('月份不连续：'+a.year+'-'+a.month+' 至 '+b.year+'-'+b.month+'。时间轨迹不能伪造缺失月份。');
  }
  var yearsUnique=[], groups=[];
  rows.forEach(function(r){
    if(!yearsUnique.length || yearsUnique[yearsUnique.length-1]!==r.year){
      yearsUnique.push(r.year); groups.push([]);
    }
    groups[groups.length-1].push(r);
  });
  if(yearsUnique.length>config.maxYears){
    throw new Error('当前'+yearsUnique.length+'个年份，超过setting.maxYears='+config.maxYears+'，请适当缩小范围。');
  }
  return {rows:rows, years:yearsUnique, groups:groups};
}
function c035Signed(n) { return (n>=0 ? '+' : '') + n.toFixed(2); }
function c035EscapeHtml(s) {
  return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')
    .replace(/"/g,'&quot;').replace(/'/g,'&#39;');
}
function c035Tooltip(p) {
  var v=p&&p.value;
  if(!Array.isArray(v)||v.length<4) return '';
  return '<div style="min-width:188px;font:12px Arial,sans-serif;color:'+setting.colors.text+'">'+
    '<div style="font-size:14px;font-weight:700;margin-bottom:8px">'+c035EscapeHtml(v[2])+'年'+c035EscapeHtml(v[3])+'月</div>'+
    '<div style="height:1px;background:'+setting.colors.grid+';margin-bottom:9px"></div>'+
    '<div style="display:flex;align-items:center;justify-content:space-between;gap:16px">'+
    '<span>'+c035EscapeHtml(setting.metricLabel)+'</span><b style="font-size:14px;color:'+setting.colors.text+'">'+
    c035Signed(Number(v[0]))+c035EscapeHtml(setting.unit)+'</b></div>'+
    '<div style="font-size:11px;margin-top:9px;color:'+setting.colors.secondaryText+'">基准：'+
    c035EscapeHtml(setting.basePeriodText)+'</div></div>';
}
function c035Point(r) {
  return [r.value, (r.month-1)*30, r.year, r.month];
}

// 根据完整月序列，以每年为一个“绘制段”：
// 段内从当年1月顺次到12月；若次年1月存在，把次年1月放在角度360°位置，
// 与下一段的0°为同一空间位置。这样不会产生12月(330°)→1月(0°)的跨圆直线。
// 这些相接的段构成按真实月份连续前进的雷达式极坐标轨迹，并允许逐年渐隐。
function c035BuildSegments(groups) {
  var segments=[];
  for(var i=0;i<groups.length;i++){
    var points=groups[i].map(function(r){return [r.value,(r.month-1)*30,r.year,r.month];});
    if(i+1<groups.length && groups[i+1][0].month===1){
      var next=groups[i+1][0];
      points.push([next.value,360,next.year,next.month]);
    }
    segments.push(points);
  }
  return segments;
}
// 自定义系列按年月序列生成一段多点折线；跨年衔接使用360°而非跳回0°。
function c035LineRenderer(segments, cfg, focus) {
  return function(params,api) {
    var index=Math.round(api.value(0));
    var current=Math.round(api.value(1));
    var segment=segments[index];
    // 正在播放的这一年不提前显示下一年1月；转入下一帧后才绘制跨年连接。
    if(focus && segment && segment.length && segment[segment.length-1][1]===360) {
      segment=segment.slice(0,-1);
    }
    if(!segment || segment.length<2) return null;
    var points=segment.map(function(p){return api.coord([p[0],p[1]]);});
    if(points.some(function(p){return !p || !Number.isFinite(p[0]) || !Number.isFinite(p[1]);}))return null;
    var age=current-index;
    var fadeYears=Math.max(1,Number(cfg.recentHistoryYears)||30);
    var minOpacity=Math.max(0,Math.min(0.2,Number(cfg.oldHistoryOpacity)));
    var maxOpacity=Math.max(minOpacity,Math.min(1,Number(cfg.recentHistoryOpacity)));
    var opacity=focus?1:(minOpacity+(maxOpacity-minOpacity)*Math.exp(-Math.max(0,age-1)/fadeYears));
    return {
      type:'polyline',
      shape:{points:points},
      style:{stroke:focus?cfg.colors.focus:(cfg.colors.historyLine || cfg.colors.history),fill:null,
        lineWidth:focus?cfg.focusLineWidth:cfg.historyLineWidth,
        opacity:opacity,lineCap:'round',lineJoin:'round'},
      silent:true
    };
  };
}

function c035BuildOption(source, config) {
  var all=source.rows, groups=source.groups, years=source.years;
  var n=years.length;
  var values=all.map(function(r){return r.value;});
  var min=Math.min.apply(null,values),max=Math.max.apply(null,values);
  var step=Number(config.ringStep);
  if(!(step>0)||!(config.axisMargin>=0))throw new Error('setting.ringStep须为正数、axisMargin不能为负数。');
  var radialMin=Math.floor((min-config.axisMargin)/step)*step;
  var radialMax=Math.ceil((max+config.axisMargin)/step)*step;
  if(radialMin===radialMax)radialMax+=step;
  var segments=c035BuildSegments(groups);
  var tickEvery=Math.max(1, Math.floor(Number(config.timelineTickEvery)||20));
  var edgeGap=Math.max(0, Math.floor(Number(config.timelineEdgeGap)||0));
  var allPoints=[];
  var timelineFrames=[];
  for(var i=0;i<n;i++) {
    var yearPts=groups[i].map(c035Point);
    Array.prototype.push.apply(allPoints,yearPts);
    var last=groups[i][groups[i].length-1];
    timelineFrames.push({
      graphic:c035Graphics(years[i],last,config),
      series:[
        {id:'c035-history-line',data:Array.from({length:i},function(_,j){return [j,i];})},
        {id:'c035-focus-line',data:[[i,i]]},
        {id:'c035-tooltip-points',data:allPoints.slice()},
        {id:'c035-head',data:[c035Point(last)]}
      ]
    });
  }
  var initialIndex=config.firstFrame==='latest'?n-1:0;
  var txtColors=config.colors;
  return {
    baseOption:{
      backgroundColor:config.colors.background,
      animation:false,
      title:{
        left:28,top:26,text:config.title,
        subtext:config.subtitle+'  ·  '+years[0]+'—'+years[n-1],itemGap:8,
        textStyle:{color:txtColors.text,fontSize:20,fontWeight:650},
        subtextStyle:{color:txtColors.secondaryText,fontSize:12}
      },
      graphic:c035Graphics(years[initialIndex],groups[initialIndex][groups[initialIndex].length-1],config),
      polar:{center:['50%','47%'],radius:['19%','68%']},
      angleAxis:{
        type:'value',min:0,max:360,interval:30,startAngle:90,clockwise:true,
        axisLine:{show:false},axisTick:{show:false},
        axisLabel:{color:txtColors.secondaryText,fontSize:11,margin:11,
          formatter:function(v){return v===360?'':String(Math.round(v/30)+1)+'月';}},
        splitLine:{show:true,lineStyle:{color:txtColors.grid,width:config.gridSpokeWidth,opacity:config.gridSpokeOpacity}}
      },
      radiusAxis:{
        type:'value',min:radialMin,max:radialMax,
        interval:step,
        axisLine:{show:false},axisTick:{show:false},
        axisLabel:{color:txtColors.secondaryText,fontSize:10,
          formatter:function(v){return(v>0?'+':'')+Number(v).toFixed(1)+config.unit;}},
        splitLine:{show:true,lineStyle:{color:txtColors.grid,width:config.gridRingWidth,opacity:config.gridRingOpacity,type:'dashed'}},
        splitArea:{show:false}
      },
      timeline:{
        axisType:'category',data:years.map(String),
        autoPlay:!!config.autoPlay,currentIndex:initialIndex,
        playInterval:config.playInterval,loop:!!config.loop,
        left:84,right:70,bottom:69,height:34,controlPosition:'left',
        symbol:'none',symbolSize:0,
        checkpointStyle:{symbol:'circle',symbolSize:11,color:txtColors.focus,
          borderColor:'#FFFFFF',borderWidth:2},
        lineStyle:{color:txtColors.grid,width:3},
        progress:{lineStyle:{color:txtColors.history,width:3}},
        controlStyle:{showPlayBtn:true,showPrevBtn:true,showNextBtn:true,itemSize:15,
          color:txtColors.text,borderColor:txtColors.secondaryText,borderWidth:1.5},
        label:{color:txtColors.secondaryText,fontSize:10,position:'bottom',
          formatter:function(text){
            var y=Number(text);
            if (y===years[0] || y===years[n-1]) return text;
            if (y%tickEvery !== 0) return '';
            if (Math.abs(y-years[0])<edgeGap || Math.abs(y-years[n-1])<edgeGap) return '';
            return text;
          }
        },
        emphasis:{label:{color:txtColors.text}}
      },
      tooltip:{
        show:true,trigger:'item',confine:true,formatter:c035Tooltip,
        backgroundColor:'#FFFFFF',borderColor:txtColors.grid,borderWidth:1,
        padding:[10,12],extraCssText:'box-shadow:0 3px 14px rgba(38,50,56,.07);border-radius:6px;'
      },
      series:[
        {id:'c035-history-line',type:'custom',coordinateSystem:'polar',
          name:'连续历史轨迹',data:[],renderItem:c035LineRenderer(segments,config,false),z:2,
          silent:true,animation:false},
        {id:'c035-focus-line',type:'custom',coordinateSystem:'polar',
          name:'当前年度轨迹',data:[],renderItem:c035LineRenderer(segments,config,true),z:5,
          silent:true,animation:false},
        {id:'c035-tooltip-points',type:'scatter',coordinateSystem:'polar',
          name:'逐月记录',data:[],encode:{radius:0,angle:1},symbol:'circle',symbolSize:8,z:7,
          itemStyle:{color:txtColors.history,opacity:0},
          emphasis:{itemStyle:{opacity:0.72,color:txtColors.focus}},
          animation:false},
        {id:'c035-head',type:'scatter',coordinateSystem:'polar',
          name:'最新月份',data:[],encode:{radius:0,angle:1},symbol:'circle',symbolSize:10,z:10,
          itemStyle:{color:txtColors.focus,borderColor:'#FFFFFF',borderWidth:2},animation:false}
      ]
    },
    options:timelineFrames
  };
}
