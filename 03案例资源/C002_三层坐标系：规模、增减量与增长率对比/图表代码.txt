// ─────────────────────────────────────────────────────────────
// 03 数据整理：按年份和品牌汇总，动态计算同比
// ─────────────────────────────────────────────────────────────
function readFields(source, fields) {
  const objects = source && source.dataObjects;
  if (!Array.isArray(objects)) throw new Error('缺少 dataArr.dataObjects，请检查数据源。');
  const years = objects[fields.year]?.values;
  const categories = objects[fields.category]?.values;
  const values = objects[fields.value]?.values;
  if (![years, categories, values].every(Array.isArray)) {
    throw new Error('年份、分类、度量的字段索引不正确，或 values 不是数组。');
  }
  if (years.length !== categories.length || years.length !== values.length) {
    throw new Error('三个字段的 values 行数不同，请检查 Power BI 数据组织。');
  }
  return years.map((year, i) => ({
    year: String(year ?? '').trim(),
    category: String(categories[i] ?? '').trim(),
    value: values[i] === null || values[i] === '' || values[i] === undefined ? null : Number(values[i])
  })).filter(row => row.year && row.category && row.value !== null && Number.isFinite(row.value));
}

function prepareData(rows, config) {
  const allYears = [...new Set(rows.map(row => row.year))]
    .sort((a, b) => Number(a) - Number(b) || a.localeCompare(b, 'zh-CN'));
  const selected = config.compareYears ? config.compareYears.map(String) : allYears.slice(-2);
  if (selected.length !== 2 || selected[0] === selected[1]) {
    throw new Error('图表必须有两个不同的比较年份，请在 Power BI 中筛选或修改 compareYears。');
  }
  const [previousYear, currentYear] = selected;
  // 汇总相同（年份、分类）的多行记录；缺失某年份不自动按 0 处理。
  const totals = new Map();
  const inputOrder = [];
  rows.filter(row => selected.includes(row.year)).forEach(row => {
    if (!inputOrder.includes(row.category)) inputOrder.push(row.category);
    const key = JSON.stringify([row.year, row.category]);
    totals.set(key, (totals.get(key) ?? 0) + row.value);
  });
  const valueAt = (year, category) => totals.get(JSON.stringify([year, category])) ?? null;
  const result = inputOrder.map(category => {
    const previous = valueAt(previousYear, category);
    const current = valueAt(currentYear, category);
    const difference = previous === null || current === null ? null : current - previous;
    const rate = difference === null || previous === 0 ? null : difference / previous * 100;
    return { category, previous, current, difference, rate };
  });
  if (config.sort === 'current-desc') result.sort((a,b) => (b.current ?? -Infinity) - (a.current ?? -Infinity));
  if (config.sort === 'previous-desc') result.sort((a,b) => (b.previous ?? -Infinity) - (a.previous ?? -Infinity));
  if (config.sort === 'name') result.sort((a,b) => a.category.localeCompare(b.category,'zh-CN'));
  return { previousYear, currentYear, records: result };
}

// ─────────────────────────────────────────────────────────────
// 04 绘图配置：三个 Grid 共用类别顺序和左右边界
// ─────────────────────────────────────────────────────────────
function buildOption(model, config) {
  const { previousYear, currentYear, records } = model;
  const { colors: c, layout: l } = config;
  const categories = records.map(row => row.category);
  const absoluteRates = records.map(row => Math.abs(row.rate ?? 0));
  const maxRate = Math.max(1, ...absoluteRates);
  const maxDelta = Math.max(1, ...records.map(row => Math.abs(row.difference ?? 0)));
  const maxValue = Math.max(1, ...records.flatMap(row => [row.previous ?? 0, row.current ?? 0]));
  const digits = config.decimals;
  const format = (value, count = 0) => value === null ? '—' : Number(value).toLocaleString('zh-CN', { minimumFractionDigits: count, maximumFractionDigits: count });
  const signed = (value, count) => value === null ? '—' : `${value > 0 ? '+' : ''}${format(value, count)}`;
  const colorFor = value => value === null || value === 0 ? c.neutral : value > 0 ? c.positive : c.negative;
  // 依据开关动态分配区域。仅启用的坐标系参与布局、索引及联动。
  // 布局始终从上到下：增长率（可选）→增减量（可选）→年度出货量（必选）。
  const sections = [
    ...(config.showGrowthRate ? ['rate'] : []),
    ...(config.showDifference ? ['difference'] : []),
    'value'
  ];
  const positions = {
    rate:       { top: 21, height: 12 },
    difference: { top: 43, height: 15 },
    value:      { top: 68 }
  };
  if (sections.length === 2) {
    positions[sections[0]] = { top: 26, height: 22 };
    positions.value = { top: 59 };
  } else if (sections.length === 1) {
    positions.value = { top: 28 };
  }
  const indexOf = type => sections.indexOf(type);
  const sectionTitle = (text, type) => ({
    text, left: l.left, top: `${positions[type].top - 5}%`,
    textStyle: { color: c.muted, fontSize: 12, fontWeight: 'normal' }
  });
  const xAxis = (type, gridIndex) => ({
    type: 'category', gridIndex, data: categories, boundaryGap: true,
    axisLine: { show: type === 'value', lineStyle: { color: c.axis } },
    axisTick: { show: false },
    axisLabel: { show: type === 'value', interval: 0, margin: 13,
      color: c.muted, fontSize: l.categoryLabelSize }
  });
  const yAxis = (type, gridIndex) => {
    const bounds = type === 'rate' ? [0, 1] :
      type === 'difference' ? [-maxDelta * 1.5, maxDelta * 1.5] : [0, maxValue * 1.22];
    return { type: 'value', gridIndex, min: bounds[0], max: bounds[1],
      axisLine: { show: false }, axisTick: { show: false },
      axisLabel: { show: false }, splitLine: { show: false } };
  };
  const yearlyBar = (year, key, color) => ({
    name: `${year}年`, type: 'bar', xAxisIndex: indexOf('value'), yAxisIndex: indexOf('value'),
    data: records.map(row => row[key]), barWidth: l.valueBarWidth, barGap: '24%',
    itemStyle: { color },
    label: { show: true, position: 'top', distance: 5, fontSize: l.valueLabelSize, color: c.title,
      formatter: params => format(params.value, digits.metric) },
    emphasis: { disabled: true }
  });

  const series = [];
  if (config.showGrowthRate) series.push({
    name: '同比增长率', type: 'scatter', xAxisIndex: indexOf('rate'), yAxisIndex: indexOf('rate'),
    symbol: 'circle',
    // 直径随增长率绝对值的平方根变化，让气泡面积体现同比幅度。
    symbolSize: value => value[2] === null ? 0 :
      Math.max(l.bubbleMinDiameter, Math.sqrt(Math.abs(value[2]) / maxRate) * l.bubbleMaxDiameter),
    data: records.map((row, i) => ({
      value: [i, 0.5, row.rate], itemStyle: { color: colorFor(row.rate), opacity: 0.9 }
    })),
    label: { show: true, position: 'bottom', distance: 8, color: c.text,
      fontSize: l.rateLabelSize, fontWeight: 'bold',
      formatter: params => params.value[2] === null ? '—' : `${signed(params.value[2], digits.rate)}%` },
    emphasis: { disabled: true }
  });
  if (config.showDifference) series.push({
    name: '同比增减量', type: 'bar', xAxisIndex: indexOf('difference'), yAxisIndex: indexOf('difference'),
    barWidth: l.differenceBarWidth,
    data: records.map(row => ({
      value: row.difference,
      itemStyle: { color: colorFor(row.difference), borderRadius: (row.difference ?? 0) >= 0 ? [3,3,0,0] : [0,0,3,3] },
      label: { position: (row.difference ?? 0) >= 0 ? 'top' : 'bottom', color: colorFor(row.difference) }
    })),
    label: { show: true, distance: 6, fontWeight: 'bold',
      formatter: params => signed(params.value, digits.difference) },
    markLine: { silent: true, symbol: ['none','none'], label: { show: false },
      lineStyle: { color: c.baseline, type: 'dashed' }, data: [{ yAxis: 0 }] },
    emphasis: { disabled: true }
  });
  series.push(yearlyBar(previousYear, 'previous', c.previous), yearlyBar(currentYear, 'current', c.current));
  const headings = {
    rate: '同比增长率（%） · 圆面积表示变化幅度',
    difference: `同比增减量（${config.unit}） · ${currentYear} − ${previousYear}`,
    value: `${config.metricName}（${config.unit}）`
  };
  return {
    backgroundColor: c.background,
    animation: false,
    title: [
      { text: config.title, subtext: `${config.subtitle} · ${previousYear} vs ${currentYear}`,
        left: l.titleLeft, top: l.titleTop,
        textStyle: { color: c.title, fontSize: 21, fontWeight: 'bold' },
        subtextStyle: { color: c.muted, fontSize: 12 } },
      ...sections.map(type => sectionTitle(headings[type], type))
    ],
    // graphic 用于不依赖数据坐标的装饰与图表来源说明。
    graphic: [
      {
        type: 'rect', silent: true, z: 100,
        left: l.markerLeft, top: l.markerTop,
        shape: { width: l.markerWidth, height: l.markerHeight },
        style: { fill: c.titleMarker }
      },
      {
        type: 'text', silent: true, z: 100,
        left: l.footerLeft, bottom: l.footerBottom,
        style: {
          text: `数据来源：${config.source}\n说明：${config.sourceNote}`,
          fill: c.footer, font: `${l.footerFontSize}px sans-serif`,
          lineHeight: l.footerFontSize + 5, textAlign: 'left'
        }
      }
    ],
    legend: { data: [`${previousYear}年`, `${currentYear}年`], right: '5%', top: '12%',
      itemWidth: 12, itemHeight: 10, textStyle: { color: c.muted, fontSize: 11 } },
    tooltip: {
      trigger: 'axis', confine: true, axisPointer: { type: 'shadow' },
      formatter(params) {
        if (!params.length) return '';
        const row = records[params[0].dataIndex];
        return [
          `<strong>${row.category}</strong>`,
          `${previousYear}年：${format(row.previous, digits.metric)} ${config.unit}`,
          `${currentYear}年：${format(row.current, digits.metric)} ${config.unit}`,
          ...(config.showDifference ? [`同比增减量：${signed(row.difference, digits.difference)} ${config.unit}`] : []),
          ...(config.showGrowthRate ? [`同比增长率：${row.rate === null ? '—（上年为0或数据缺失）' : signed(row.rate, digits.rate) + '%'}`] : [])
        ].join('<br>');
      }
    },
    axisPointer: { link: [{ xAxisIndex: sections.map((_, i) => i) }] },
    // 可选模块保持百分比高度；底部年度柱状图使用 gridBottom 像素边距。
    grid: sections.map(type => ({
      left: l.left, right: l.right,
      top: `${positions[type].top}%`,
      ...(positions[type].height != null
        ? { height: `${positions[type].height}%` }
        : { bottom: l.gridBottom })
    })),
    xAxis: sections.map((type, i) => xAxis(type, i)),
    yAxis: sections.map((type, i) => yAxis(type, i)),
    series
  };
}

// ECharts 官方编辑器识别全局 option；Power BI 同样使用此配置。
const chartData = prepareData(readFields(chartSource, settings.fields), settings);
option = buildOption(chartData, settings);
