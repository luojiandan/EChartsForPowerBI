// ===== 03 计算与数据校验：无业务特定常量 =====
function c026S(v){return v==null?'':String(v).trim();}
function c026Esc(v){return c026S(v).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;').replace(/'/g,'&#39;');}
function c026Clamp(v,a,b){return Math.max(a,Math.min(b,v));}
function c026Num(v){return Number(v||0).toLocaleString('zh-CN');}
function c026Fail(m){throw Error('C026 输入错误：'+m);}
function c026Label(n){return n.kind==='group'?n.name:(setting.productAliases[n.sku]||n.name);}
function c026BuildData(){
  var arrays=[客户群,商品编号,商品名称,购买客户数,群体客户数];
  if(arrays.some(function(x){return !Array.isArray(x);}))c026Fail('5个输入字段必须都是一维数组。');
  var length=arrays[0].length;
  if(!length||arrays.some(function(x){return x.length!==length;}))c026Fail('5个数组必须非空且长度一致。');
  var groups=new Map(),goods=new Map(),links=[],seen=new Set();
  for(var i=0;i<length;i++){
    var g=c026S(客户群[i]),sku=c026S(商品编号[i]),name=c026S(商品名称[i]);
    var weight=Number(购买客户数[i]),base=Number(群体客户数[i]);
    if(!g||!sku||!name)c026Fail('第'+(i+1)+'条存在空的客户群/商品编号/商品名称。');
    if(!Number.isInteger(weight)||weight<=0)c026Fail('第'+(i+1)+'条购买客户数需为正整数。');
    if(!Number.isInteger(base)||base<=0||base<weight)c026Fail('第'+(i+1)+'条群体人数无效。');
    var k=JSON.stringify([g,sku]);if(seen.has(k))c026Fail('第'+(i+1)+'条客户群×商品重复：'+k);seen.add(k);
    if(!groups.has(g))groups.set(g,{key:'g:'+g,name:g,kind:'group',size:base,edges:[],sum:0});
    var group=groups.get(g);if(group.size!==base)c026Fail('客户群 '+g+' 在不同记录中的群体客户数不一致。');
    if(!goods.has(sku))goods.set(sku,{key:'p:'+sku,sku:sku,name:name,kind:'product',edges:[],sum:0});
    var product=goods.get(sku);if(product.name!==name)c026Fail('商品编号 '+sku+' 对应多个不同商品名称，请在上游统一。');
    var row={id:'e:'+i,g:group.key,p:product.key,group:g,sku:sku,weight:weight};
    links.push(row);group.edges.push(row);product.edges.push(row);group.sum+=weight;product.sum+=weight;
  }
  var gg=[...groups.values()],pp=[...goods.values()];
  var color=new Map(gg.map(function(g,i){return[g.name,setting.colors[i%setting.colors.length]];}));
  pp.forEach(function(p){p.dominant=p.edges.slice().sort(function(a,b){return b.weight-a.weight||gg.findIndex(x=>x.name===a.group)-gg.findIndex(x=>x.name===b.group);})[0].group;});
  var order=new Map(gg.map(function(g,i){return[g.name,i];}));
  pp.sort(function(a,b){return order.get(a.dominant)-order.get(b.dominant)||b.sum-a.sum||a.sku.localeCompare(b.sku,'zh-CN',{numeric:true});});
  var maxW=Math.max.apply(null,links.map(function(x){return x.weight;}));
  return {groups:gg,products:pp,links:links,color:color,maxW:maxW,groupMap:groups,productMap:goods, shared3:pp.filter(function(p){return p.edges.length>=3;}).length};
}

// ===== 04 固定环形布局：按群组主偏好排序的商品，内部为数据驱动贝塞尔曲线 =====
function c026BuildModel(){
  var W=c026Dims().w,H=c026Dims().h;
  var panel=c026Clamp(setting.panelWidth,226,Math.max(226,W*.34));
  var right=W-panel-22, plotW=right-28;
  var top=158, footer=65, availableH=H-top-footer;
  var R=Math.max(65,Math.min(plotW/2-44,availableH/2-24));
  R*=setting.ringScale;
  var cx=28+plotW/2,cy=top+availableH/2;
  var gArr=c026Data.groups,pArr=c026Data.products,perGroup=new Map();
  gArr.forEach(function(g){perGroup.set(g.name,pArr.filter(function(p){return p.dominant===g.name;}));});
  var nodes=[],nodeMap=new Map(),nG=gArr.length;
  gArr.forEach(function(g,i){
    var start=-Math.PI/2+i*(2*Math.PI/nG),end=-Math.PI/2+(i+1)*(2*Math.PI/nG);
    var seg=perGroup.get(g.name),n=seg.length;
    // 顶部首部区域给组节点留出位置；商品排到同一个扇区内，体现客户群分组
    var a=(start+end)/2;
    var center={key:g.key,kind:'group',name:g.name,entity:g,theta:a,x:Math.cos(a),y:Math.sin(a),size:Math.min(27,15+Math.sqrt(g.size)*.45),color:c026Color(g.name)};
    nodes.push(center);nodeMap.set(center.key,center);
    seg.forEach(function(p,j){
      var slot=(j+.5)/n;
      var angle=start+.065+(end-start-.13)*slot;
      // 避开组节点的中心角，避免同一圆周的节点重叠
      if(Math.abs(angle-a)<.070)angle+=(angle<a?-.076:.076);
      var nP={key:p.key,kind:'product',name:p.name,sku:p.sku,entity:p,theta:angle,x:Math.cos(angle),y:Math.sin(angle),size:c026Clamp(4.1+Math.sqrt(p.sum)*.95,6,13),color:c026Color(p.dominant)};
      nodes.push(nP);nodeMap.set(nP.key,nP);
    });
  });
  // 每次生成模型都从原始汇总重算商品排名；不依赖上一次绘制的残留状态。
  var ranked=c026Data.products.slice().sort(function(a,b){return b.sum-a.sum||String(a.sku).localeCompare(String(b.sku));});
  ranked.forEach(function(p,i){p.rank=i;});
  // 先锁定4个客户群标签，再根据实时缩放级别选择更多商品标签。
  // 标签在当前屏幕坐标系内重新计算防碰撞，且不会把已被缩出画布的节点强行标注。
  var zoom=c026View.zoom, thresholds=setting.zoomLabelThresholds;
  var tier=0;
  for(var ti=0;ti<thresholds.length;ti++)if(zoom>=thresholds[ti])tier=ti+1;
  var globalRankLimit=tier===0?setting.showTopLabels:Infinity;
  var perGroupLimit=setting.zoomLabelLimits[tier]||99;
  var textLength=setting.zoomLabelLengths[tier]||18;
  var visualRadius=R*.95/1.10;
  var occupied=[];
  function labelBox(n){
    var angle=n.theta,dx=Math.cos(angle),dy=-Math.sin(angle),grp=n.kind==='group';
    var px=cx+visualRadius*dx*zoom+c026View.panX;
    var py=cy+visualRadius*dy*zoom+c026View.panY;
    var txt=c026Human(c026Label(n),grp?12:textLength),size=grp?12:10;
    n.displayLabel=txt;
    var units=0;
    for(var j=0;j<txt.length;j++)units+=txt.charCodeAt(j)>255?1:.60;
    var w=units*size+5,h=size+5,off=n.size/2+8;
    var side=(dx>0.22?'right':dx<-.22?'left':dy<0?'top':'bottom');
    var x=px,y=py;
    if(side==='right'){x=px+off;y=py-h/2;}
    else if(side==='left'){x=px-off-w;y=py-h/2;}
    else if(side==='top'){x=px-w/2;y=py-off-h;}
    else{x=px-w/2;y=py+off;}
    n.labelSide=side;
    return {left:x-3,right:x+w+3,top:y-3,bottom:y+h+3};
  }
  function overlaps(a,b){return a.left<b.right && b.left<a.right && a.top<b.bottom && b.top<a.bottom;}
  function visible(b){return b.left>=19 && b.right<=right-5 && b.top>=171 && b.bottom<=H-78;}
  nodes.filter(function(n){return n.kind==='group';}).forEach(function(n){
    n.labelVisible=true;var b=labelBox(n);if(visible(b))occupied.push(b);
  });
  gArr.forEach(function(g){
    var count=0,groupNode=nodeMap.get(g.key);
    var candidates=nodes.filter(function(n){return n.kind==='product' && n.entity.dominant===g.name;});
    candidates.sort(function(a,b){return b.entity.sum-a.entity.sum || String(a.sku).localeCompare(String(b.sku));});
    candidates.forEach(function(n){
      n.labelVisible=false;
      if(count>=perGroupLimit || n.entity.rank>=globalRankLimit)return;
      if(tier===0){
        var distance=Math.abs(Math.atan2(Math.sin(n.theta-groupNode.theta),Math.cos(n.theta-groupNode.theta)));
        if(distance<setting.labelMinAngle)return;
      }
      var box=labelBox(n);
      if(!visible(box) || occupied.some(function(b){return overlaps(box,b);}))return;
      occupied.push(box);n.labelVisible=true;count++;
    });
  });
  var focused=c026FocusKey();
  return {W:W,H:H,panel:panel,right:right,cx:cx,cy:cy,R:R,nodes:nodes,nodeMap:nodeMap,focus:focused,links:c026Data.links,range:{left:cx-R,top:cy-R,right:W-(cx+R),bottom:H-(cy+R)}};
}

function c026ZoomAt(factor,x,y){
  var old=c026View.zoom;
  var next=c026Clamp(old*factor,setting.zoomMin,setting.zoomMax);
  if(Math.abs(next-old)<.000001)return;
  var k=next/old,cx=c026Model.cx,cy=c026Model.cy;
  if(!Number.isFinite(x))x=cx;
  if(!Number.isFinite(y))y=cy;
  c026View.panX=(x-cx)*(1-k)+c026View.panX*k;
  c026View.panY=(y-cy)*(1-k)+c026View.panY*k;
  c026View.zoom=next;c026ClampPan();
  c026Redraw();
}
