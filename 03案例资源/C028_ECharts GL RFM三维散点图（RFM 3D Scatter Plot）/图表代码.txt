function parseMetric(value) {
  if (value === null || value === undefined || String(value).trim() === '') return null;
  var result = Number(String(value).replace(/[,，\s]/g,''));
  return isFinite(result) && result >= 0 ? result : null;
}
function esc(value) {
  return String(value).replace(/&/g,'&amp;').replace(/</g,'&lt;')
    .replace(/>/g,'&gt;').replace(/"/g,'&quot;').replace(/'/g,'&#39;');
}
function fmt(value, digits) {
  var x = Number(value);
  return isFinite(x) ? x.toLocaleString('zh-CN',{
    minimumFractionDigits: digits || 0,
    maximumFractionDigits: digits || 0
  }) : '—';
}
var byId = Object.create(null), customers = [], invalidRows = 0, duplicateRows = 0;
for (var i=0; i<customerIds.length; i++) {
  var id = customerIds[i] == null ? '' : String(customerIds[i]).trim();
  var r = parseMetric(recencyDays[i]);
  var f = parseMetric(frequencyValues[i]);
  var m = parseMetric(monetaryValues[i]);
  if (!id || r === null || f === null || m === null) { invalidRows++; continue; }
  var key = '$'+id;
  if (Object.prototype.hasOwnProperty.call(byId,key)) {
    duplicateRows++;
    var found=customers[byId[key]];
    // 客户级记录重复可能来自 BI 表连接，避免将汇总 F/M 再次累加。
    found.r=Math.min(found.r,r);
    found.f=Math.max(found.f,f);
    found.m=Math.max(found.m,m);
  } else {
    byId[key]=customers.length;
    customers.push({id:id,r:r,f:f,m:m});
  }
}
// 采用并列平均秩：R的原始天数越少评分越高，F/M 原始值越多评分越高。
// 全部相同的字段给予所有客户 50 分；样本量 1 时亦如此。
function percentRank(list, prop, reverse) {
  var result=new Array(list.length), n=list.length;
  if (!n) return result;
  if (n===1) { result[0]=50; return result; }
  var sorted=list.map(function(v,index){return {index:index,value:v[prop]};});
  sorted.sort(function(a,b){return a.value-b.value;});
  for(var p=0;p<n;) {
    var q=p+1;
    while(q<n && sorted[q].value===sorted[p].value) q++;
    var score=100*(p+q-1)/(2*(n-1));
    if (reverse) score=100-score;
    for(var j=p;j<q;j++) result[sorted[j].index]=score;
    p=q;
  }
  return result;
}

function pointSize(score){
  var t=Math.max(0,Math.min(1,score/100));
  // 圆点的面积随 M 分位值增长（直径按开方映射）；同尺寸不代表相同金额。
  return setting.display.minPointSize+(setting.display.maxPointSize-setting.display.minPointSize)*Math.sqrt(t);
}
function legendBubbleRadius(count){
  if (!count) return 2.3;
  var ratio=Math.sqrt(count/maxGroupCount);
  return setting.display.minLegendBubbleRadius+
    (setting.display.maxLegendBubbleRadius-setting.display.minLegendBubbleRadius)*ratio;
}

// ========================= 05. ECharts GL：八区外壳和中心剖面 =========================
// 24个分区外表面 + 12个中心剖面小面片，让每区边界始终可辨识。
// 与 V4 的24面/0.031透明度不同：单面0.072且绘制0.29透明度的边框。

// surface 不参加 legend 筛选；它是坐标背景，点击客户群只筛选 scatter3D。
function surfaceOnPlane(name,fill,bounds,axis,at,opacity,edgeOpacity){
  var dirs=['x','y','z'],free=dirs.filter(function(d){return d!==axis;});
  function coord(d,u,v){
    if(d===axis)return at;
    var t=d===free[0]?u:v;
    return bounds[d][0]+t*(bounds[d][1]-bounds[d][0]);
  }
  return {
    id:name,name:name,type:'surface',coordinateSystem:'cartesian3D',
    silent:true,animation:false,parametric:true,shading:'color',
    tooltip:{show:false},
    itemStyle:{color:fill,opacity:opacity},
    wireframe:{show:setting.display.showZoneEdges,
      lineStyle:{color:'#A8B6C2',opacity:edgeOpacity,width:0.8}},
    parametricEquation:{
      u:{min:0,max:1,step:1},v:{min:0,max:1,step:1},
      x:function(u,v){return coord('x',u,v);},
      y:function(u,v){return coord('y',u,v);},
      z:function(u,v){return coord('z',u,v);}
    }
  };
}

      ['x','y','z'].forEach(function(axis,j){

          bits[j]?hi:lo,setting.display.zoneOpacity,setting.display.zoneEdgeOpacity));
      });

  if(setting.display.showThresholdPlanes){
    ['x','y','z'].forEach(function(axis){

          '#BCC7CF',bounds,axis,mid,
          setting.display.thresholdOpacity,setting.display.thresholdEdgeOpacity));
      }
    });
  }
  return s;
}

function findChartFromEvent(element,evt){
  if(typeof myChart!=='undefined'&&myChart&&typeof myChart.dispatchAction==='function')return myChart;
  if(typeof chart!=='undefined'&&chart&&typeof chart.dispatchAction==='function')return chart;
  if(typeof echartsInstance!=='undefined'&&echartsInstance&&typeof echartsInstance.dispatchAction==='function')return echartsInstance;
  if(typeof echarts==='undefined'||!echarts||typeof echarts.getInstanceByDom!=='function')return null;
  var cursor=element||(evt&&evt.target),zr=null;
  for(var n=0;cursor&&n<12;n++,cursor=cursor.parent){
    zr=cursor.__zr||cursor._zr;
    if(zr)break;
  }
  if(!zr||!zr.painter||typeof zr.painter.getViewportRoot!=='function')return null;
  var node=zr.painter.getViewportRoot();
  for(var j=0;node&&j<12;j++,node=node.parentNode){
    var instance=echarts.getInstanceByDom(node);
    if(instance&&typeof instance.dispatchAction==='function')return instance;
  }
  return null;
}

    });
    updateZrStyle(item.childAt(5),{fill:visible?z.color:'#A8B3BE'});
    updateZrStyle(item.childAt(6),{
      text:visible?'显示中':'已隐藏',
      fill:visible?setting.colors.muted:'#8995A1'
    });
  }
}
function onZoneClick(index,element,evt){
  var target=findChartFromEvent(element,evt);
  if(!target)return;
  target.dispatchAction({type:'legendToggleSelect',name:ZONES[index].cn});
  zoneVisible[index]=!zoneVisible[index];
  renderLegendVisibility(element);
  if(typeof target.getZr==='function'){
    var zr=target.getZr();
    if(zr&&typeof zr.refresh==='function')zr.refresh();
  }
}
function togglePanel(element,evt){
  // 可靠折叠与是否能获取echarts实例无关：直接在ZRender对象树中设置 ignore。
  // Button -> state group -> root group；root下固定 [anchor, body, tab] 顺序。
  var stateGroup=element&&element.parent;
  var root=stateGroup&&stateGroup.parent;
  if(!root||typeof root.childAt!=='function')return;
  var body=root.childAt(1),tab=root.childAt(2);
  if(!body||!tab)return;
  panelFolded=!panelFolded;
  body.ignore=panelFolded;
  tab.ignore=!panelFolded;
  if(typeof body.dirty==='function')body.dirty();
  if(typeof tab.dirty==='function')tab.dirty();
  var chartInstance=findChartFromEvent(element,evt);
  if(chartInstance&&typeof chartInstance.getZr==='function'){
    chartInstance.getZr().refresh();
  }else{
    // 没有宿主实例也能切换对象 ignore；事件自身刷新ZRender渲染树。
    var zr=element.__zr||stateGroup.__zr||root.__zr;
    if(zr&&typeof zr.refresh==='function')zr.refresh();
  }
}

// 3D 三轴在远端公共顶点可能同时绘制多个“0”，透视角度下形成“000”。
// 仅保留 F 轴的 0 标签；R 和 M 轴仍保留 25/50/75/100，其原点由 F 轴 0 表达。
// 该处理只影响刻度文字，不影响 R/F/M 数据、轴范围或网格/分区坐标。
function axis3D(name, hideZeroLabel){
  return {
    type:'value',name:name,min:setting.score.min,max:setting.score.max,
    interval:setting.score.interval,
    nameTextStyle:{color:setting.colors.text,fontSize:11},
    axisLine:{show:true,lineStyle:{color:setting.colors.muted,width:1}},
    axisLabel:{
      show:true,
      formatter:function(value){
        return hideZeroLabel && Math.abs(Number(value)-setting.score.min)<1e-9
          ? '' : String(value);
      },
      textStyle:{color:setting.colors.muted,fontSize:10}
    },
    axisTick:{show:false},
    splitLine:{show:true,lineStyle:{color:setting.colors.grid,width:1}},
    splitArea:{show:false}
  };
}
